cmake_minimum_required(VERSION 3.20)
project(foundation_followthrough C)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private reference root")
set(FF_FOUNDATION_SOURCE "" CACHE FILEPATH "Explicit completed foundation depth.c")
set(FF_FOUNDATION_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../foundation_depth/include" CACHE PATH "Public provider headers")
option(FF_LEARNER "Build unsolved companion with explicit completed provider" OFF)
option(CMASTERY_SANITIZE "Address/undefined sanitizers" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/foundation_followthrough/follow.c")
    set(private "${CMASTERY_ASSESSOR_ROOT}/foundation_followthrough/follow.c")
    if(NOT FF_FOUNDATION_SOURCE)
      set(FF_FOUNDATION_SOURCE "${CMASTERY_ASSESSOR_ROOT}/foundation_depth/depth.c")
    endif()
  elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/follow.c")
    set(private "${CMASTERY_ASSESSOR_ROOT}/follow.c")
  else()
    message(FATAL_ERROR "Supply explicit completed companion and foundation provider")
  endif()
endif()
if(NOT EXISTS "${FF_FOUNDATION_SOURCE}")
  if(CMASTERY_ASSESSOR_ROOT)
    message(FATAL_ERROR "Supply explicit completed companion and foundation provider")
  endif()
  set(FF_FOUNDATION_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../foundation_depth/src/depth.c")
endif()
if(NOT private OR FF_LEARNER)
  set(private "${CMAKE_CURRENT_LIST_DIR}/follow.c")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_executable(ff-tests tests.c "${private}" "${FF_FOUNDATION_SOURCE}")
add_executable(ff-history history_driver.c "${FF_FOUNDATION_SOURCE}")
foreach(t ff-tests ff-history)
  target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${FF_FOUNDATION_INCLUDE}")
endforeach()
foreach(case interval copy codec partition lifecycle)
  add_test(NAME foundation_followthrough.${case} COMMAND ff-tests ${case})
endforeach()
