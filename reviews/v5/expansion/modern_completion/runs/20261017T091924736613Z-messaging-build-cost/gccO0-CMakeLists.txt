cmake_minimum_required(VERSION 3.20)
project(messaging_build_cost C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(measure cost_driver.c cost_reference.c registration.c weak.c stream.c owned.c operation.c supervisor.c messaging.c)
target_include_directories(measure PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(measure PRIVATE -O0 -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(measure PRIVATE Threads::Threads)
