cmake_minimum_required(VERSION 3.16)
project(network_http_variants C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NV_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/variant.c" CACHE FILEPATH "Explicit learner/reference implementation")
set(NV_OWNED_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/owned.c" CACHE FILEPATH "Explicit owned descriptor implementation")
set(NV_CHECKSUM_SOURCE "${ROOT}/labs/b045_systems/src/systems.c" CACHE FILEPATH "Explicit checksum provider, optionally exact author slice")
set(NV_DEPTH_SOURCE "${ROOT}/labs/http_depth/src/http_depth.c" CACHE FILEPATH "Explicit original date provider")
set(NV_FLOW_SOURCE "${ROOT}/labs/hosted_followthrough/http/flow.c" CACHE FILEPATH "Explicit serialized pool provider")
foreach(p NV_SOURCE NV_OWNED_SOURCE NV_CHECKSUM_SOURCE NV_DEPTH_SOURCE NV_FLOW_SOURCE)
  if(NOT EXISTS "${${p}}")
    message(FATAL_ERROR "Missing explicit provider ${p}: ${${p}}")
  endif()
endforeach()
add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffunction-sections -fdata-sections)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}" "${ROOT}/labs/b045_systems/include" "${ROOT}/labs/http_depth/include" "${ROOT}/labs/hosted_followthrough/http" "${ROOT}/labs/communications/include")
add_executable(nv_core tests.c "${NV_SOURCE}" "${NV_CHECKSUM_SOURCE}" "${NV_DEPTH_SOURCE}")
add_executable(nv_owned owned_tests.c "${NV_OWNED_SOURCE}" "${NV_FLOW_SOURCE}")
target_link_options(nv_core PRIVATE -Wl,--gc-sections)
target_link_options(nv_owned PRIVATE -Wl,--gc-sections)
find_package(Threads REQUIRED)
target_link_libraries(nv_owned PRIVATE Threads::Threads)
enable_testing()
foreach(case checksum length date)
  add_test(NAME network_http_variants.${case} COMMAND nv_core ${case})
endforeach()
add_test(NAME network_http_variants.owned COMMAND nv_owned)
