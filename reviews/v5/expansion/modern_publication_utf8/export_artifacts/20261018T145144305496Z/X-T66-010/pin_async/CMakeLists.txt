cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_pin_async C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(OPA_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OPA_PRIVATE "${OPA_DIR}/src")
set(OPA_OWNERSHIP "${OPA_DIR}/../src/ownership.c")
set(CMASTERY_FOUNDATIONS_ROOT "${OPA_DIR}/../../foundations" CACHE PATH "Completed foundations workspace with io.async")
set(OPA_FOUNDATION "${CMASTERY_FOUNDATIONS_ROOT}/src/foundation.c")
set(OPA_PD "${OPA_DIR}/imports/dma.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OPA_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
  set(OPA_FOUNDATION "${CMASTERY_ASSESSOR_ROOT}/foundations/foundation.c")
 else()
  set(OPA_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OPA_PRIVATE "${OPA_ASSESSOR}/pin_async")
 set(OPA_OWNERSHIP "${OPA_ASSESSOR}/ownership.c")
 set(OPA_PD "${OPA_ASSESSOR}/../peripheral_depth/dma/dma.c")
endif()
if(CMASTERY_FOUNDATIONS_ASSESSOR_ROOT)
 set(OPA_FOUNDATION "${CMASTERY_FOUNDATIONS_ASSESSOR_ROOT}/foundation.c")
endif()
if(DEFINED CMASTERY_PD_SOURCE)
 set(OPA_PD "${CMASTERY_PD_SOURCE}")
endif()
if(NOT EXISTS "${OPA_PD}")
 message(FATAL_ERROR "Pin DMA composition requires an explicit completed Pd source: CMASTERY_PD_SOURCE")
endif()
if(NOT EXISTS "${OPA_FOUNDATION}")
 message(STATUS "Pin composition requires explicit completed io.async foundations workspace")
 return()
endif()
if(CMASTERY_OPA_LEARNER)
 set(OPA_PRIVATE "${OPA_DIR}/src")
 set(OPA_OWNERSHIP "${OPA_DIR}/../src/ownership.c")
endif()
function(opa_target name)
 add_executable(${name} "${OPA_DIR}/tests/tests.c" "${OPA_DIR}/tests/dma_tests.c" "${OPA_PRIVATE}/pin_async.c" "${OPA_PRIVATE}/pin_host.c" "${OPA_PRIVATE}/pin_dma.c" "${OPA_OWNERSHIP}" "${OPA_FOUNDATION}" "${OPA_PD}")
 target_include_directories(${name} PRIVATE "${OPA_DIR}/include" "${OPA_DIR}/imports" "${OPA_DIR}/../include" "${CMASTERY_FOUNDATIONS_ROOT}/include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
opa_target(ownership-pin_async)
foreach(case pin_async_base pin_async_timeout pin_async_notification pin_async_identity pin_async_counters pin_async_host pin_async_dma)
 add_test(NAME ownership.${case} COMMAND ownership-pin_async ${case})
 set_tests_properties(ownership.${case} PROPERTIES TIMEOUT 30)
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OPA_CASES pin_async_timeout pin_async_notification pin_async_identity pin_async_counters pin_async_base pin_async_base pin_async_dma pin_async_dma)
 foreach(i RANGE 1 8)
  opa_target(ownership-pin_async-mutant-${i})
  if(i EQUAL 5)
   target_compile_definitions(ownership-pin_async-mutant-${i} PRIVATE OW_MUTANT=8)
  elseif(i EQUAL 6)
   target_compile_definitions(ownership-pin_async-mutant-${i} PRIVATE OW_MUTANT=18)
  elseif(i EQUAL 7)
   target_compile_definitions(ownership-pin_async-mutant-${i} PRIVATE OPA_MUTATION=5)
  elseif(i EQUAL 8)
   target_compile_definitions(ownership-pin_async-mutant-${i} PRIVATE PD_MUTANT=5)
  else()
   target_compile_definitions(ownership-pin_async-mutant-${i} PRIVATE OPA_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OPA_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OPA_DIR}/expect_failure.py" $<TARGET_FILE:ownership-pin_async-mutant-${i}> ${case})
 endforeach()
endif()
