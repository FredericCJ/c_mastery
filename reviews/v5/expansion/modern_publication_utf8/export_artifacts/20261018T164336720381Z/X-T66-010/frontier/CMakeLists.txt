cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_frontier C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OCF_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OCF_PRIVATE "${OCF_DIR}/src")
set(OCF_OWNERSHIP "${OCF_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OCF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OCF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OCF_PRIVATE "${OCF_ASSESSOR}/frontier")
 set(OCF_OWNERSHIP "${OCF_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OCF_LEARNER)
 set(OCF_PRIVATE "${OCF_DIR}/src")
 set(OCF_OWNERSHIP "${OCF_DIR}/../src/ownership.c")
endif()
function(ocf_target name)
 add_executable(${name} "${OCF_DIR}/tests/tests.c" "${OCF_PRIVATE}/frontier.c" "${OCF_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OCF_DIR}/include" "${OCF_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ocf_target(ownership-frontier)
foreach(case frontier_base frontier_dag frontier_partial frontier_admission frontier_callbacks)
 add_test(NAME ownership.${case} COMMAND ownership-frontier ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OCF_CASES frontier_partial frontier_dag frontier_partial frontier_admission frontier_partial frontier_dag frontier_base frontier_base frontier_callbacks)
 foreach(i RANGE 1 9)
  ocf_target(ownership-frontier-mutant-${i})
  if(i EQUAL 7)
   target_compile_definitions(ownership-frontier-mutant-${i} PRIVATE OW_MUTANT=1)
  elseif(i EQUAL 8)
   target_compile_definitions(ownership-frontier-mutant-${i} PRIVATE OW_MUTANT=11)
  else()
   target_compile_definitions(ownership-frontier-mutant-${i} PRIVATE OCF_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OCF_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OCF_DIR}/expect_failure.py" $<TARGET_FILE:ownership-frontier-mutant-${i}> ${case})
 endforeach()
endif()
