cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_references C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(ORC_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ORC_PRIVATE "${ORC_DIR}/src")
set(ORC_OWNERSHIP "${ORC_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(ORC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(ORC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ORC_PRIVATE "${ORC_ASSESSOR}/references")
 set(ORC_OWNERSHIP "${ORC_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_ORC_LEARNER)
 set(ORC_PRIVATE "${ORC_DIR}/src")
 set(ORC_OWNERSHIP "${ORC_DIR}/../src/ownership.c")
endif()
function(omm_target name)
 add_executable(${name} "${ORC_DIR}/tests/tests.c" "${ORC_PRIVATE}/references.c" "${ORC_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${ORC_DIR}/include" "${ORC_DIR}/../include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
omm_target(ownership-references)
foreach(case references_base references_overflow references_cycle references_failures references_threads references_race)
 add_test(NAME ownership.${case} COMMAND ownership-references ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ORC_CASES references_cycle references_base references_base)
 foreach(i RANGE 1 3)
  omm_target(ownership-references-mutant-${i})
  if(i EQUAL 2)
   target_compile_definitions(ownership-references-mutant-${i} PRIVATE OW_MUTANT=6)
  elseif(i EQUAL 3)
   target_compile_definitions(ownership-references-mutant-${i} PRIVATE OW_MUTANT=16)
  else()
   target_compile_definitions(ownership-references-mutant-${i} PRIVATE ORC_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET ORC_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ORC_DIR}/expect_failure.py" $<TARGET_FILE:ownership-references-mutant-${i}> ${case})
 endforeach()
endif()
