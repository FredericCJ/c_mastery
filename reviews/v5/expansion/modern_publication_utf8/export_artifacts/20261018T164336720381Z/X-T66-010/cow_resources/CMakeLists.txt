cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_cow_resources C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(OCR_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OCR_PRIVATE "${OCR_DIR}/src")
set(OCR_OWNERSHIP "${OCR_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OCR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OCR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OCR_PRIVATE "${OCR_ASSESSOR}/cow_resources")
 set(OCR_OWNERSHIP "${OCR_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OCR_LEARNER)
 set(OCR_PRIVATE "${OCR_DIR}/src")
 set(OCR_OWNERSHIP "${OCR_DIR}/../src/ownership.c")
endif()
function(omm_target name)
 add_executable(${name} "${OCR_DIR}/tests/tests.c" "${OCR_PRIVATE}/cow_resources.c" "${OCR_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OCR_DIR}/include" "${OCR_DIR}/../include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
omm_target(ownership-cow_resources)
foreach(case cow_resources_base cow_resources_snapshots cow_resources_failures cow_resources_identity cow_resources_close cow_resources_threads)
 add_test(NAME ownership.${case} COMMAND ownership-cow_resources ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OCR_CASES cow_resources_snapshots cow_resources_failures cow_resources_identity cow_resources_close cow_resources_base cow_resources_base)
 foreach(i RANGE 1 6)
  omm_target(ownership-cow_resources-mutant-${i})
  if(i EQUAL 5)
   target_compile_definitions(ownership-cow_resources-mutant-${i} PRIVATE OW_MUTANT=7)
  elseif(i EQUAL 6)
   target_compile_definitions(ownership-cow_resources-mutant-${i} PRIVATE OW_MUTANT=17)
  else()
   target_compile_definitions(ownership-cow_resources-mutant-${i} PRIVATE OCR_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OCR_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OCR_DIR}/expect_failure.py" $<TARGET_FILE:ownership-cow_resources-mutant-${i}> ${case})
 endforeach()
endif()
