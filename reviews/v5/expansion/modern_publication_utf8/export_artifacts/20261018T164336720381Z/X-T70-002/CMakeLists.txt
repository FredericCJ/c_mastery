cmake_minimum_required(VERSION 3.20)
project(dod C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or dod directory")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
set(IMPL src/dod.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/dod/dod.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/dod/dod.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/dod.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(dod-tests tests/tests.c tests/histories.c ${IMPL})
target_include_directories(dod-tests PRIVATE include)
enable_testing()
foreach(case columns small swar sparse ecs persistent hamt rope)
 add_test(NAME dod.${case} COMMAND dod-tests ${case})
 set_tests_properties(dod.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2)
   add_executable(dod-mutant-${case}-${mutant} tests/tests.c tests/histories.c ${IMPL})
   target_include_directories(dod-mutant-${case}-${mutant} PRIVATE include)
   string(TOUPPER ${case} CASE_UPPER)
   if(mutant EQUAL 1)
    target_compile_definitions(dod-mutant-${case}-${mutant} PRIVATE D_MUTATE_${CASE_UPPER})
   else()
    target_compile_definitions(dod-mutant-${case}-${mutant} PRIVATE D_MUTATE_${CASE_UPPER}2)
   endif()
   add_test(NAME dod.mutation.${case}.${mutant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:dod-mutant-${case}-${mutant}> ${case})
   set_tests_properties(dod.mutation.${case}.${mutant} PROPERTIES TIMEOUT 20)
  endforeach()
 endif()
endforeach()
get_filename_component(TEXT_DIR "${IMPL}" DIRECTORY)
function(dod_text_program target)
 add_executable(${target} tests/text_ext.c ${IMPL} "${TEXT_DIR}/dod_text.c")
 target_include_directories(${target} PRIVATE include)
endfunction()
dod_text_program(dod-text)
set(text_index 0)
foreach(case rope_edit editors utf8 rope_splice)
 add_test(NAME dod.${case} COMMAND dod-text ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 1 2)
   math(EXPR mutant "${text_index}*2+${variant}")
   dod_text_program(dod-text-mutant-${mutant})
   target_compile_definitions(dod-text-mutant-${mutant} PRIVATE D_TEXT_MUTANT=${mutant})
   add_test(NAME dod.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:dod-text-mutant-${mutant}> ${case})
  endforeach()
 endif()
 math(EXPR text_index "${text_index}+1")
endforeach()
