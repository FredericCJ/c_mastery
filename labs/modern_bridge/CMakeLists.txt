cmake_minimum_required(VERSION 3.20)
project(modern_bridge C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or modern_bridge assessor root")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
set(IMPL src/bridge.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/modern_bridge/bridge.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/modern_bridge/bridge.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/bridge.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(mb_program target)
 add_executable(${target} tests/tests.c ${IMPL})
 target_include_directories(${target} PRIVATE include)
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
mb_program(modern_bridge-tests)
enable_testing()
foreach(case region once hazard epoch)
 add_test(NAME modern_bridge.${case} COMMAND modern_bridge-tests ${case})
 set_tests_properties(modern_bridge.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2)
   mb_program(modern_bridge-mutant-${case}-${mutant})
   string(TOUPPER ${case} CASE_UPPER)
   target_compile_definitions(modern_bridge-mutant-${case}-${mutant} PRIVATE MB_MUTATE_${CASE_UPPER}${mutant})
   add_test(NAME modern_bridge.mutation.${case}.${mutant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:modern_bridge-mutant-${case}-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
