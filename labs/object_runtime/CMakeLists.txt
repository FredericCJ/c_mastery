cmake_minimum_required(VERSION 3.20)
project(c_mastery_object_runtime C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or family")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
option(CMASTERY_TSAN "Separate hosted thread sanitizer" OFF)
set(CMASTERY_OWNERSHIP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../ownership" CACHE PATH "Accepted ownership learner workspace")
set(CMASTERY_OWNERSHIP_ASSESSOR_ROOT "" CACHE PATH "Explicit ownership assessor root or family")
set(CMASTERY_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../runtime" CACHE PATH "Completed runtime learner workspace")
set(CMASTERY_ALGORITHMS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../algorithms" CACHE PATH "Completed algorithms learner workspace")
set(CMASTERY_RUNTIME_ASSESSOR_ROOT "" CACHE PATH "Explicit runtime assessor root or family")
set(CMASTERY_ALGORITHMS_ASSESSOR_ROOT "" CACHE PATH "Explicit algorithms assessor root or family")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "ASan and TSan require separate executions")
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/object_runtime.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}")
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/object_runtime/object_runtime.c")
  set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}/object_runtime")
  if(NOT CMASTERY_OWNERSHIP_ASSESSOR_ROOT)
   set(CMASTERY_OWNERSHIP_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/ownership")
  endif()
  foreach(family RUNTIME ALGORITHMS)
   string(TOLOWER ${family} lower)
   if(NOT CMASTERY_${family}_ASSESSOR_ROOT)
    set(CMASTERY_${family}_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/${lower}")
   endif()
  endforeach()
 endif()
 set(IMPL "${ASSESSOR_FAMILY}/object_runtime.c")
endif()
add_library(object_runtime STATIC "${IMPL}")
target_include_directories(object_runtime PUBLIC include)
add_executable(object_runtime-tests tests/tests.c)
target_link_libraries(object_runtime-tests PRIVATE object_runtime)
enable_testing()
foreach(case value intern shapes cache mark nursery cards compact weak trial safepoint)
 add_test(NAME object_runtime.${case} COMMAND object_runtime-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(number 1 2)
   string(TOUPPER ${case} upper)
   set(mutation "OR_MUTATE_${upper}")
   if(number EQUAL 2)
    string(APPEND mutation 2)
   endif()
   add_executable(mutant-${case}-${number} tests/tests.c "${IMPL}")
   target_include_directories(mutant-${case}-${number} PRIVATE include)
   target_compile_definitions(mutant-${case}-${number} PRIVATE ${mutation})
   add_test(NAME object_runtime.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}> ${case})
  endforeach()
 endif()
endforeach()

foreach(family RUNTIME ALGORITHMS)
 string(TOLOWER ${family} lower)
 set(${family}_IMPL "${CMASTERY_${family}_ROOT}/src")
 if(CMASTERY_${family}_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_${family}_ASSESSOR_ROOT}/${lower}")
   set(${family}_IMPL "${CMASTERY_${family}_ASSESSOR_ROOT}/${lower}")
  else()
   set(${family}_IMPL "${CMASTERY_${family}_ASSESSOR_ROOT}")
  endif()
 endif()
endforeach()
set_source_files_properties("${ALGORITHMS_IMPL}/algo.c" PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")

                "${RUNTIME_IMPL}/cvm.c" "${RUNTIME_IMPL}/compiler.c" "${ALGORITHMS_IMPL}/algo.c")
 target_include_directories(${target} PRIVATE include "${CMASTERY_RUNTIME_ROOT}/include" "${CMASTERY_ALGORITHMS_ROOT}/include")
 target_link_libraries(${target} PRIVATE m)
endfunction()

            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py
                    $<TARGET_FILE:object-runtime-vm-${case}-mutant-${number}> ${case})
  endif()
 endforeach()
endforeach()
