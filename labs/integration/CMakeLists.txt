cmake_minimum_required(VERSION 3.20)
project(cmastery_integration C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
set(CMASTERY_COMPONENTS_FILE "" CACHE FILEPATH "Verified local exported-component target definitions")
if(CMASTERY_COMPONENTS_FILE)
 if(NOT IS_ABSOLUTE "${CMASTERY_COMPONENTS_FILE}" OR NOT EXISTS "${CMASTERY_COMPONENTS_FILE}")
  message(FATAL_ERROR "Component definitions must name an existing absolute file")
 endif()
 include("${CMASTERY_COMPONENTS_FILE}")
endif()
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/integration directory")
set(IN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor root must be absolute")
 endif()
 set(IN_SOURCE "${CMASTERY_ASSESSOR_ROOT}")
endif()
# Roots name the exported implementation source directory. Includes are explicit
# because source snapshots may use a different layout from the course scaffold.
foreach(dep FOUNDATIONS ALGORITHMS FIRMWARE_IO RUNTIME STORAGE)
 set(CMASTERY_COMPONENTS_${dep}_TARGETS "${CMASTERY_COMPONENTS_${dep}_TARGETS}" CACHE STRING "Verified component target(s) for ${dep}")
 set(HAVE_${dep} FALSE)
 foreach(kind ROOT INCLUDE)
  set(CMASTERY_${dep}_${kind} "$ENV{CMASTERY_${dep}_${kind}}" CACHE PATH "Explicit ${dep} ${kind}")
  if(CMASTERY_${dep}_${kind} AND NOT IS_ABSOLUTE "${CMASTERY_${dep}_${kind}}")
   message(FATAL_ERROR "Dependency paths must be absolute")
  endif()
 endforeach()
 if(CMASTERY_COMPONENTS_${dep}_TARGETS)
  foreach(component ${CMASTERY_COMPONENTS_${dep}_TARGETS})
   if(NOT TARGET ${component})
    message(FATAL_ERROR "Missing exported target ${component}")
   endif()
  endforeach()
  set(HAVE_${dep} TRUE)
 elseif(CMASTERY_${dep}_ROOT AND CMASTERY_${dep}_INCLUDE)
  set(HAVE_${dep} TRUE)
 endif()
endforeach()
if(NOT HAVE_FOUNDATIONS OR NOT HAVE_ALGORITHMS)
 message(STATUS "Integration unavailable: import actual foundations and algorithms components. No assessor fallback.")
 return()
endif()
set(IN_DEPS "")
set(IN_LINKS "")
set(IN_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/include")
set(IN_DEFINES "")
set(IN_CASES p1 p2 p3 p5 p6)
set(IN_ROLES FOUNDATIONS ALGORITHMS)
if(HAVE_FIRMWARE_IO)
 list(APPEND IN_ROLES FIRMWARE_IO)
 list(APPEND IN_DEFINES IN_FIRMWARE=1)
 list(APPEND IN_CASES p4 mcu)
endif()
if(HAVE_RUNTIME AND HAVE_STORAGE)
 list(APPEND IN_ROLES RUNTIME STORAGE)
 list(APPEND IN_DEFINES IN_VM_STORE=1)
 list(APPEND IN_CASES linux)
endif()
set(SOURCES_FOUNDATIONS foundation.c v4_foundation.c)
set(SOURCES_ALGORITHMS algo.c v4_algo.c)
set(SOURCES_FIRMWARE_IO firmware_io.c)
set(SOURCES_RUNTIME cvm.c compiler.c)
set(SOURCES_STORAGE store.c)
foreach(dep ${IN_ROLES})
 if(CMASTERY_COMPONENTS_${dep}_TARGETS)
  list(APPEND IN_LINKS ${CMASTERY_COMPONENTS_${dep}_TARGETS})
 else()
  list(APPEND IN_INCLUDES "${CMASTERY_${dep}_INCLUDE}")
  foreach(source ${SOURCES_${dep}})
   list(APPEND IN_DEPS "${CMASTERY_${dep}_ROOT}/${source}")
  endforeach()
 endif()
endforeach()
function(in_target name)
 add_executable(${name} "${IN_SOURCE}/integration.c" tests/contracts.c ${IN_DEPS})
 target_include_directories(${name} PRIVATE ${IN_INCLUDES})
 target_compile_definitions(${name} PRIVATE ${IN_DEFINES})
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${name} PRIVATE ${IN_LINKS})
 if(UNIX)
  target_link_libraries(${name} PRIVATE m)
 endif()
endfunction()
in_target(integration_contracts)
foreach(case ${IN_CASES})
 add_test(NAME integration.${case} COMMAND integration_contracts ${case})
 set_tests_properties(integration.${case} PROPERTIES TIMEOUT 10 LABELS "host;integration;explicit-imports")
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 in_target(integration_lookahead)
 target_compile_definitions(integration_lookahead PRIVATE IN_LOOKAHEAD=1)
 add_test(NAME integration.architecture_lookahead COMMAND integration_lookahead p1)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 if(NOT IN_LINKS)
  include(tests/mutations.cmake)
 endif()
endif()
