cmake_minimum_required(VERSION 3.16)
project(corpus_tdigest C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(TD_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/tdigest.c" CACHE FILEPATH "Completed original provider")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private provider")
option(CMASTERY_SANITIZE "Hosted sanitizers" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_tdigest/tdigest.c")
    set(TD_PROVIDER "${CMASTERY_ASSESSOR_ROOT}/corpus_tdigest/tdigest.c")
  else()
    set(TD_PROVIDER "${CMASTERY_ASSESSOR_ROOT}/tdigest.c")
  endif()
endif()
if(NOT EXISTS "${TD_PROVIDER}")
  message(FATAL_ERROR "Supply explicit completed tdigest provider")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(tdigest STATIC "${TD_PROVIDER}")
target_include_directories(tdigest PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(tdigest_tests tests.c)
target_link_libraries(tdigest_tests PRIVATE tdigest)
add_executable(tdigest_driver driver.c)
target_link_libraries(tdigest_driver PRIVATE tdigest)
enable_testing()
add_test(NAME corpus_tdigest.contract COMMAND tdigest_tests)
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(m RANGE 1 3)
    add_executable(tdigest_mutant_${m} tests.c "${TD_PROVIDER}")
    target_include_directories(tdigest_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(tdigest_mutant_${m} PRIVATE TD_MUTANT=${m})
    add_test(NAME corpus_tdigest.mutation.contract.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:tdigest_mutant_${m}>)
  endforeach()
endif()
