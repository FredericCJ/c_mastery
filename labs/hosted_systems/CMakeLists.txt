cmake_minimum_required(VERSION 3.20)
project(cmastery_hosted_systems C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/hosted_systems directory")
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor directory must be absolute")
 endif()
 set(IMPL "${CMASTERY_ASSESSOR_ROOT}")
endif()
set(HS_SOURCES)
foreach(part descriptor copy deadline process mapping frame signal ipc reactor pool queue clocks)
 list(APPEND HS_SOURCES "${IMPL}/${part}.c")
 add_library(hosted_${part} STATIC "${IMPL}/${part}.c")
 target_include_directories(hosted_${part} PUBLIC include)
 target_link_libraries(hosted_${part} PUBLIC Threads::Threads)
 target_compile_options(hosted_${part} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 install(TARGETS hosted_${part} ARCHIVE DESTINATION lib)
endforeach()
add_library(hosted STATIC ${HS_SOURCES})
target_include_directories(hosted PUBLIC include)
target_link_libraries(hosted PUBLIC Threads::Threads)
target_compile_options(hosted PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(hosted_tests tests/tests.c)
target_compile_options(hosted_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(hosted_tests PRIVATE hosted)
enable_testing()
foreach(case descriptor copy deadline process mapping frame signal ipc reactor pool queue clocks)
 add_test(NAME hosted_systems.${case} COMMAND hosted_tests ${case})
 set_tests_properties(hosted_systems.${case} PROPERTIES TIMEOUT 10)
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 add_executable(hosted_alternate tests/tests.c ${HS_SOURCES})
 target_include_directories(hosted_alternate PRIVATE include)
 target_compile_definitions(hosted_alternate PRIVATE HS_ALTERNATE)
 target_compile_options(hosted_alternate PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 target_link_libraries(hosted_alternate PRIVATE Threads::Threads)
 add_test(NAME hosted_systems.pool_alternative COMMAND hosted_alternate pool)
 set_tests_properties(hosted_systems.pool_alternative PROPERTIES TIMEOUT 10)
endif()
install(TARGETS hosted ARCHIVE DESTINATION lib)
install(FILES include/hosted.h DESTINATION include)
