cmake_minimum_required(VERSION 3.20)
project(modern_composition C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit own assessor or common root")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
option(CMASTERY_TSAN "Separate hosted thread sanitizer" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/composition.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/modern_composition/composition.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/modern_composition/composition.c")
  set(COMMON_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/composition.c")
 endif()
endif()
set(IMPORT_SOURCES "")
set(IMPORT_INCLUDES "")
foreach(dependency errors ownership coroutines messaging)
 string(TOUPPER ${dependency} upper)
 set(CMASTERY_${upper}_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../${dependency}" CACHE PATH "Explicit prerequisite learner workspace")
 set(CMASTERY_${upper}_ASSESSOR_ROOT "" CACHE PATH "Explicit prerequisite assessor")
 set(source "${CMASTERY_${upper}_ROOT}/src/${dependency}.c")
 set(reference "${CMASTERY_${upper}_ASSESSOR_ROOT}")
 if(NOT reference AND COMMON_ASSESSOR)
  set(reference "${COMMON_ASSESSOR}/${dependency}")
 endif()
 if(reference)
  if(EXISTS "${reference}/${dependency}/${dependency}.c")
   set(source "${reference}/${dependency}/${dependency}.c")
  else()
   set(source "${reference}/${dependency}.c")
  endif()
 endif()
 list(APPEND IMPORT_SOURCES "${source}")
 list(APPEND IMPORT_INCLUDES "${CMASTERY_${upper}_ROOT}/include")
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "ASan and TSan need separate executions")
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
function(adapter_program target)
 add_executable(${target} tests/tests.c "${IMPL}" ${IMPORT_SOURCES})
 target_include_directories(${target} PRIVATE include ${IMPORT_INCLUDES})
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
adapter_program(composition-tests)
enable_testing()
foreach(case publish async)
 add_test(NAME modern_composition.${case} COMMAND composition-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 1 2)
   set(mutant ${variant})
   if(case STREQUAL "async")
    math(EXPR mutant "${variant}+2")
   endif()
   adapter_program(composition-mutant-${mutant})
   target_compile_definitions(composition-mutant-${mutant} PRIVATE MC_MUTANT=${mutant})
   add_test(NAME modern_composition.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:composition-mutant-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
