cmake_minimum_required(VERSION 3.20)
project(corpus_shunting C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or family assessor root")
set(CMASTERY_PARSING_DEPTH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../parsing_depth" CACHE PATH "Actual prerequisite learner workspace")
set(CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT "" CACHE PATH "Explicit completed prerequisite provider")
option(CMASTERY_SANITIZE "Address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL src/shunting.c)
set(PROVIDER "${CMASTERY_PARSING_DEPTH_ROOT}/src/parsing.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_shunting/shunting.c")
  set(TC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/corpus_shunting")
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/parsing_depth")
 else()
  set(TC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/../parsing_depth")
 endif()
 set(IMPL "${TC_ASSESSOR}/shunting.c")
 set(PROVIDER "${PD_ASSESSOR}/pratt.c")
endif()
if(CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/parsing_depth/pratt.c")
  set(PROVIDER "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/parsing_depth/pratt.c")
 else()
  set(PROVIDER "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/pratt.c")
 endif()
endif()
if(NOT EXISTS "${CMASTERY_PARSING_DEPTH_ROOT}/parsing.h" OR NOT EXISTS "${PROVIDER}")
 message(FATAL_ERROR "Supply the actual parsing_depth prerequisite workspace/provider")
endif()
function(sy_program target driver)
 add_executable(${target} ${driver} ${IMPL} "${PROVIDER}")
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMASTERY_PARSING_DEPTH_ROOT}")
endfunction()
sy_program(shunting-tests tests.c)
sy_program(shunting-driver driver.c)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
foreach(anchor semantics bounds stream)
 add_test(NAME corpus_shunting.${anchor} COMMAND shunting-tests ${anchor})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 add_test(NAME corpus_shunting.oracle COMMAND "${Python3_EXECUTABLE}" "${TC_ASSESSOR}/oracle.py" $<TARGET_FILE:shunting-driver>)
 set_tests_properties(corpus_shunting.oracle PROPERTIES TIMEOUT 180)
 foreach(m RANGE 1 7)
  sy_program(shunting-mutant-${m} tests.c)
  target_compile_definitions(shunting-mutant-${m} PRIVATE SY_MUTANT=${m})
  if(m LESS 4)
   set(anchor semantics)
  elseif(m EQUAL 4 OR m EQUAL 7)
   set(anchor bounds)
  else()
   set(anchor stream)
  endif()
  add_test(NAME corpus_shunting.mutation.${anchor}.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:shunting-mutant-${m}> ${anchor})
 endforeach()
endif()
