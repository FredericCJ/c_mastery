cmake_minimum_required(VERSION 3.20)
project(linux_supervision LANGUAGES C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(cases namespace pids shutdown network telemetry)
set(implementation "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
    set(implementation "${CMASTERY_ASSESSOR_ROOT}")
endif()
add_executable(supervise supervisor_main.c)
foreach(case IN LISTS cases)
    target_sources(supervise PRIVATE "${implementation}/${case}.c")
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
    target_sources(supervise PRIVATE "${implementation}/support.c")
endif()
target_include_directories(supervise PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(supervise PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(supervise PRIVATE -static)
enable_testing()
foreach(case IN LISTS cases)
    add_test(NAME supervision.${case} COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/submission.py" check "${case}"
        --program "$<TARGET_FILE:supervise>" --out "${CMAKE_CURRENT_BINARY_DIR}/targets/${case}")
    set_tests_properties(supervision.${case} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 150)
endforeach()
