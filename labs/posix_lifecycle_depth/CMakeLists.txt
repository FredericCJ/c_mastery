cmake_minimum_required(VERSION 3.20)
project(posix_lifecycle_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or family assessor root")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
option(CMASTERY_TSAN "ThreadSanitizer" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/posix_lifecycle.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/posix_lifecycle_depth/posix_lifecycle.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/posix_lifecycle_depth/posix_lifecycle.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/posix_lifecycle.c")
 endif()
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -g -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
add_executable(pl-helper tests/helper.c)
function(program target)
 add_executable(${target} tests/tests.c "${IMPL}")
 target_include_directories(${target} PRIVATE include)
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
program(pl-tests)
enable_testing()
foreach(case launch pool)
 add_test(NAME posix_lifecycle_depth.${case} COMMAND pl-tests ${case} $<TARGET_FILE:pl-helper>)
 set_tests_properties(posix_lifecycle_depth.${case} PROPERTIES TIMEOUT 15)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 1 2)
   set(mutant ${variant})
   if(case STREQUAL "pool")
    math(EXPR mutant "${variant}+2")
   endif()
   program(pl-mutant-${mutant})
   target_compile_definitions(pl-mutant-${mutant} PRIVATE PL_MUTANT=${mutant})
   add_test(NAME posix_lifecycle_depth.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:pl-mutant-${mutant}> ${case} $<TARGET_FILE:pl-helper>)
  endforeach()
 endif()
endforeach()
