if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")

  set(EJD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")

 set(EJD_OWNERSHIP "${CMASTERY_OWNERSHIP_ASSESSOR_ROOT}/ownership.c")

function(ejd_target name)
 add_executable(${name} "${EJD_DIR}/tests/tests.c" "${EJD_CORE}" "${EJD_ERRORS}" "${EJD_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${EJD_DIR}/include" "${EJD_DIR}/../include" "${CMASTERY_OWNERSHIP_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ejd_target(errors-jump_scope)
foreach(case jump_stack jump_resources jump_nested jump_effects jump_admission)
 add_test(NAME errors.${case} COMMAND errors-jump_scope ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EJD_CASES jump_resources jump_effects jump_nested jump_nested jump_effects jump_nested jump_stack jump_stack jump_resources)
 foreach(i RANGE 1 9)
  ejd_target(errors-jump_scope-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(errors-jump_scope-mutant-${i} PRIVATE EJD_MUTATION=${i})
  elseif(i LESS 9)
   math(EXPR baseline "${i}+4")
   target_compile_definitions(errors-jump_scope-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  else()
   target_compile_definitions(errors-jump_scope-mutant-${i} PRIVATE OW_MUTANT=12)
  endif()
  math(EXPR index "${i}-1")
  list(GET EJD_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EJD_DIR}/expect_failure.py" $<TARGET_FILE:errors-jump_scope-mutant-${i}> ${case})
 endforeach()
endif()
