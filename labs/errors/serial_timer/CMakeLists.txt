cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_serial_timer C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EAT_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(EAT_PRIVATE "${EAT_DIR}/src")
set(EAT_ERRORS "${EAT_DIR}/../src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(EAT_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
 else()
  set(EAT_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(EAT_PRIVATE "${EAT_ASSESSOR}/serial_timer")
 set(EAT_ERRORS "${EAT_ASSESSOR}/errors.c")
endif()
if(CMASTERY_EAT_LEARNER)
 set(EAT_PRIVATE "${EAT_DIR}/src")
 set(EAT_ERRORS "${EAT_DIR}/../src/errors.c")
endif()
function(eat_target name)
 add_executable(${name} "${EAT_DIR}/tests/tests.c" "${EAT_PRIVATE}/serial_timer.c" "${EAT_ERRORS}")
 target_include_directories(${name} PRIVATE "${EAT_PRIVATE}" "${EAT_DIR}/include" "${EAT_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
eat_target(errors-serial_timer)
foreach(case timer_base timer_relation timer_window timer_lifetime timer_epoch timer_admission)
 add_test(NAME errors.${case} COMMAND errors-serial_timer ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EAT_CASES timer_lifetime timer_lifetime timer_window timer_window timer_epoch timer_epoch timer_base timer_base)
 foreach(i RANGE 1 8)
  eat_target(errors-serial_timer-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(errors-serial_timer-mutant-${i} PRIVATE EAT_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+6")
   target_compile_definitions(errors-serial_timer-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET EAT_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EAT_DIR}/expect_failure.py" $<TARGET_FILE:errors-serial_timer-mutant-${i}> ${case})
 endforeach()
endif()
