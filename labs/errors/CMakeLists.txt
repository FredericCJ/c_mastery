cmake_minimum_required(VERSION 3.20)
project(c_mastery_errors C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  else()
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}/errors.c")
  endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_executable(errors-tests tests/tests.c "${IMPL}")
target_include_directories(errors-tests PRIVATE include)
add_executable(errors-oracle-driver tests/oracle_driver.c "${IMPL}")
target_include_directories(errors-oracle-driver PRIVATE include)
enable_testing()
set(index 0)
foreach(case optional propagation progress composition validation boundary arithmetic contracts refinement)
  add_test(NAME errors.${case} COMMAND errors-tests ${case})
  set_tests_properties(errors.${case} PROPERTIES TIMEOUT 20)
  if(CMASTERY_ASSESSOR_ROOT)
    foreach(variant RANGE 1 2)
      math(EXPR index "${index}+1")
      add_executable(errors-mutant-${index} tests/tests.c "${IMPL}")
      target_include_directories(errors-mutant-${index} PRIVATE include)
      target_compile_definitions(errors-mutant-${index} PRIVATE ER_MUTANT=${index})
      add_test(NAME errors.mutation.${case}.${variant}
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:errors-mutant-${index}> ${case})
      set_tests_properties(errors.mutation.${case}.${variant} PROPERTIES TIMEOUT 20)
    endforeach()
  endif()
endforeach()
