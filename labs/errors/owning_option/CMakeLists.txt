cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_owning_option C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EOP_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(EOP_CORE "${EOP_DIR}/src/owning_option.c")
set(EOP_ERRORS "${EOP_DIR}/../src/errors.c")
set(CMASTERY_OWNERSHIP_ROOT "${EOP_DIR}/../../ownership" CACHE PATH "Completed T66 ownership workspace for late follow-through")
set(EOP_OWNERSHIP "${CMASTERY_OWNERSHIP_ROOT}/src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(EOP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
  set(EOP_OWNERSHIP "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
 else()
  set(EOP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(EOP_CORE "${EOP_ASSESSOR}/owning_option/owning_option.c")
 set(EOP_ERRORS "${EOP_ASSESSOR}/errors.c")
endif()
if(CMASTERY_OWNERSHIP_ASSESSOR_ROOT)
 set(EOP_OWNERSHIP "${CMASTERY_OWNERSHIP_ASSESSOR_ROOT}/ownership.c")
endif()
if(NOT EXISTS "${EOP_OWNERSHIP}")
 message(STATUS "Late owning-option stage requires an explicit completed T66 ownership workspace")
 return()
endif()
if(CMASTERY_EOP_LEARNER)
 set(EOP_CORE "${EOP_DIR}/src/owning_option.c")
 set(EOP_ERRORS "${EOP_DIR}/../src/errors.c")
endif()
function(eop_target name)
 add_executable(${name} "${EOP_DIR}/tests/tests.c" "${EOP_CORE}" "${EOP_ERRORS}" "${EOP_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${EOP_DIR}/include" "${EOP_DIR}/../include" "${CMASTERY_OWNERSHIP_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
eop_target(errors-owning_option)
foreach(case option_scalar option_owner option_failure option_base)
 add_test(NAME errors.${case} COMMAND errors-owning_option ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EOP_CASES option_owner option_owner option_owner option_owner option_scalar option_scalar option_failure option_failure option_base option_base)
 foreach(i RANGE 1 10)
  eop_target(errors-owning_option-mutant-${i})
  if(i LESS 9)
   target_compile_definitions(errors-owning_option-mutant-${i} PRIVATE EOP_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-8")
   target_compile_definitions(errors-owning_option-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET EOP_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EOP_DIR}/expect_failure.py" $<TARGET_FILE:errors-owning_option-mutant-${i}> ${case})
 endforeach()
endif()
