cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_contract_modes C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(ECM_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ECM_PRIVATE "${ECM_DIR}/src")
set(ECM_ERRORS "${ECM_DIR}/../src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(ECM_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
 else()
  set(ECM_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ECM_PRIVATE "${ECM_ASSESSOR}/contract_modes")
 set(ECM_ERRORS "${ECM_ASSESSOR}/errors.c")
endif()
if(CMASTERY_ECM_LEARNER)
 set(ECM_PRIVATE "${ECM_DIR}/src")
 set(ECM_ERRORS "${ECM_DIR}/../src/errors.c")
endif()
function(ecm_target name)
 add_executable(${name} "${ECM_DIR}/tests/tests.c" "${ECM_PRIVATE}/contract_modes.c" "${ECM_ERRORS}")
 target_include_directories(${name} PRIVATE "${ECM_PRIVATE}" "${ECM_DIR}/include" "${ECM_DIR}/../include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ecm_target(errors-contract_modes)
foreach(case modes_base modes_bounds modes_exhaustion modes_admission modes_concurrent)
 add_test(NAME errors.${case} COMMAND errors-contract_modes ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ECM_CASES modes_bounds modes_exhaustion modes_bounds modes_bounds modes_concurrent modes_base modes_base)
 foreach(i RANGE 1 7)
  ecm_target(errors-contract_modes-mutant-${i})
  if(i LESS 6)
   target_compile_definitions(errors-contract_modes-mutant-${i} PRIVATE ECM_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+9")
   target_compile_definitions(errors-contract_modes-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET ECM_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ECM_DIR}/expect_failure.py" $<TARGET_FILE:errors-contract_modes-mutant-${i}> ${case})
 endforeach()
endif()
