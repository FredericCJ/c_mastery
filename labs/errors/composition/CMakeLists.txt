set(CMASTERY_OWNERSHIP_ROOT "${ERC_DIR}/../../ownership" CACHE PATH "Completed T66 ownership workspace for late follow-through")
set(ERC_OWNERSHIP "${CMASTERY_OWNERSHIP_ROOT}/src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")

  set(ERC_OWNERSHIP "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")

 set(ERC_OWNERSHIP "${CMASTERY_OWNERSHIP_ASSESSOR_ROOT}/ownership.c")

function(erc_target name)
 add_executable(${name} "${ERC_DIR}/tests/tests.c" "${ERC_CORE}" "${ERC_ERRORS}" "${ERC_OPTION}" "${ERC_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${ERC_DIR}/include" "${ERC_DIR}/../include" "${ERC_DIR}/../owning_option/include" "${CMASTERY_OWNERSHIP_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
erc_target(errors-composition)
foreach(case compose_laws compose_effects compose_owner compose_failure compose_cause compose_base)
 add_test(NAME errors.${case} COMMAND errors-composition ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ERC_CASES compose_owner compose_failure compose_owner compose_cause compose_cause compose_failure compose_owner compose_base compose_base)
 foreach(i RANGE 1 9)
  erc_target(errors-composition-mutant-${i})
  if(i LESS 8)
   target_compile_definitions(errors-composition-mutant-${i} PRIVATE ERC_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-1")
   target_compile_definitions(errors-composition-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET ERC_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ERC_DIR}/expect_failure.py" $<TARGET_FILE:errors-composition-mutant-${i}> ${case})
 endforeach()
endif()
