if(CMASTERY_ASSESSOR_ROOT)

function(ept_target name)
 if(ARGC GREATER 1)
  set(test_source "${ARGV1}")
 else()
  set(test_source "${EPT_DIR}/tests/tests.c")
 endif()
 add_executable(${name} "${test_source}" "${EPT_CORE}" "${EPT_ERRORS}" "${EPT_STREAM}")
 target_include_directories(${name} PRIVATE "${EPT_DIR}/include" "${EPT_DIR}/../include" "${CMASTERY_API_DEPTH_ROOT}/stream")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()

 endforeach()
