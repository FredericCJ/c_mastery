cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_rational C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EFR_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(EFR_PRIVATE "${EFR_DIR}/src")
set(EFR_ERRORS "${EFR_DIR}/../src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(EFR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
 else()
  set(EFR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(EFR_PRIVATE "${EFR_ASSESSOR}/rational")
 set(EFR_ERRORS "${EFR_ASSESSOR}/errors.c")
endif()
if(CMASTERY_EFR_LEARNER)
 set(EFR_PRIVATE "${EFR_DIR}/src")
 set(EFR_ERRORS "${EFR_DIR}/../src/errors.c")
endif()
function(efr_target name)
 add_executable(${name} "${EFR_DIR}/tests/tests.c" "${EFR_PRIVATE}/rational.c" "${EFR_PRIVATE}/opaque.c" "${EFR_ERRORS}")
 target_include_directories(${name} PRIVATE "${EFR_PRIVATE}" "${EFR_DIR}/include" "${EFR_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
efr_target(errors-rational)
foreach(case rational_base rational_consumer rational_add rational_signed rational_opaque rational_admission)
 add_test(NAME errors.${case} COMMAND errors-rational ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EFR_CASES rational_consumer rational_add rational_add rational_add rational_signed rational_signed rational_base rational_base)
 foreach(i RANGE 1 8)
  efr_target(errors-rational-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(errors-rational-mutant-${i} PRIVATE EFR_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+10")
   target_compile_definitions(errors-rational-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET EFR_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EFR_DIR}/expect_failure.py" $<TARGET_FILE:errors-rational-mutant-${i}> ${case})
 endforeach()
endif()
