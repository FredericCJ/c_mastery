cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_validation C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EVV_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(EVV_PRIVATE "${EVV_DIR}/src")
set(EVV_ERRORS "${EVV_DIR}/../src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(EVV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
 else()
  set(EVV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(EVV_PRIVATE "${EVV_ASSESSOR}/validation")
 set(EVV_ERRORS "${EVV_ASSESSOR}/errors.c")
endif()
if(CMASTERY_EVV_LEARNER)
 set(EVV_PRIVATE "${EVV_DIR}/src")
 set(EVV_ERRORS "${EVV_DIR}/../src/errors.c")
endif()
function(evv_target name)
 add_executable(${name} "${EVV_DIR}/tests/tests.c" "${EVV_PRIVATE}/validation.c" "${EVV_ERRORS}")
 target_include_directories(${name} PRIVATE "${EVV_PRIVATE}" "${EVV_DIR}/include" "${EVV_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
evv_target(errors-validation)
foreach(case validate_base validate_ratio validate_nested validate_admission validate_security)
 add_test(NAME errors.${case} COMMAND errors-validation ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EVV_CASES validate_security validate_nested validate_ratio validate_ratio validate_nested validate_nested validate_admission validate_base validate_base)
 foreach(i RANGE 1 9)
  evv_target(errors-validation-mutant-${i})
  if(i LESS 8)
   target_compile_definitions(errors-validation-mutant-${i} PRIVATE EVV_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+1")
   target_compile_definitions(errors-validation-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET EVV_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EVV_DIR}/expect_failure.py" $<TARGET_FILE:errors-validation-mutant-${i}> ${case})
 endforeach()
endif()
