cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(errors_propagation C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(EPR_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(EPR_PRIVATE "${EPR_DIR}/src")
set(EPR_ERRORS "${EPR_DIR}/../src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
  set(EPR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/errors")
 else()
  set(EPR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(EPR_PRIVATE "${EPR_ASSESSOR}/propagation")
 set(EPR_ERRORS "${EPR_ASSESSOR}/errors.c")
endif()
if(CMASTERY_EPR_LEARNER)
 set(EPR_PRIVATE "${EPR_DIR}/src")
 set(EPR_ERRORS "${EPR_DIR}/../src/errors.c")
endif()
function(epr_target name)
 add_executable(${name} "${EPR_DIR}/tests/tests.c" "${EPR_PRIVATE}/propagation.c" "${EPR_ERRORS}")
 target_include_directories(${name} PRIVATE "${EPR_PRIVATE}" "${EPR_DIR}/include" "${EPR_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
epr_target(errors-propagation)
foreach(case prop_ordinary prop_stream prop_macro prop_base)
 add_test(NAME errors.${case} COMMAND errors-propagation ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(EPR_CASES prop_macro prop_stream prop_stream prop_stream prop_macro prop_ordinary prop_ordinary prop_base prop_base)
 foreach(i RANGE 1 9)
  epr_target(errors-propagation-mutant-${i})
  if(i LESS 8)
   target_compile_definitions(errors-propagation-mutant-${i} PRIVATE EPR_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-5")
   target_compile_definitions(errors-propagation-mutant-${i} PRIVATE ER_MUTANT=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET EPR_CASES ${index} case)
  add_test(NAME errors.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${EPR_DIR}/expect_failure.py" $<TARGET_FILE:errors-propagation-mutant-${i}> ${case})
 endforeach()
endif()
