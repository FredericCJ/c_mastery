cmake_minimum_required(VERSION 3.20)
project(corpus_index C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or corpus_index assessor root")
option(CMASTERY_SANITIZE "Address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL src/index.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_index/fenwick.c")
  set(CI_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/corpus_index")
 else()
  set(CI_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(IMPL "${CI_ASSESSOR}/fenwick.c" "${CI_ASSESSOR}/segment.c" "${CI_ASSESSOR}/interval.c" "${CI_ASSESSOR}/suffix.c")
endif()
function(ci_program target driver)
 add_executable(${target} ${driver} ${IMPL})
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
ci_program(corpus-index-tests tests.c)
ci_program(corpus-index-driver driver.c)
ci_program(corpus-index-benchmark benchmark.c)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
set(CI_CASES fenwick segment interval suffix)
set(CI_ORDINAL 0)
foreach(case IN LISTS CI_CASES)
 math(EXPR CI_ORDINAL "${CI_ORDINAL}+1")
 add_test(NAME corpus_index.${case} COMMAND corpus-index-tests ${CI_ORDINAL})
 set_tests_properties(corpus_index.${case} PROPERTIES TIMEOUT 30)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant 1 2)
   math(EXPR mutation "(${CI_ORDINAL}-1)*2+${variant}")
   ci_program(corpus-index-mutant-${mutation} tests.c)
   target_compile_definitions(corpus-index-mutant-${mutation} PRIVATE CI_MUTANT=${mutation})
   add_test(NAME corpus_index.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject_mutant.py" $<TARGET_FILE:corpus-index-mutant-${mutation}> ${CI_ORDINAL})
  endforeach()
 endif()
endforeach()
