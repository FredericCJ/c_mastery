cmake_minimum_required(VERSION 3.20)
project(c_mastery_b04_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor family directory")
option(CMASTERY_SANITIZE "Host ASan/UBSan only" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor root must be absolute")
 endif()
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/b04_depth/depth.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/b04_depth/depth.c")
 else()
 set(IMPL "${CMASTERY_ASSESSOR_ROOT}/depth.c")
 endif()
endif()
add_library(depth STATIC "${IMPL}")
target_include_directories(depth PUBLIC include)
add_executable(depth-tests tests/tests.c)
target_link_libraries(depth-tests PRIVATE depth)
enable_testing()
foreach(case serial certificate framing dma memory)
 add_test(NAME b04_depth.${case} COMMAND depth-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  add_executable(mutant-${case} tests/tests.c "${IMPL}")
  target_include_directories(mutant-${case} PRIVATE include)
  string(TOUPPER ${case} upper)
  target_compile_definitions(mutant-${case} PRIVATE B4_MUTATE_${upper})
  add_test(NAME b04_depth.mutation.${case} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}> ${case})
 endif()
endforeach()
