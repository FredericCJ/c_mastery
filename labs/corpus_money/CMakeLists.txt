cmake_minimum_required(VERSION 3.16)
project(corpus_money C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CM_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/money.c" CACHE FILEPATH "Explicit completed component")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private provider root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_money/money.c")
    set(CM_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/corpus_money")
  else()
    set(CM_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
  endif()
  set(CM_PROVIDER "${CM_PRIVATE}/money.c")
endif()
if(NOT EXISTS "${CM_PROVIDER}")
  message(FATAL_ERROR "Supply explicit completed money provider")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(money STATIC "${CM_PROVIDER}")
target_include_directories(money PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(money_tests tests.c)
target_link_libraries(money_tests PRIVATE money)
add_executable(money_driver driver.c)
target_link_libraries(money_driver PRIVATE money)
enable_testing()
foreach(anchor currency rounding allocation settlement)
  add_test(NAME corpus_money.${anchor} COMMAND money_tests ${anchor})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME corpus_money.oracle COMMAND "${Python3_EXECUTABLE}" "${CM_PRIVATE}/oracle.py" $<TARGET_FILE:money_driver>)
  set_tests_properties(corpus_money.oracle PROPERTIES TIMEOUT 180)
  foreach(m RANGE 1 8)
    add_executable(money_mutant_${m} tests.c "${CM_PROVIDER}")
    target_include_directories(money_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(money_mutant_${m} PRIVATE CM_MUTANT=${m})
    if(m LESS 3)
      set(anchor currency)
    elseif(m LESS 5)
      set(anchor rounding)
    elseif(m LESS 7)
      set(anchor allocation)
    else()
      set(anchor settlement)
    endif()
    add_test(NAME corpus_money.mutation.${anchor}.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:money_mutant_${m}> ${anchor})
  endforeach()
endif()
