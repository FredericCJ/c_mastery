cmake_minimum_required(VERSION 3.20)
project(corpus_treiber C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/treiber.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_treiber/treiber.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/corpus_treiber/treiber.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/treiber.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/treiber.c")
 else()
  message(FATAL_ERROR "Supply the explicit completed corpus_treiber implementation")
 endif()
endif()
function(stack_target name short mutation)
 add_executable(${name} tests.c "${IMPL}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_definitions(${name} PRIVATE TS_MUTANT=${mutation})
 if(short)
  target_compile_definitions(${name} PRIVATE TS_GEN_MAX=3)
 endif()
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_libraries(${name} PRIVATE Threads::Threads)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=thread -no-pie)
 endif()
endfunction()
stack_target(treiber 0 0)
stack_target(treiber_short 1 0)
add_test(NAME corpus_treiber.stack COMMAND treiber stack)
add_test(NAME corpus_treiber.reclaim COMMAND treiber reclaim)
add_test(NAME corpus_treiber.retirement COMMAND treiber_short reclaim)
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutant 1 2 3 4)
  stack_target(treiber_mutant${mutant} 0 ${mutant})
  if(mutant LESS 3)
   set(anchor stack)
  else()
   set(anchor reclaim)
  endif()
  add_test(NAME corpus_treiber.mutation.${anchor}.${mutant}
   COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:treiber_mutant${mutant}> ${anchor})
  if(mutant GREATER 2)
   stack_target(treiber_short_mutant${mutant} 1 ${mutant})
   add_test(NAME corpus_treiber.mutation.retirement.${mutant}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:treiber_short_mutant${mutant}> reclaim)
  endif()
 endforeach()
endif()
