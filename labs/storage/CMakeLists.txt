cmake_minimum_required(VERSION 3.20)
project(cmastery_storage C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/storage directory; empty builds learner source")
option(CMASTERY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMASTERY_RUNTIME_ROOT "" CACHE PATH "Optional explicit labs/runtime path for executed VM integration")
set(CMASTERY_RUNTIME_ASSESSOR_ROOT "" CACHE PATH "Optional explicit assessor/runtime path")
enable_testing()
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
  set(ENGINE "${CMASTERY_ASSESSOR_ROOT}/store.c")
  if(NOT EXISTS "${ENGINE}")
    message(FATAL_ERROR "Explicit assessor source does not exist")
  endif()
endif()
set(ADAPTER "")
if(UNIX)
 set(ADAPTER src/store_file.c)
endif()
function(storage_target target)
 add_executable(${target} ${ENGINE} tests/test_store.c ${ADAPTER})
 target_include_directories(${target} PRIVATE include)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZERS)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
 endif()
endfunction()
storage_target(storage_tests)
if(CMASTERY_RUNTIME_ROOT)
 set(RUNTIME_ENGINE "${CMASTERY_RUNTIME_ROOT}/src/cvm.c")
 if(CMASTERY_RUNTIME_ASSESSOR_ROOT)
  set(RUNTIME_ENGINE "${CMASTERY_RUNTIME_ASSESSOR_ROOT}/cvm.c")
 endif()
 if(NOT EXISTS "${RUNTIME_ENGINE}")
  message(FATAL_ERROR "Explicit runtime dependency source does not exist")
 endif()
 target_sources(storage_tests PRIVATE "${CMASTERY_RUNTIME_ROOT}/adapters/store_host.c" "${RUNTIME_ENGINE}")
 target_include_directories(storage_tests PRIVATE "${CMASTERY_RUNTIME_ROOT}/include" "${CMASTERY_RUNTIME_ROOT}/adapters")
 target_compile_definitions(storage_tests PRIVATE STORAGE_WITH_RUNTIME=1)
 add_test(NAME storage.vm COMMAND storage_tests vm)
 set_tests_properties(storage.vm PROPERTIES TIMEOUT 30)
endif()
foreach(case codec initial_image cache lookup insertion deletion transaction publish checkpoint_order format tree differential crash checkpoint recovery corruption capacity file)
 add_test(NAME storage.${case} COMMAND storage_tests ${case})
 set_tests_properties(storage.${case} PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutation NO_COMMIT_FLUSH EARLY_RECLAIM BAD_SEPARATOR)
  storage_target(storage_mutant_${mutation})
  target_compile_definitions(storage_mutant_${mutation} PRIVATE STORAGE_MUTANT_${mutation}=1)
 endforeach()
 add_test(NAME storage.mutation_no_commit_flush COMMAND storage_mutant_NO_COMMIT_FLUSH crash)
 add_test(NAME storage.mutation_early_reclaim COMMAND storage_mutant_EARLY_RECLAIM checkpoint)
 add_test(NAME storage.mutation_bad_separator COMMAND storage_mutant_BAD_SEPARATOR tree)
 set_tests_properties(storage.mutation_no_commit_flush storage.mutation_early_reclaim storage.mutation_bad_separator PROPERTIES WILL_FAIL TRUE TIMEOUT 120)
endif()
