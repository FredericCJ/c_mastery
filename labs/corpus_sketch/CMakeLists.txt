cmake_minimum_required(VERSION 3.16)
project(corpus_sketch C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SK_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src" CACHE PATH "Explicit completed providers")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Private provider root")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_sketch")
    set(SK_PROVIDER "${CMASTERY_ASSESSOR_ROOT}/corpus_sketch")
  else()
    set(SK_PROVIDER "${CMASTERY_ASSESSOR_ROOT}")
  endif()
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(sketch STATIC "${SK_PROVIDER}/count_min.c" "${SK_PROVIDER}/hyperloglog.c" "${SK_PROVIDER}/minhash.c")
target_include_directories(sketch PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(sketch_driver driver.c)
target_link_libraries(sketch_driver PRIVATE sketch)
enable_testing()
set(index 0)
foreach(name count_min hyperloglog minhash)
  math(EXPR index "${index}+1")
  add_executable(sketch_${name} tests.c)
  target_compile_definitions(sketch_${name} PRIVATE SK_TEST=${index})
  target_link_libraries(sketch_${name} PRIVATE sketch)
  add_test(NAME corpus_sketch.${name} COMMAND sketch_${name})
  if(CMASTERY_ASSESSOR_ROOT)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(m RANGE 1 2)
      add_executable(sketch_${name}_mutant_${m} tests.c "${SK_PROVIDER}/${name}.c")
      target_include_directories(sketch_${name}_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
      target_compile_definitions(sketch_${name}_mutant_${m} PRIVATE SK_TEST=${index} SK_MUTANT=${m})
      add_test(NAME corpus_sketch.mutation.${name}.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:sketch_${name}_mutant_${m}>)
    endforeach()
  endif()
endforeach()
