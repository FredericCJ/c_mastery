cmake_minimum_required(VERSION 3.16)
project(corpus_query C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(QI_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/query.c" CACHE FILEPATH "Explicit completed component")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private provider root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_query/query.c")
    set(QI_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/corpus_query")
  else()
    set(QI_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
  endif()
  set(QI_PROVIDER "${QI_PRIVATE}/query.c")
endif()
if(NOT EXISTS "${QI_PROVIDER}")
  message(FATAL_ERROR "Supply explicit completed query provider")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(query STATIC "${QI_PROVIDER}")
target_include_directories(query PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(query_tests tests.c)
target_link_libraries(query_tests PRIVATE query)
add_executable(query_driver driver.c)
target_link_libraries(query_driver PRIVATE query)
enable_testing()
foreach(anchor bitmap postings zones)
  add_test(NAME corpus_query.${anchor} COMMAND query_tests ${anchor})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME corpus_query.oracle COMMAND "${Python3_EXECUTABLE}" "${QI_PRIVATE}/oracle.py" $<TARGET_FILE:query_driver>)
  set_tests_properties(corpus_query.oracle PROPERTIES TIMEOUT 180)
  foreach(m RANGE 1 6)
    add_executable(query_mutant_${m} tests.c "${QI_PROVIDER}")
    target_include_directories(query_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(query_mutant_${m} PRIVATE QI_MUTANT=${m})
    if(m LESS 3)
      set(anchor bitmap)
    elseif(m LESS 5)
      set(anchor postings)
    else()
      set(anchor zones)
    endif()
    add_test(NAME corpus_query.mutation.${anchor}.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:query_mutant_${m}> ${anchor})
  endforeach()
endif()
