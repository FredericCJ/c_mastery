cmake_minimum_required(VERSION 3.20)
project(text_composition C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or family assessor root")
set(CMASTERY_PARSING_DEPTH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../parsing_depth" CACHE PATH "Actual prerequisite learner workspace")
set(CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT "" CACHE PATH "Explicit completed prerequisite provider")
option(CMASTERY_SANITIZE "Address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL src/text.c)
set(PROVIDER "${CMASTERY_PARSING_DEPTH_ROOT}/src/parsing.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/text_composition/recovery.c")
  set(TC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/text_composition")
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/parsing_depth")
 else()
  set(TC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/../parsing_depth")
 endif()
 set(IMPL "${TC_ASSESSOR}/recovery.c" "${TC_ASSESSOR}/layout.c" "${TC_ASSESSOR}/template.c")
 set(PROVIDER "${PD_ASSESSOR}/pratt.c")
endif()
if(CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/parsing_depth/pratt.c")
  set(PROVIDER "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/parsing_depth/pratt.c")
 else()
  set(PROVIDER "${CMASTERY_PARSING_DEPTH_ASSESSOR_ROOT}/pratt.c")
 endif()
endif()
if(NOT EXISTS "${CMASTERY_PARSING_DEPTH_ROOT}/parsing.h" OR NOT EXISTS "${PROVIDER}")
 message(FATAL_ERROR "Supply the actual parsing_depth prerequisite workspace/provider")
endif()
function(tc_program target driver)
 add_executable(${target} ${driver} ${IMPL} "${PROVIDER}")
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMASTERY_PARSING_DEPTH_ROOT}")
endfunction()
tc_program(text-composition-tests tests.c)
tc_program(text-composition-driver driver.c)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
set(ORDINAL 0)
foreach(case recovery layout template)
 math(EXPR ORDINAL "${ORDINAL}+1")
 add_test(NAME text_composition.${case} COMMAND text-composition-tests ${ORDINAL})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant 1 2)
   math(EXPR mutation "(${ORDINAL}-1)*2+${variant}")
   tc_program(text-composition-mutant-${mutation} tests.c)
   target_compile_definitions(text-composition-mutant-${mutation} PRIVATE TC_MUTANT=${mutation})
   add_test(NAME text_composition.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject_mutant.py" $<TARGET_FILE:text-composition-mutant-${mutation}> ${ORDINAL})
  endforeach()
 endif()
endforeach()
