cmake_minimum_required(VERSION 3.20)
project(c_mastery_systems_certificates C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or family")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/systems_certificates.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}")
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/systems_certificates/systems_certificates.c")
  set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}/systems_certificates")
 endif()
 set(IMPL "${ASSESSOR_FAMILY}/systems_certificates.c")
endif()
add_library(systems_certificates STATIC "${IMPL}")
target_include_directories(systems_certificates PUBLIC include)
add_executable(systems_certificates-tests tests/tests.c)
target_link_libraries(systems_certificates-tests PRIVATE systems_certificates)
enable_testing()
foreach(case tree matches clock)
 add_test(NAME systems_certificates.${case} COMMAND systems_certificates-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(number 1 2)
   string(TOUPPER ${case} upper)
   set(mutation "SC_MUTATE_${upper}")
   if(number EQUAL 2)
    string(APPEND mutation 2)
   endif()
   add_executable(mutant-${case}-${number} tests/tests.c "${IMPL}")
   target_include_directories(mutant-${case}-${number} PRIVATE include)
   target_compile_definitions(mutant-${case}-${number} PRIVATE ${mutation})
   add_test(NAME systems_certificates.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}> ${case})
  endforeach()
 endif()
endforeach()
