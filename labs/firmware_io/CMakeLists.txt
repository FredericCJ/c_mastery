cmake_minimum_required(VERSION 3.20)
project(cmastery_firmware_io C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/firmware_io directory")
set(FW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor root must be absolute")
 endif()
 set(FW_SOURCE "${CMASTERY_ASSESSOR_ROOT}")
endif()
add_library(firmware_io STATIC "${FW_SOURCE}/firmware_io.c")
target_include_directories(firmware_io PUBLIC include)
target_compile_options(firmware_io PRIVATE -Wall -Wextra -Wpedantic)
add_executable(firmware_contracts tests/contracts.c)
target_link_libraries(firmware_contracts PRIVATE firmware_io)
enable_testing()
foreach(case startup vectors gpio uart spi i2c timer adc dma ethernet rs485 watchdog)
 add_test(NAME firmware_io.${case} COMMAND firmware_contracts ${case})
 set_tests_properties(firmware_io.${case} PROPERTIES TIMEOUT 5 LABELS "host;register-model;not-physical")
endforeach()
find_program(ARM_GCC arm-none-eabi-gcc)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(ARM_GCC)
 add_test(NAME firmware_io.cortex_compile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cortex_compile.py" --cc "${ARM_GCC}" --source "${FW_SOURCE}/firmware_io.c" --include "${CMAKE_CURRENT_SOURCE_DIR}/include")
endif()
if(CMASTERY_ASSESSOR_ROOT)
 include(tests/mutations.cmake)
endif()
