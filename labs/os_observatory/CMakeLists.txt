cmake_minimum_required(VERSION 3.20)
project(cmastery_os_observatory C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OS_IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
  set(OS_IMPL "${CMASTERY_ASSESSOR_ROOT}")
endif()
add_executable(os_tests tests/test_os.c "${OS_IMPL}/oslab.c" "${OS_IMPL}/platform.c")
target_include_directories(os_tests PRIVATE include)
target_compile_options(os_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(os_tests PRIVATE Threads::Threads)
foreach(case schedulers fairness clock process thread descriptor signal memory io)
  add_test(NAME os_${case} COMMAND os_tests ${case})
  set_tests_properties(os_${case} PROPERTIES TIMEOUT 10)
endforeach()
add_executable(os_linux_compare "${OS_IMPL}/linux_compare.c")
add_executable(os_policy_probe policy_probe.c)
target_compile_options(os_policy_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(os_linux_compare PRIVATE Threads::Threads)
target_compile_options(os_linux_compare PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME os_linux_comparison COMMAND os_linux_compare)
set_tests_properties(os_linux_comparison PROPERTIES TIMEOUT 15)
add_test(NAME os_mutations COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/mutations.py"
  --compiler "${CMAKE_C_COMPILER}" --impl "${OS_IMPL}" --lab "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMASTERY_FREERTOS_ROOT)
  set(FR "${CMASTERY_FREERTOS_ROOT}")
  set(PORT "${FR}/portable/ThirdParty/GCC/Posix")
  add_executable(os_freertos_compare "${OS_IMPL}/freertos_compare.c"
    "${FR}/tasks.c" "${FR}/queue.c" "${FR}/list.c" "${PORT}/port.c" "${PORT}/utils/wait_for_event.c")
  target_include_directories(os_freertos_compare PRIVATE compare "${FR}/include" "${PORT}" "${PORT}/utils")
  target_link_libraries(os_freertos_compare PRIVATE Threads::Threads)
  target_compile_options(os_freertos_compare PRIVATE -UNDEBUG)
  add_test(NAME os_freertos_comparison COMMAND os_freertos_compare)
  set_tests_properties(os_freertos_comparison PROPERTIES TIMEOUT 15)
endif()
