cmake_minimum_required(VERSION 3.20)
project(c_mastery_protocols C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor family directory")
option(CMASTERY_SANITIZE "Host ASan/UBSan" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/protocols.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/protocols/protocols.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/protocols/protocols.c")
 else()
 set(IMPL "${CMASTERY_ASSESSOR_ROOT}/protocols.c")
 endif()
endif()
add_library(protocols STATIC "${IMPL}")
target_include_directories(protocols PUBLIC include)
add_executable(protocols-tests tests/tests.c)
target_link_libraries(protocols-tests PRIVATE protocols)
enable_testing()
foreach(case typestate builder session hsm active thread behavior driver)
 add_test(NAME protocols.${case} COMMAND protocols-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(number 1 2)
   set(mutation "TS_MUTATE_")
   string(TOUPPER ${case} upper)
   string(APPEND mutation ${upper})
   if(number EQUAL 2)
    string(APPEND mutation 2)
   endif()
   add_executable(mutant-${case}-${number} tests/tests.c "${IMPL}")
   target_include_directories(mutant-${case}-${number} PRIVATE include)
   target_compile_definitions(mutant-${case}-${number} PRIVATE ${mutation})
   add_test(NAME protocols.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}> ${case})
  endforeach()
 endif()
endforeach()
