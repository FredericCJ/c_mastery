cmake_minimum_required(VERSION 3.20)
project(b02_fixed_point_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_B02_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth" CACHE PATH "Explicit completed original B02 provider")
set(BF_CORE "${CMAKE_CURRENT_LIST_DIR}/fixed_point.c")
set(BF_B02 "${CMASTERY_B02_ROOT}/src/depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(BF_B02 "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
 if(NOT CMASTERY_BF_LEARNER)
  set(BF_CORE "${CMASTERY_ASSESSOR_ROOT}/b02_followthrough/fixed_point/fixed_point.c")
 endif()
endif()
foreach(source "${BF_CORE}" "${BF_B02}")
 if(NOT EXISTS "${source}")
  message(FATAL_ERROR "Supply the explicit completed fixed_point and B02 implementations")
 endif()
endforeach()
function(bf_target name)
 add_executable(${name} tests.c "${BF_CORE}" "${BF_B02}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_B02_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
bf_target(b02-fixed_point)
foreach(case q7 even q11 units scale16 mixed cascade)
 add_test(NAME b02_followthrough.${case} COMMAND b02-fixed_point ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_BF_LEARNER)
 set(cases q7 even q11 q11 scale16 mixed cascade)
 foreach(i RANGE 1 7)
  bf_target(b02-fixed_point-mutant-${i})
  target_compile_definitions(b02-fixed_point-mutant-${i} PRIVATE BF_MUTANT=${i})
  if(i EQUAL 1)
   target_compile_definitions(b02-fixed_point-mutant-${i} PRIVATE B2_MUTATE_Q7=1)
  endif()
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME b02_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:b02-fixed_point-mutant-${i}> ${case})
 endforeach()
endif()
