cmake_minimum_required(VERSION 3.20)
project(b02_map_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_B02_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth" CACHE PATH "Explicit original B02 public provider")
set(CMASTERY_CD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../containers_depth" CACHE PATH "Explicit actual container public provider")
set(BF_CORE "${CMAKE_CURRENT_LIST_DIR}/map.c")
set(BF_B02 "${CMASTERY_B02_ROOT}/src/depth.c")
set(BF_CD "${CMASTERY_CD_ROOT}/learner.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(BF_B02 "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
 set(BF_CD "${CMASTERY_ASSESSOR_ROOT}/containers_depth/indexed.c")
 if(NOT CMASTERY_BF_LEARNER)
  set(BF_CORE "${CMASTERY_ASSESSOR_ROOT}/b02_followthrough/map/map.c")
 endif()
endif()
foreach(source "${BF_CORE}" "${BF_B02}" "${BF_CD}")
 if(NOT EXISTS "${source}")
  message(FATAL_ERROR "Supply the explicit map and container implementations")
 endif()
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/instrument.py" "${BF_B02}" "${CMAKE_CURRENT_BINARY_DIR}/b02_instrumented.c" COMMAND_ERROR_IS_FATAL ANY)
 set(BF_B02 "${CMAKE_CURRENT_BINARY_DIR}/b02_instrumented.c")
endif()
function(bf_target name)
 add_executable(${name} tests.c "${BF_CORE}" "${BF_B02}" "${BF_CD}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_B02_ROOT}/include" "${CMASTERY_CD_ROOT}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_ASSESSOR_ROOT)
  target_compile_definitions(${name} PRIVATE BF_INSTRUMENTED=1)
 endif()
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
bf_target(b02-map)
foreach(case reachability rehash keys)
 add_test(NAME b02_followthrough.${case} COMMAND b02-map ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_BF_LEARNER)
 foreach(i RANGE 1 3)
  bf_target(b02-map-mutant-${i})
  target_compile_definitions(b02-map-mutant-${i} PRIVATE BF_MUTANT=${i})
  if(i EQUAL 1)
   target_compile_definitions(b02-map-mutant-${i} PRIVATE B2_MUTATE_MAP=1)
   set(case reachability)
  else()
   set(case rehash)
  endif()
  add_test(NAME b02_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:b02-map-mutant-${i}> ${case})
 endforeach()
endif()
