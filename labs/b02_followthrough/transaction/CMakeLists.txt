cmake_minimum_required(VERSION 3.20)
project(b02_transaction_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_B02_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth" CACHE PATH "Explicit completed original B02 provider")
set(BF_CORE "${CMAKE_CURRENT_LIST_DIR}/transaction.c")
set(BF_B02 "${CMASTERY_B02_ROOT}/src/depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(BF_B02 "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
 if(NOT CMASTERY_BF_LEARNER)
  set(BF_CORE "${CMASTERY_ASSESSOR_ROOT}/b02_followthrough/transaction/transaction.c")
 endif()
endif()
foreach(source "${BF_CORE}" "${BF_B02}")
 if(NOT EXISTS "${source}")
  message(FATAL_ERROR "Supply the explicit completed transaction and B02 implementations")
 endif()
endforeach()
function(bf_target name)
 add_executable(${name} tests.c "${BF_CORE}" "${BF_B02}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_B02_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
bf_target(b02-transaction)
foreach(case styles recovery registration admission)
 add_test(NAME b02_followthrough.${case} COMMAND b02-transaction ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_BF_LEARNER)
 set(cases styles styles recovery recovery recovery registration)
 foreach(i RANGE 1 6)
  bf_target(b02-transaction-mutant-${i})
  target_compile_definitions(b02-transaction-mutant-${i} PRIVATE BF_MUTANT=${i})
  if(i EQUAL 1)
   target_compile_definitions(b02-transaction-mutant-${i} PRIVATE B2_MUTATE_TRANSACTION=1)
  endif()
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME b02_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:b02-transaction-mutant-${i}> ${case})
 endforeach()
endif()
