cmake_minimum_required(VERSION 3.20)
project(b02_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or b02_depth directory")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
set(IMPL src/depth.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/depth.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_executable(b02-tests tests/tests.c ${IMPL})
if(CMASTERY_ASSESSOR_ROOT)
 get_filename_component(ASSESSOR_DIR "${IMPL}" DIRECTORY)
 target_include_directories(b02-tests BEFORE PRIVATE "${ASSESSOR_DIR}")
endif()
target_include_directories(b02-tests PRIVATE include)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(b02-oracle-driver tests/oracle_driver.c ${IMPL})
target_include_directories(b02-oracle-driver PRIVATE include)
add_test(NAME b02_depth.oracle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/oracle.py" $<TARGET_FILE:b02-oracle-driver>)
set_tests_properties(b02_depth.oracle PROPERTIES TIMEOUT 60)
foreach(case macro parse map sort arena q7 transaction variant)
 add_test(NAME b02_depth.${case} COMMAND b02-tests ${case})
 set_tests_properties(b02_depth.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(b02-mutant-${case} tests/tests.c ${IMPL})
  target_include_directories(b02-mutant-${case} BEFORE PRIVATE "${ASSESSOR_DIR}")
  target_include_directories(b02-mutant-${case} PRIVATE include)
  string(TOUPPER ${case} CASE_UPPER)
  target_compile_definitions(b02-mutant-${case} PRIVATE B2_MUTATE_${CASE_UPPER})
  add_test(NAME b02_depth.mutation.${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:b02-mutant-${case}> ${case})
  set_tests_properties(b02_depth.mutation.${case} PROPERTIES TIMEOUT 20)
 endif()
endforeach()
