cmake_minimum_required(VERSION 3.16)
project(corpus_model C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CM_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/model.c" CACHE FILEPATH "Completed domain/session provider")
set(CM_PARSER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/parser_dependency.c" CACHE FILEPATH "Accepted X-T15-012 Pratt provider")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "Hosted sanitizers" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_model/model.c")
    set(CM_PROVIDER "${CMASTERY_ASSESSOR_ROOT}/corpus_model/model.c")
    set(CM_PARSER_SOURCE "${CMASTERY_ASSESSOR_ROOT}/parsing_depth/pratt.c")
  else()
    set(CM_PROVIDER "${CMASTERY_ASSESSOR_ROOT}/model.c")
    set(CM_PARSER_SOURCE "${CMASTERY_ASSESSOR_ROOT}/../parsing_depth/pratt.c")
  endif()
endif()
if(NOT EXISTS "${CM_PROVIDER}" OR NOT EXISTS "${CM_PARSER_SOURCE}")
  message(FATAL_ERROR "Supply explicit completed model and accepted Pratt providers")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(model STATIC "${CM_PROVIDER}" "${CM_PARSER_SOURCE}")
target_include_directories(model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/provider")
foreach(name tests driver repl session_driver)
  add_executable(model_${name} ${name}.c)
  target_link_libraries(model_${name} PRIVATE model)
endforeach()
enable_testing()
add_test(NAME corpus_model.contract COMMAND model_tests)
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(m RANGE 1 6)
    add_executable(model_mutant_${m} tests.c "${CM_PROVIDER}" "${CM_PARSER_SOURCE}")
    target_include_directories(model_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/provider")
    target_compile_definitions(model_mutant_${m} PRIVATE CM_MUTANT=${m})
    add_test(NAME corpus_model.mutation.contract.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:model_mutant_${m}>)
  endforeach()
endif()
