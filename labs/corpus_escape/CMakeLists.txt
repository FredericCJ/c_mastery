cmake_minimum_required(VERSION 3.16)
project(corpus_escape C)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CMASTERY_SANITIZE "Host address/undefined checks" OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit external completed implementation")
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_escape/escape.c")
  set(source "${CMASTERY_ASSESSOR_ROOT}/corpus_escape/escape.c")
 else()
  set(source "${CMASTERY_ASSESSOR_ROOT}/escape.c")
 endif()
else()
 set(source src/escape.c)
endif()
add_library(core STATIC ${source})
target_include_directories(core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(escape_tests tests.c)
target_link_libraries(escape_tests core)
add_test(NAME corpus_escape.escape COMMAND escape_tests)
add_executable(escape_driver driver.c)
target_link_libraries(escape_driver core)
if(CMASTERY_ASSESSOR_ROOT)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 foreach(n RANGE 1 2)
  add_library(mutant${n} STATIC ${source})
  target_include_directories(mutant${n} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(mutant${n} PRIVATE ES_MUTANT=${n})
  add_executable(mutant_test${n} tests.c)
  target_link_libraries(mutant_test${n} mutant${n})
  add_test(NAME corpus_escape.mutation.escape.${n} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:mutant_test${n}>)
 endforeach()
 # Defaults apply to repository layout only. Exported workspaces supply explicit
 # public roots and completed source files; no private provider is copied.
 set(CMASTERY_TEXT_COMPOSITION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../text_composition" CACHE PATH "Public template prerequisite")
 set(CMASTERY_CORPUS_EVENTS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../corpus_events" CACHE PATH "Public events prerequisite")
 set(CMASTERY_PARSING_DEPTH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../parsing_depth" CACHE PATH "Public template header dependency")
 set(CMASTERY_TEMPLATE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../assessor/text_composition/template.c" CACHE FILEPATH "Explicit completed template implementation")
 set(CMASTERY_EVENTS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../assessor/corpus_events/events.c" CACHE FILEPATH "Explicit completed events implementation")
 foreach(bits 16 1)
  add_executable(composition${bits} composition.c ${CMASTERY_TEMPLATE_SOURCE} ${CMASTERY_EVENTS_SOURCE})
  target_include_directories(composition${bits} PRIVATE "${CMASTERY_TEXT_COMPOSITION_ROOT}" "${CMASTERY_CORPUS_EVENTS_ROOT}" "${CMASTERY_PARSING_DEPTH_ROOT}")
  target_compile_definitions(composition${bits} PRIVATE EV_TEXT=${bits})
  target_link_libraries(composition${bits} core)
  add_test(NAME corpus_escape.composition${bits} COMMAND composition${bits})
 endforeach()
endif()
