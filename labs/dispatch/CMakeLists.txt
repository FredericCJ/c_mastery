cmake_minimum_required(VERSION 3.20)
project(dispatch C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
set(CMASTERY_DOD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../dod" CACHE PATH "Accepted dod learner workspace")
set(CMASTERY_DOD_ASSESSOR_ROOT "" CACHE PATH "Explicit dod assessor directory or common assessor root")
set(IMPL src/dispatch.c)
set(STORAGE "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/dispatch/dispatch.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/dispatch/dispatch.c")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/dispatch.c")
 endif()
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_DOD_ASSESSOR_ROOT}/dod/dod.c")
  set(STORAGE "${CMASTERY_DOD_ASSESSOR_ROOT}/dod/dod.c")
 else()
  set(STORAGE "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(variant plugin1 plugin2 bad)
 add_library(dispatch-${variant} SHARED tests/plugin.c)
 target_include_directories(dispatch-${variant} PRIVATE include "${CMASTERY_DOD_ROOT}/include")
endforeach()
target_compile_definitions(dispatch-plugin2 PRIVATE PLUGIN_SCHEMA=2)
target_compile_definitions(dispatch-bad PRIVATE PLUGIN_ABI=9)
function(dispatch_program target)
 add_executable(${target} tests/tests.c ${IMPL} ${STORAGE})
 target_include_directories(${target} PRIVATE include "${CMASTERY_DOD_ROOT}/include")
 target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
 if(CMASTERY_SANITIZE)
  target_compile_options(${target} PRIVATE -fno-pie)
  target_link_options(${target} PRIVATE -no-pie)
 endif()
endfunction()
dispatch_program(dispatch-tests)
enable_testing()
foreach(probe qualifier_reject callback_reject empty_struct_reject)
 add_test(NAME dispatch.constraint.${probe} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_reject.py" "${CMAKE_C_COMPILER}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/${probe}.c")
endforeach()
foreach(case traits fat dictionary closures erased multiple intrusive modules capabilities plugin reload)
 add_test(NAME dispatch.${case} COMMAND dispatch-tests ${case} $<TARGET_FILE:dispatch-plugin1> $<TARGET_FILE:dispatch-plugin2> $<TARGET_FILE:dispatch-bad>)
 set_tests_properties(dispatch.${case} PROPERTIES TIMEOUT 30)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2)
   dispatch_program(dispatch-mutant-${case}-${mutant})
   string(TOUPPER ${case} CASE_UPPER)
   target_compile_definitions(dispatch-mutant-${case}-${mutant} PRIVATE DS_MUTATE_${CASE_UPPER}${mutant})
   add_test(NAME dispatch.mutation.${case}.${mutant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:dispatch-mutant-${case}-${mutant}> ${case} $<TARGET_FILE:dispatch-plugin1> $<TARGET_FILE:dispatch-plugin2> $<TARGET_FILE:dispatch-bad>)
  endforeach()
 endif()
endforeach()
