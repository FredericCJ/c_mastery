cmake_minimum_required(VERSION 3.20)
project(c_mastery_edu7_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or family")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/edu7_depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}")
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/edu7_depth/edu7_depth.c")
  set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}/edu7_depth")
 endif()
 set(IMPL "${ASSESSOR_FAMILY}/edu7_depth.c")
endif()
add_library(edu7_depth STATIC "${IMPL}")
target_include_directories(edu7_depth PUBLIC include)
add_executable(edu7_depth-tests tests/tests.c)
target_link_libraries(edu7_depth-tests PRIVATE edu7_depth)
enable_testing()
foreach(case grant offer session)
 add_test(NAME edu7_depth.${case} COMMAND edu7_depth-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(number 1 2)
   string(TOUPPER ${case} upper)
   set(mutation "E7_MUTATE_${upper}")
   if(number EQUAL 2)
    string(APPEND mutation 2)
   endif()
   add_executable(mutant-${case}-${number} tests/tests.c "${IMPL}")
   target_include_directories(mutant-${case}-${number} PRIVATE include)
   target_compile_definitions(mutant-${case}-${number} PRIVATE ${mutation})
   add_test(NAME edu7_depth.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}> ${case})
  endforeach()
 endif()
endforeach()
