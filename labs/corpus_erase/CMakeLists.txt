cmake_minimum_required(VERSION 3.16)
project(corpus_erase C CXX)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CMASTERY_SANITIZE "Host address/undefined checks" OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit external author provider")
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_erase/erase.c")
  set(source "${CMASTERY_ASSESSOR_ROOT}/corpus_erase/erase.c")
 else()
  set(source "${CMASTERY_ASSESSOR_ROOT}/erase.c")
 endif()
else()
 set(source src/erase.c)
endif()
add_library(core STATIC ${source})
target_include_directories(core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(erase_tests tests.c)
target_link_libraries(erase_tests core)
add_test(NAME corpus_erase.compact COMMAND erase_tests)
add_executable(erase_driver driver.c)
target_link_libraries(erase_driver core)
add_executable(cpp_comparison comparison.cpp)
add_test(NAME corpus_erase.cpp_comparison COMMAND cpp_comparison)
if(CMASTERY_ASSESSOR_ROOT)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 foreach(n RANGE 1 2)
  add_library(mutant${n} STATIC ${source})
  target_include_directories(mutant${n} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(mutant${n} PRIVATE ER_MUTANT=${n})
  add_executable(mutant_test${n} tests.c)
  target_link_libraries(mutant_test${n} mutant${n})
  add_test(NAME corpus_erase.mutation.compact.${n} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:mutant_test${n}>)
 endforeach()
endif()
