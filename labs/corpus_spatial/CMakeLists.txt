cmake_minimum_required(VERSION 3.16)
project(corpus_spatial C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CMASTERY_SANITIZE "Address/undefined sanitizers" OFF)
if(CMASTERY_SANITIZE)
  add_compile_options(-O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit completed implementation")
set(source "${CMAKE_CURRENT_SOURCE_DIR}/src/spatial.c")
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_spatial/spatial.c")
    set(source "${CMASTERY_ASSESSOR_ROOT}/corpus_spatial/spatial.c")
  elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/spatial.c")
    set(source "${CMASTERY_ASSESSOR_ROOT}/spatial.c")
  else()
    message(FATAL_ERROR "Supply the explicit completed corpus_spatial implementation")
  endif()
endif()
add_executable(spatial_tests tests.c "${source}")
target_include_directories(spatial_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(spatial_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
foreach(case RANGE 1 5)
  add_test(NAME corpus_spatial.${case} COMMAND spatial_tests ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(number RANGE 1 10)
    add_executable(spatial_mutant${number} tests.c "${source}")
    target_include_directories(spatial_mutant${number} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(spatial_mutant${number} PRIVATE SP_MUTANT=${number})
    target_compile_options(spatial_mutant${number} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
  foreach(case RANGE 1 5)
    foreach(control RANGE 1 2)
      math(EXPR number "(${case}-1)*2+${control}")
      add_test(NAME corpus_spatial.mutation.${case}.${control} COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:spatial_mutant${number}> ${case})
    endforeach()
  endforeach()
endif()
