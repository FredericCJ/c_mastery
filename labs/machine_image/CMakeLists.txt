cmake_minimum_required(VERSION 3.20)
project(cmastery_machine_image C ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/machine_image directory")
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor directory must be absolute")
 endif()
 set(IMPL "${CMASTERY_ASSESSOR_ROOT}")
endif()
add_library(machine_core STATIC "${IMPL}/abi.c" "${IMPL}/sum.S")
add_library(machine_bytes STATIC "${IMPL}/bytes.c")
add_library(machine_arithmetic STATIC "${IMPL}/wide.c")
foreach(target machine_bytes machine_arithmetic)
 target_include_directories(${target} PUBLIC include)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffreestanding -fno-builtin)
endforeach()
install(TARGETS machine_bytes machine_arithmetic ARCHIVE DESTINATION lib)
install(FILES include/machine.h DESTINATION include)
target_include_directories(machine_core PUBLIC include)
target_compile_options(machine_core PRIVATE -Wall -Wextra -Wpedantic -ffreestanding -fno-builtin)
add_executable(machine_tests tests/runtime.c tests/hostile.S)
target_link_libraries(machine_tests PRIVATE machine_core machine_bytes machine_arithmetic)
target_compile_options(machine_tests PRIVATE -UNDEBUG)
# Compile all learner C sources even when a particular unit is not yet selected.
add_library(machine_objects OBJECT "${IMPL}/archive.c" "${IMPL}/relocation.c" "${IMPL}/tls.c" "${IMPL}/entry.c" "${IMPL}/syscall.c" "${IMPL}/image.c")
target_compile_options(machine_objects PRIVATE -Wall -Wextra -Wpedantic)
add_executable(machine_audit "${IMPL}/audit.c")
enable_testing()
foreach(case abi hostile bytes wide)
 add_test(NAME machine_image.${case} COMMAND machine_tests ${case})
 set_tests_properties(machine_image.${case} PROPERTIES TIMEOUT 10)
endforeach()
foreach(case archive relocation tls budget entry syscall helpers byte_closure closure image)
 add_test(NAME machine_image.${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/artifacts.py" --case ${case} --cc "${CMAKE_C_COMPILER}" --impl "${IMPL}" --lab "${CMAKE_CURRENT_SOURCE_DIR}" --work "${CMAKE_CURRENT_BINARY_DIR}/artifacts/${case}" "--optimization=$<IF:$<CONFIG:Debug>,-O0,-O2>")
 set_tests_properties(machine_image.${case} PROPERTIES TIMEOUT 30)
endforeach()
