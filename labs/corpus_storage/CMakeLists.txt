cmake_minimum_required(VERSION 3.16)
project(corpus_storage C)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CMASTERY_SANITIZE "Host address and undefined checks" OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit outside learner provider")
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_storage/skip.c")
    set(provider "${CMASTERY_ASSESSOR_ROOT}/corpus_storage")
  else()
    set(provider "${CMASTERY_ASSESSOR_ROOT}")
  endif()
  set(sources "${provider}/skip.c" "${provider}/tree.c" "${provider}/lsm.c")
else()
  set(sources src/storage.c)
endif()
add_library(core STATIC ${sources})
target_include_directories(core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(storage_tests tests.c)
target_link_libraries(storage_tests core)
add_executable(storage_driver driver.c)
target_link_libraries(storage_driver core)
foreach(case skip tree lsm)
  add_test(NAME corpus_storage.${case} COMMAND storage_tests ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(n RANGE 1 6)
    if(n LESS 3)
      set(case skip)
    elseif(n LESS 5)
      set(case tree)
    else()
      set(case lsm)
    endif()
    add_library(mutant${n} STATIC ${sources})
    target_include_directories(mutant${n} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(mutant${n} PRIVATE SS_MUTANT=${n})
    add_executable(mutant_test${n} tests.c)
    target_link_libraries(mutant_test${n} mutant${n})
    math(EXPR variant "(${n}-1)%2+1")
    add_test(NAME corpus_storage.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:mutant_test${n}> ${case})
  endforeach()
endif()
