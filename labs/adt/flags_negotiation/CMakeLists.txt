cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_flags_negotiation C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADF_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADF_BASE "${ADF_DIR}/../src/adt.c")
set(ADF_CORE "${ADF_DIR}/flags_negotiation.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADF_BASE "${ADF_ASSESSOR}/adt.c")
 set(ADF_CORE "${ADF_ASSESSOR}/flags_negotiation/flags_negotiation.c")
endif()
if(CMASTERY_ADF_LEARNER)
 set(ADF_CORE "${ADF_DIR}/flags_negotiation.c")
 set(ADF_BASE "${ADF_DIR}/../src/adt.c")
endif()
function(adf_target name)
 add_executable(${name} "${ADF_DIR}/tests.c" ${ADF_CORE} ${ADF_BASE})
 target_include_directories(${name} PRIVATE "${ADF_DIR}" "${ADF_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(${name} PRIVATE -fsanitize=thread)
 endif()
endfunction()
adf_target(adt-flags_negotiation)
foreach(case flags_negotiation flags_sets flags_base)
 add_test(NAME adt.${case} COMMAND adt-flags_negotiation ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADF_CASES flags_negotiation flags_negotiation flags_sets flags_sets flags_base flags_base)
 foreach(i RANGE 1 6)
  adf_target(adt-flags-negotiation-mutant-${i})
  if(i LESS 5)
   target_compile_definitions(adt-flags-negotiation-mutant-${i} PRIVATE ADF_MUTATION=${i})
  else()
   set(old "${i}")
   target_compile_definitions(adt-flags-negotiation-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADF_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADF_DIR}/expect_failure.py" $<TARGET_FILE:adt-flags-negotiation-mutant-${i}> ${case})
 endforeach()
endif()
