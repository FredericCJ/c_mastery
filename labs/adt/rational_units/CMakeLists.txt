cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_rational_units C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADQ_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADQ_BASE "${ADQ_DIR}/../src/adt.c")
set(ADQ_CORE "${ADQ_DIR}/rational_units.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADQ_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADQ_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADQ_BASE "${ADQ_ASSESSOR}/adt.c")
 set(ADQ_CORE "${ADQ_ASSESSOR}/rational_units/rational_units.c")
endif()
if(CMASTERY_ADQ_LEARNER)
 set(ADQ_CORE "${ADQ_DIR}/rational_units.c")
 set(ADQ_BASE "${ADQ_DIR}/../src/adt.c")
endif()
function(ado_target name)
 add_executable(${name} "${ADQ_DIR}/tests.c" ${ADQ_CORE} ${ADQ_BASE})
 target_include_directories(${name} PRIVATE "${ADQ_DIR}" "${ADQ_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(${name} PRIVATE -fsanitize=thread)
 endif()
endfunction()
ado_target(adt-rational_units)
foreach(case rational_fraction rational_boundary rational_base)
 add_test(NAME adt.${case} COMMAND adt-rational_units ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADQ_CASES rational_fraction rational_fraction rational_boundary rational_boundary rational_base rational_base)
 foreach(i RANGE 1 6)
  ado_target(adt-rational-units-mutant-${i})
  if(i LESS 5)
   target_compile_definitions(adt-rational-units-mutant-${i} PRIVATE ADQ_MUTATION=${i})
  else()
   math(EXPR old "${i}-2")
   target_compile_definitions(adt-rational-units-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADQ_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADQ_DIR}/expect_failure.py" $<TARGET_FILE:adt-rational-units-mutant-${i}> ${case})
 endforeach()
endif()
