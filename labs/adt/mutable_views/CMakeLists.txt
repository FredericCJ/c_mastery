cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_mutable_views C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADV_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADV_BASE "${ADV_DIR}/../src/adt.c")
set(ADV_CORE "${ADV_DIR}/mutable_views.c")
set(ADV_VECTOR "${ADV_DIR}/vector.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADV_BASE "${ADV_ASSESSOR}/adt.c")
 set(ADV_CORE "${ADV_ASSESSOR}/mutable_views/mutable_views.c")
 set(ADV_VECTOR "${ADV_ASSESSOR}/mutable_views/vector.c")
endif()
if(CMASTERY_ADV_LEARNER)
 set(ADV_BASE "${ADV_DIR}/../src/adt.c")
 set(ADV_CORE "${ADV_DIR}/mutable_views.c")
 set(ADV_VECTOR "${ADV_DIR}/vector.c")
endif()
function(adv_target name)
 add_executable(${name} "${ADV_DIR}/tests.c" ${ADV_CORE} ${ADV_VECTOR} ${ADV_BASE})
 target_include_directories(${name} PRIVATE "${ADV_DIR}" "${ADV_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -msse2 -mno-avx)
 if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_options(${name} PRIVATE -fno-tree-vectorize)
 else()
  target_compile_options(${name} PRIVATE -fno-vectorize -fno-slp-vectorize)
 endif()
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
adv_target(adt-mutable_views)
foreach(case view_geometry view_transform view_vector view_reduction view_base)
 add_test(NAME adt.${case} COMMAND adt-mutable_views ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADV_CASES view_geometry view_geometry view_transform view_transform view_vector view_vector view_reduction view_reduction view_base view_base)
 foreach(i RANGE 1 10)
  adv_target(adt-view-mutant-${i})
  if(i LESS 9)
   target_compile_definitions(adt-view-mutant-${i} PRIVATE ADV_MUTATION=${i})
  else()
   math(EXPR old "${i}+2")
   target_compile_definitions(adt-view-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADV_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADV_DIR}/expect_failure.py" $<TARGET_FILE:adt-view-mutant-${i}> ${case})
 endforeach()
endif()
