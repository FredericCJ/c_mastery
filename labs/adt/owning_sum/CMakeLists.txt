cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_owning_sum C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADO_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADO_BASE "${ADO_DIR}/../src/adt.c")
set(ADO_CORE "${ADO_DIR}/owning_sum.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADO_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADO_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADO_BASE "${ADO_ASSESSOR}/adt.c")
 set(ADO_CORE "${ADO_ASSESSOR}/owning_sum/owning_sum.c")
endif()
if(CMASTERY_ADO_LEARNER)
 set(ADO_CORE "${ADO_DIR}/owning_sum.c")
 set(ADO_BASE "${ADO_DIR}/../src/adt.c")
endif()
function(ado_target name)
 add_executable(${name} "${ADO_DIR}/tests.c" ${ADO_CORE} ${ADO_BASE})
 target_include_directories(${name} PRIVATE "${ADO_DIR}" "${ADO_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(${name} PRIVATE -fsanitize=thread)
 endif()
endfunction()
ado_target(adt-owning_sum)
foreach(case sum_value sum_owner sum_rollback sum_base)
 add_test(NAME adt.${case} COMMAND adt-owning_sum ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADO_CASES sum_owner sum_rollback sum_value sum_rollback sum_owner sum_value sum_base sum_base)
 foreach(i RANGE 1 8)
  ado_target(adt-owning-sum-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-owning-sum-mutant-${i} PRIVATE ADO_MUTATION=${i})
  else()
   math(EXPR old "${i}-6")
   target_compile_definitions(adt-owning-sum-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADO_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADO_DIR}/expect_failure.py" $<TARGET_FILE:adt-owning-sum-mutant-${i}> ${case})
 endforeach()
endif()
