cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_relocating_pool C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(ADP_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADP_BASE "${ADP_DIR}/../src/adt.c")
set(ADP_CORE "${ADP_DIR}/relocating_pool.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADP_BASE "${ADP_ASSESSOR}/adt.c")
 set(ADP_CORE "${ADP_ASSESSOR}/relocating_pool/relocating_pool.c")
endif()
if(CMASTERY_ADP_LEARNER)
 set(ADP_BASE "${ADP_DIR}/../src/adt.c")
 set(ADP_CORE "${ADP_DIR}/relocating_pool.c")
endif()
function(adp_target name)
 add_executable(${name} "${ADP_DIR}/tests.c" ${ADP_CORE} ${ADP_BASE})
 target_include_directories(${name} PRIVATE "${ADP_DIR}" "${ADP_DIR}/../include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=thread -no-pie)
 endif()
endfunction()
adp_target(adt-relocating_pool)
foreach(case pool_identity pool_retirement pool_relocation pool_native pool_base)
 add_test(NAME adt.${case} COMMAND adt-relocating_pool ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADP_CASES pool_identity pool_identity pool_retirement pool_retirement pool_relocation pool_relocation pool_native pool_native pool_base pool_base)
 foreach(i RANGE 1 10)
  adp_target(adt-pool-mutant-${i})
  if(i LESS 9)
   target_compile_definitions(adt-pool-mutant-${i} PRIVATE ADP_MUTATION=${i})
  else()
   target_compile_definitions(adt-pool-mutant-${i} PRIVATE MOD_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADP_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADP_DIR}/expect_failure.py" $<TARGET_FILE:adt-pool-mutant-${i}> ${case})
 endforeach()
endif()
