cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_partial_optics C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADL_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADL_BASE "${ADL_DIR}/../src/adt.c")
set(ADL_CORE "${ADL_DIR}/partial_optics.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADL_BASE "${ADL_ASSESSOR}/adt.c")
 set(ADL_CORE "${ADL_ASSESSOR}/partial_optics/partial_optics.c")
endif()
if(CMASTERY_ADL_LEARNER)
 set(ADL_BASE "${ADL_DIR}/../src/adt.c")
 set(ADL_CORE "${ADL_DIR}/partial_optics.c")
endif()
function(adl_target name)
 add_executable(${name} "${ADL_DIR}/tests.c" ${ADL_CORE} ${ADL_BASE})
 target_include_directories(${name} PRIVATE "${ADL_DIR}" "${ADL_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
adl_target(adt-partial_optics)
foreach(case optics_nested optics_prism optics_effects optics_base)
 add_test(NAME adt.${case} COMMAND adt-partial_optics ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADL_CASES optics_nested optics_nested optics_prism optics_prism optics_effects optics_effects optics_base optics_base)
 foreach(i RANGE 1 8)
  adl_target(adt-optics-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-optics-mutant-${i} PRIVATE ADL_MUTATION=${i})
  else()
   math(EXPR old "${i}+8")
   target_compile_definitions(adt-optics-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADL_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADL_DIR}/expect_failure.py" $<TARGET_FILE:adt-optics-mutant-${i}> ${case})
 endforeach()
endif()
