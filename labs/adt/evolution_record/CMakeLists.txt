cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_evolution_record C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_FOUNDATIONS_ROOT "${ADE_DIR}/../../foundations" CACHE PATH "Completed foundations workspace")
set(CMASTERY_API_DEPTH_ROOT "${ADE_DIR}/../../api_depth" CACHE PATH "Completed architecture workspace")
set(CMASTERY_REPLICATED_ROOT "${ADE_DIR}/../../replicated" CACHE PATH "Completed hashing workspace")
set(ADE_CURSOR "${CMASTERY_FOUNDATIONS_ROOT}/src/foundation.c")
set(ADE_CODEC "${CMASTERY_API_DEPTH_ROOT}/architecture/learner.c")
set(ADE_HASH "${CMASTERY_REPLICATED_ROOT}/hashing/hashing.c")
set(ADE_BASE "${ADE_DIR}/../src/adt.c")
set(ADE_CORE "${ADE_DIR}/evolution_record.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADE_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
  set(ADE_CURSOR "${CMASTERY_ASSESSOR_ROOT}/foundations/foundation.c")
  set(ADE_CODEC "${CMASTERY_ASSESSOR_ROOT}/api_depth/architecture/core.c")
  set(ADE_HASH "${CMASTERY_ASSESSOR_ROOT}/replicated/hashing/hashing.c")
 else()
  set(ADE_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADE_BASE "${ADE_ASSESSOR}/adt.c")
 set(ADE_CORE "${ADE_ASSESSOR}/evolution_record/evolution_record.c")
endif()
if(CMASTERY_ADE_LEARNER)
 set(ADE_BASE "${ADE_DIR}/../src/adt.c")
 set(ADE_CORE "${ADE_DIR}/evolution_record.c")
endif()
foreach(dependency FOUNDATIONS API_DEPTH REPLICATED)
 if(CMASTERY_${dependency}_ASSESSOR_ROOT)
  if(dependency STREQUAL "FOUNDATIONS")
   set(ADE_CURSOR "${CMASTERY_FOUNDATIONS_ASSESSOR_ROOT}/foundation.c")
  elseif(dependency STREQUAL "API_DEPTH")
   set(ADE_CODEC "${CMASTERY_API_DEPTH_ASSESSOR_ROOT}/architecture/core.c")
  else()
   set(ADE_HASH "${CMASTERY_REPLICATED_ASSESSOR_ROOT}/hashing/hashing.c")
  endif()
 endif()
endforeach()
if(NOT EXISTS "${ADE_CURSOR}" OR NOT EXISTS "${ADE_CODEC}" OR NOT EXISTS "${ADE_HASH}")
 message(STATUS "evolution_record requires explicit completed foundations, api_depth and replicated hashing workspaces")
 return()
endif()
function(ade_target name)
 add_executable(${name} "${ADE_DIR}/tests.c" ${ADE_CORE} ${ADE_BASE} ${ADE_CURSOR} ${ADE_CODEC} ${ADE_HASH} "${ADE_DIR}/target.c")
 target_include_directories(${name} PRIVATE "${ADE_DIR}" "${ADE_DIR}/../include" "${CMASTERY_FOUNDATIONS_ROOT}/include" "${CMASTERY_API_DEPTH_ROOT}/architecture" "${CMASTERY_REPLICATED_ROOT}/hashing")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ade_target(adt-evolution_record)
foreach(case evolution_records evolution_auth evolution_policy evolution_base)
 add_test(NAME adt.${case} COMMAND adt-evolution_record ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADE_CASES evolution_records evolution_records evolution_auth evolution_auth evolution_policy evolution_policy evolution_base evolution_base)
 foreach(i RANGE 1 8)
  ade_target(adt-evolution-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-evolution-mutant-${i} PRIVATE ADE_MUTATION=${i})
  else()
   math(EXPR old "${i}+10")
   target_compile_definitions(adt-evolution-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADE_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADE_DIR}/expect_failure.py" $<TARGET_FILE:adt-evolution-mutant-${i}> ${case})
 endforeach()
endif()
