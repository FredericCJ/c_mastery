cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_arena_flow C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADF_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(ADF_BASE "${ADF_DIR}/../src/adt.c")
set(ADF_CORE "${ADF_DIR}/arena_flow.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
 else()
  set(ADF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADF_BASE "${ADF_ASSESSOR}/adt.c")
 set(ADF_CORE "${ADF_ASSESSOR}/arena_flow/arena_flow.c")
endif()
if(CMASTERY_ADF_LEARNER)
 set(ADF_BASE "${ADF_DIR}/../src/adt.c")
 set(ADF_CORE "${ADF_DIR}/arena_flow.c")
endif()
function(adf_target name)
 add_executable(${name} "${ADF_DIR}/tests.c" ${ADF_CORE} ${ADF_BASE})
 target_include_directories(${name} PRIVATE "${ADF_DIR}" "${ADF_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
adf_target(adt-arena_flow)
foreach(case arena_certificate arena_worklist arena_outcomes arena_base)
 add_test(NAME adt.${case} COMMAND adt-arena_flow ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADF_CASES arena_certificate arena_certificate arena_worklist arena_worklist arena_outcomes arena_outcomes arena_base arena_base)
 foreach(i RANGE 1 8)
  adf_target(adt-flow-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-flow-mutant-${i} PRIVATE ADF_MUTATION=${i})
  else()
   math(EXPR old "${i}+6")
   target_compile_definitions(adt-flow-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADF_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADF_DIR}/expect_failure.py" $<TARGET_FILE:adt-flow-mutant-${i}> ${case})
 endforeach()
endif()
