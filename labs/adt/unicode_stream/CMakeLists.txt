find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(ADU_CORE "${ADU_DIR}/unicode_stream.c")
if(CMASTERY_ASSESSOR_ROOT)

function(adu_target name)
 add_executable(${name} "${ADU_DIR}/tests.c" ${ADU_CORE} ${ADU_OWNER} ${ADU_BASE})
 target_include_directories(${name} PRIVATE "${ADU_DIR}" "${ADU_DIR}/../include" "${ADU_DIR}/../owning_sum")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
adu_target(adt-unicode_stream)
foreach(case utf8_chunks utf8_finalization utf8_owner utf8_base)
 add_test(NAME adt.${case} COMMAND adt-unicode_stream ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADU_CASES utf8_chunks utf8_finalization utf8_finalization utf8_chunks utf8_owner utf8_owner utf8_base utf8_base)
 foreach(i RANGE 1 8)
  adu_target(adt-utf8-stream-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-utf8-stream-mutant-${i} PRIVATE ADU_MUTATION=${i})
  else()
   target_compile_definitions(adt-utf8-stream-mutant-${i} PRIVATE MOD_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADU_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADU_DIR}/expect_failure.py" $<TARGET_FILE:adt-utf8-stream-mutant-${i}> ${case})
 endforeach()
endif()
