cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(adt_packet_ring C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADP_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_FOUNDATIONS_ROOT "${ADP_DIR}/../../foundations" CACHE PATH "Completed foundations workspace")
set(CMASTERY_API_DEPTH_ROOT "${ADP_DIR}/../../api_depth" CACHE PATH "Completed architecture workspace")
set(CMASTERY_HOSTED_SYSTEMS_ROOT "${ADP_DIR}/../../hosted_systems" CACHE PATH "Completed hosted frame workspace")
set(ADP_CURSOR "${CMASTERY_FOUNDATIONS_ROOT}/src/foundation.c")
set(ADP_CODEC "${CMASTERY_API_DEPTH_ROOT}/architecture/learner.c")
set(ADP_FRAME "${CMASTERY_HOSTED_SYSTEMS_ROOT}/src/frame.c")
set(ADP_BASE "${ADP_DIR}/../src/adt.c")
set(ADP_CORE "${ADP_DIR}/packet_ring.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/adt/adt.c")
  set(ADP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/adt")
  set(ADP_CURSOR "${CMASTERY_ASSESSOR_ROOT}/foundations/foundation.c")
  set(ADP_CODEC "${CMASTERY_ASSESSOR_ROOT}/api_depth/architecture/core.c")
  set(ADP_FRAME "${CMASTERY_ASSESSOR_ROOT}/hosted_systems/frame.c")
 else()
  set(ADP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(ADP_BASE "${ADP_ASSESSOR}/adt.c")
 set(ADP_CORE "${ADP_ASSESSOR}/packet_ring/packet_ring.c")
endif()
if(CMASTERY_ADP_LEARNER)
 set(ADP_BASE "${ADP_DIR}/../src/adt.c")
 set(ADP_CORE "${ADP_DIR}/packet_ring.c")
endif()
foreach(dependency FOUNDATIONS API_DEPTH HOSTED_SYSTEMS)
 if(CMASTERY_${dependency}_ASSESSOR_ROOT)
  if(dependency STREQUAL "FOUNDATIONS")
   set(ADP_CURSOR "${CMASTERY_FOUNDATIONS_ASSESSOR_ROOT}/foundation.c")
  elseif(dependency STREQUAL "API_DEPTH")
   set(ADP_CODEC "${CMASTERY_API_DEPTH_ASSESSOR_ROOT}/architecture/core.c")
  else()
   set(ADP_FRAME "${CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT}/frame.c")
  endif()
 endif()
endforeach()
if(NOT EXISTS "${ADP_CURSOR}" OR NOT EXISTS "${ADP_CODEC}" OR NOT EXISTS "${ADP_FRAME}")
 message(STATUS "packet_ring requires explicit completed foundations, api_depth and hosted_systems frame workspaces")
 return()
endif()
function(adp_target name)
 add_executable(${name} "${ADP_DIR}/tests.c" ${ADP_CORE} ${ADP_BASE} ${ADP_CURSOR} ${ADP_CODEC} ${ADP_FRAME} "${ADP_DIR}/packet_ring.h")
 target_include_directories(${name} PRIVATE "${ADP_DIR}" "${ADP_DIR}/../include" "${CMASTERY_FOUNDATIONS_ROOT}/include" "${CMASTERY_API_DEPTH_ROOT}/architecture" "${CMASTERY_HOSTED_SYSTEMS_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
adp_target(adt-packet_ring)
foreach(case packet_ring packet_lifetime packet_pipe packet_base)
 add_test(NAME adt.${case} COMMAND adt-packet_ring ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(ADP_CASES packet_ring packet_ring packet_lifetime packet_lifetime packet_pipe packet_pipe packet_base packet_base)
 foreach(i RANGE 1 8)
  adp_target(adt-packet-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(adt-packet-mutant-${i} PRIVATE ADP_MUTATION=${i})
  else()
   math(EXPR old "${i}+12")
   target_compile_definitions(adt-packet-mutant-${i} PRIVATE MOD_MUTATION=${old})
  endif()
  math(EXPR index "${i}-1")
  list(GET ADP_CASES ${index} case)
  add_test(NAME adt.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${ADP_DIR}/expect_failure.py" $<TARGET_FILE:adt-packet-mutant-${i}> ${case})
 endforeach()
endif()
