cmake_minimum_required(VERSION 3.20)
project(c_mastery_runtime C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private assessor/runtime directory")
option(CMASTERY_SANITIZERS "Enable hosted address and undefined behavior diagnostics" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
    message(FATAL_ERROR "CMASTERY_ASSESSOR_ROOT must be absolute")
  endif()
  set(CVM_IMPL "${CMASTERY_ASSESSOR_ROOT}")
else()
  set(CVM_IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
endif()
add_library(cvm_core STATIC "${CVM_IMPL}/cvm.c")
target_include_directories(cvm_core PUBLIC include)
add_library(cvm_compiler STATIC "${CVM_IMPL}/compiler.c")
target_link_libraries(cvm_compiler PUBLIC cvm_core)
foreach(t cvm_core cvm_compiler)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(CMASTERY_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(cvm_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(cvm_compiler PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(cvm-run src/main.c)
target_link_libraries(cvm-run PRIVATE cvm_compiler)
add_executable(cvm-compile src/compile_main.c)
target_link_libraries(cvm-compile PRIVATE cvm_compiler)
add_executable(runtime-tests tests/runtime_tests.c)
target_link_libraries(runtime-tests PRIVATE cvm_compiler)
enable_testing()
foreach(case wire image_verifier verifier arithmetic calls compiler control arena gc fuel host portability integration)
  add_test(NAME runtime.${case} COMMAND runtime-tests ${case})
  set_tests_properties(runtime.${case} PROPERTIES TIMEOUT 5)
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  add_executable(runtime-roots-mutant tests/runtime_tests.c "${CVM_IMPL}/cvm.c" "${CVM_IMPL}/compiler.c")
  target_include_directories(runtime-roots-mutant PRIVATE include)
  target_compile_definitions(runtime-roots-mutant PRIVATE CVM_MUTATE_SKIP_ROOTS)
  add_test(NAME runtime.mutation.roots COMMAND runtime-roots-mutant gc)
  set_tests_properties(runtime.mutation.roots PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
endif()
