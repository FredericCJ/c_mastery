cmake_minimum_required(VERSION 3.16)
project(verification_fuzzing C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
set(CORE src/core.c)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/verification_fuzzing/core.c")
    set(CORE "${CMASTERY_ASSESSOR_ROOT}/verification_fuzzing/core.c")
  else()
    set(CORE "${CMASTERY_ASSESSOR_ROOT}/core.c")
  endif()
endif()
add_executable(b06-tests ${CORE} tests/tests.c)
target_include_directories(b06-tests PRIVATE include)
if(NOT MSVC)
  target_compile_options(b06-tests PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(b06-tests PRIVATE m)
endif()
add_test(NAME verification_fuzzing.add COMMAND b06-tests add)
if(CMASTERY_ASSESSOR_ROOT)
  add_executable(b06-bad-1 ${CORE} tests/tests.c)
  target_include_directories(b06-bad-1 PRIVATE include)
  target_compile_definitions(b06-bad-1 PRIVATE B06_MUTATION=1)
  if(NOT MSVC)
    target_link_libraries(b06-bad-1 PRIVATE m)
  endif()
  add_test(NAME verification_fuzzing.mutation.add COMMAND b06-bad-1 add)
  set_tests_properties(verification_fuzzing.mutation.add PROPERTIES WILL_FAIL TRUE)
endif()
add_test(NAME verification_fuzzing.search COMMAND b06-tests search)
if(CMASTERY_ASSESSOR_ROOT)
