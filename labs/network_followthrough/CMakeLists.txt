cmake_minimum_required(VERSION 3.20)
project(c_mastery_network_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit repository assessor root or network_followthrough directory")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan verification only" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/ip.c")
set(SYSTEMS "${CMAKE_CURRENT_SOURCE_DIR}/src/checksum.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor root must be explicit and absolute")
 endif()
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/network_followthrough/ip/ip.c")
  set(PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(PRIVATE "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(IMPL "${PRIVATE}/network_followthrough/ip/ip.c")
 set(SYSTEMS "${PRIVATE}/b045_systems/systems.c")
 if(NOT EXISTS "${IMPL}" OR NOT EXISTS "${SYSTEMS}")
  message(FATAL_ERROR "Explicit original IP and checksum providers required")
 endif()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ip/checksum_slice.py"
   "${SYSTEMS}" "${CMAKE_CURRENT_BINARY_DIR}/checksum.c" COMMAND_ERROR_IS_FATAL ANY)
 set(SYSTEMS "${CMAKE_CURRENT_BINARY_DIR}/checksum.c")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-pie -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_executable(network-ip ip/tests.c "${IMPL}" "${SYSTEMS}")
target_include_directories(network-ip PRIVATE ip include)
enable_testing()
add_test(NAME network_followthrough.ip COMMAND network-ip)

set(CONGESTION "${CMAKE_CURRENT_SOURCE_DIR}/src/congestion.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(CONGESTION "${PRIVATE}/network_followthrough/congestion/congestion.c")
 if(NOT EXISTS "${CONGESTION}")
  message(FATAL_ERROR "Explicit original congestion provider required")
 endif()
endif()
add_executable(network-congestion congestion/tests.c "${CONGESTION}")
target_include_directories(network-congestion PRIVATE congestion)
add_test(NAME network_followthrough.congestion COMMAND network-congestion)
