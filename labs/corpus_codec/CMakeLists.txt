cmake_minimum_required(VERSION 3.16)
project(corpus_codec C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CC_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/codec.c" CACHE FILEPATH "Explicit completed component")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private provider root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_codec/codec.c")
    set(CC_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/corpus_codec")
  else()
    set(CC_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
  endif()
  set(CC_PROVIDER "${CC_PRIVATE}/codec.c")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(NOT EXISTS "${CC_PROVIDER}")
  message(FATAL_ERROR "Supply explicit completed codec provider")
endif()
add_library(codec STATIC "${CC_PROVIDER}")
target_include_directories(codec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(codec PRIVATE -Wall -Wextra -Werror -pedantic-errors)
add_executable(codec_tests tests.c)
target_link_libraries(codec_tests PRIVATE codec)
add_executable(codec_driver driver.c)
target_link_libraries(codec_driver PRIVATE codec)
add_executable(codec_measure measure.c)
target_link_libraries(codec_measure PRIVATE codec)
enable_testing()
add_test(NAME corpus_codec.contract COMMAND codec_tests)
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME corpus_codec.oracle COMMAND "${Python3_EXECUTABLE}" "${CC_PRIVATE}/oracle.py" $<TARGET_FILE:codec_driver>)
  set_tests_properties(corpus_codec.oracle PROPERTIES TIMEOUT 180)
  foreach(m RANGE 1 4)
    add_executable(codec_mutant_${m} tests.c "${CC_PROVIDER}")
    target_include_directories(codec_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(codec_mutant_${m} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    target_compile_definitions(codec_mutant_${m} PRIVATE CC_MUTANT=${m})
    add_test(NAME corpus_codec.mutation.contract.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:codec_mutant_${m}>)
  endforeach()
endif()
