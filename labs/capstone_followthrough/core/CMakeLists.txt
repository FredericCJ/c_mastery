cmake_minimum_required(VERSION 3.20)
project(capcore VERSION 1.0 LANGUAGES C)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CAP_BUILD_ADAPTER "Build the explicitly supplied Linux adapter" OFF)
option(CAP_BUILD_EXAMPLES "Build the installed-API consumer" OFF)
foreach(variable CAP_CORE_SOURCE CAP_PROTOCOL_SOURCE CAP_PROTOCOL_INCLUDE)
  if(NOT DEFINED ${variable})
    message(FATAL_ERROR "Supply ${variable}; no assessor discovery is performed")
  endif()
endforeach()
add_library(capcore ${CAP_CORE_SOURCE} ${CAP_PROTOCOL_SOURCE})
target_include_directories(capcore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<BUILD_INTERFACE:${CAP_PROTOCOL_INCLUDE}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(capcore PROPERTIES VERSION 1.0 SOVERSION 1)
if(CAP_BUILD_ADAPTER)
  if(NOT DEFINED CAP_LINUX_SOURCE OR NOT DEFINED CAP_LINUX_INCLUDE)
    message(FATAL_ERROR "Linux adapter requires explicit source and include paths")
  endif()
  add_library(caplinux ${CAP_LINUX_SOURCE})
  target_include_directories(caplinux PRIVATE ${CAP_LINUX_INCLUDE})
  target_link_libraries(caplinux PUBLIC capcore)
endif()
if(CAP_BUILD_EXAMPLES)
  add_executable(capconsumer consumer.c)
  target_link_libraries(capconsumer PRIVATE capcore)
endif()
install(TARGETS capcore LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES service.h ${CAP_PROTOCOL_INCLUDE}/checkpoint_depth.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
