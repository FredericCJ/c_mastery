cmake_minimum_required(VERSION 3.20)
project(portability_facade C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(BF_CORE "${CMAKE_CURRENT_LIST_DIR}/facade.c")
set(BF_HEADER "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth/include/facade.h")
set(BF_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth/src/depth.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(BF_CORE "${CMASTERY_ASSESSOR_ROOT}/language_portability/facade/facade.c")
 set(BF_HEADER "${CMASTERY_ASSESSOR_ROOT}/b02_depth/facade.h")
 set(BF_SOURCE "${CMASTERY_ASSESSOR_ROOT}/b02_depth/depth.c")
endif()
if(NOT EXISTS "${BF_CORE}" OR NOT EXISTS "${BF_HEADER}" OR NOT EXISTS "${BF_SOURCE}")
 message(FATAL_ERROR "Supply the explicit completed facade and B02 implementations")
endif()
function(bf_target name)
 add_executable(${name} tests.c "${BF_CORE}" "${BF_SOURCE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/../../b02_depth/include")
 target_compile_definitions(${name} PRIVATE BF_B02_FACADE="${BF_HEADER}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
bf_target(portability-facade)
foreach(case borrow binary sequence)
 add_test(NAME portability.${case} COMMAND portability-facade ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(cases borrow binary sequence)
 foreach(i RANGE 1 3)
  bf_target(portability-facade-mutant-${i})
  target_compile_definitions(portability-facade-mutant-${i} PRIVATE BF_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME portability.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:portability-facade-mutant-${i}> ${case})
 endforeach()
endif()
