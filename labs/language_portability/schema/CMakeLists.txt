cmake_minimum_required(VERSION 3.20)
project(portability_schema C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LS_PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(LS_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/language_portability/schema")
endif()
if(NOT EXISTS "${LS_PRIVATE}/schema.c" OR NOT EXISTS "${LS_PRIVATE}/generate.py")
 message(FATAL_ERROR "Supply the explicit completed schema implementation")
endif()
set(LS_GENERATED)
foreach(file config.h rules.h fields.h codec.h manifest.json)
 list(APPEND LS_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/${file}")
endforeach()
add_custom_command(OUTPUT ${LS_GENERATED}
 COMMAND "${Python3_EXECUTABLE}" "${LS_PRIVATE}/generate.py" "${CMAKE_CURRENT_LIST_DIR}/schema.json" "${CMAKE_CURRENT_BINARY_DIR}/generated"
 DEPENDS "${LS_PRIVATE}/generate.py" "${CMAKE_CURRENT_LIST_DIR}/schema.json" VERBATIM)
add_custom_target(schema-generated DEPENDS ${LS_GENERATED})
function(ls_target name)
 add_executable(${name} tests.c "${LS_PRIVATE}/schema.c")
 add_dependencies(${name} schema-generated)
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ls_target(portability-schema)
foreach(case wire identity boundary)
 add_test(NAME portability.${case} COMMAND portability-schema ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(cases wire identity boundary)
 foreach(i RANGE 1 3)
  ls_target(portability-schema-mutant-${i})
  target_compile_definitions(portability-schema-mutant-${i} PRIVATE LS_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME portability.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:portability-schema-mutant-${i}> ${case})
 endforeach()
endif()
