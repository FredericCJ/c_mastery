cmake_minimum_required(VERSION 3.20)
project(portability_capability C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LP_CORE "${CMAKE_CURRENT_LIST_DIR}/capability.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(LP_CORE "${CMASTERY_ASSESSOR_ROOT}/language_portability/capability/capability.c")
endif()
if(NOT EXISTS "${LP_CORE}")
 message(FATAL_ERROR "Supply the explicit completed capability implementation")
endif()
function(lp_target name)
 add_executable(${name} tests.c "${LP_CORE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
lp_target(portability-capability)
foreach(case record transform matrix trace)
 add_test(NAME portability.${case} COMMAND portability-capability ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(cases transform matrix trace record)
 foreach(i RANGE 1 4)
  lp_target(portability-capability-mutant-${i})
  target_compile_definitions(portability-capability-mutant-${i} PRIVATE LP_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME portability.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:portability-capability-mutant-${i}> ${case})
 endforeach()
endif()
