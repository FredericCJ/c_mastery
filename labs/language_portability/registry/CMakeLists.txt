cmake_minimum_required(VERSION 3.20)
project(portability_registry C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LR_GENERATED)
foreach(file ids.h names.inc dispatch.inc aliases.inc validation.json manifest.json)
 list(APPEND LR_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/${file}")
endforeach()
add_custom_command(OUTPUT ${LR_GENERATED}
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/generate.py" "${CMAKE_CURRENT_LIST_DIR}/registry.json" "${CMAKE_CURRENT_BINARY_DIR}/generated"
 DEPENDS "${CMAKE_CURRENT_LIST_DIR}/generate.py" "${CMAKE_CURRENT_LIST_DIR}/registry.json"
 VERBATIM)
add_custom_target(registry-generated DEPENDS ${LR_GENERATED})
set(LP_CORE "${CMAKE_CURRENT_LIST_DIR}/registry.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(LP_CORE "${CMASTERY_ASSESSOR_ROOT}/language_portability/registry/registry.c")
endif()
if(NOT EXISTS "${LP_CORE}")
 message(FATAL_ERROR "Supply the explicit completed registry implementation")
endif()
function(lp_target name)
 get_filename_component(LP_DIRECTORY "${LP_CORE}" DIRECTORY)
 add_executable(${name} tests.c "${LP_CORE}" "${LP_DIRECTORY}/generic.c")
 add_dependencies(${name} registry-generated)
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
lp_target(portability-registry)
foreach(case unknown association generic)
 add_test(NAME portability.${case} COMMAND portability-registry ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_PORTABILITY_LEARNER)
 set(cases unknown association generic)
 foreach(i RANGE 1 3)
  lp_target(portability-registry-mutant-${i})
  target_compile_definitions(portability-registry-mutant-${i} PRIVATE LR_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME portability.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:portability-registry-mutant-${i}> ${case})
 endforeach()
endif()
