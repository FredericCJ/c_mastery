cmake_minimum_required(VERSION 3.20)
project(storage_transaction_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit learner storage provider directory")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h")
 message(FATAL_ERROR "Supply the completed storage provider using CMASTERY_STORAGE_ROOT")
endif()
set(ST_SOURCE "${CMAKE_CURRENT_LIST_DIR}/transactions.c")
set(ST_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(ST_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(ST_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(ST_ORIGINAL "${ST_ASSESSOR}/storage/store.c")
 file(SHA256 "${ST_ORIGINAL}" ST_ORIGINAL_HASH)
 set(ST_STORE "${CMAKE_CURRENT_BINARY_DIR}/store-instrumented.c")
 add_custom_command(OUTPUT "${ST_STORE}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/instrument_store.py" --input "${ST_ORIGINAL}" --output "${ST_STORE}" --sha256 "${ST_ORIGINAL_HASH}" DEPENDS "${ST_ORIGINAL}" "${CMAKE_CURRENT_LIST_DIR}/instrument_store.py" VERBATIM)
 if(NOT CMASTERY_ST_LEARNER)
  set(ST_SOURCE "${ST_ASSESSOR}/storage_followthrough/transactions/transactions.c")
 endif()
endif()
function(st_target name)
 add_executable(${name} tests.c "${ST_SOURCE}" "${ST_STORE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
st_target(storage-transactions)
foreach(case original_abort operation savepoint lifecycle nosteal capacity cost uncertain)
 add_test(NAME storage_followthrough.${case} COMMAND storage-transactions ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_ST_LEARNER)
 set(ST_CONTROLS operation savepoint lifecycle nosteal)
 foreach(i RANGE 1 4)
  st_target(storage-transactions-mutant-${i})
  target_compile_definitions(storage-transactions-mutant-${i} PRIVATE ST_MUTANT=${i})
  math(EXPR index "${i}-1")
  list(GET ST_CONTROLS ${index} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-transactions-mutant-${i}> ${case})
 endforeach()
endif()
