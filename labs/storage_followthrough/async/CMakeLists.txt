cmake_minimum_required(VERSION 3.20)
project(storage_async_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/async.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SA_LEARNER)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/async/async.c")
  set(SA_SOURCE "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/async/async.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/async.c")
  set(SA_SOURCE "${CMASTERY_ASSESSOR_ROOT}/async.c")
 else()
  message(FATAL_ERROR "Supply the explicit completed async implementation")
 endif()
endif()
function(sa_target name)
 add_executable(${name} tests.c "${SA_SOURCE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sa_target(storage-async)
foreach(case retained unknown admission)
 add_test(NAME storage_followthrough.${case} COMMAND storage-async ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SA_LEARNER)
 set(cases retained unknown retained)
 foreach(i RANGE 1 3)
  sa_target(storage-async-mutant-${i})
  target_compile_definitions(storage-async-mutant-${i} PRIVATE SA_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-async-mutant-${i}> ${case})
 endforeach()
endif()
