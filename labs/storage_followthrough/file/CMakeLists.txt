cmake_minimum_required(VERSION 3.20)
project(storage_file_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit completed storage provider")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h" OR NOT EXISTS "${CMASTERY_STORAGE_ROOT}/src/store_file.c")
 message(FATAL_ERROR "Supply completed storage and original file provider using CMASTERY_STORAGE_ROOT")
endif()
set(SFI_SOURCE "${CMAKE_CURRENT_LIST_DIR}/file.c")
set(SFI_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(SFI_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SFI_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SFI_STORE "${SFI_ASSESSOR}/storage/store.c")
 if(NOT CMASTERY_SFI_LEARNER)
  set(SFI_SOURCE "${SFI_ASSESSOR}/storage_followthrough/file/file.c")
 endif()
endif()
function(sfi_target name)
 add_executable(${name} tests.c "${SFI_SOURCE}" "${SFI_STORE}" "${CMASTERY_STORAGE_ROOT}/src/store_file.c")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_options(${name} PRIVATE -Wl,--wrap=pread,--wrap=pwrite,--wrap=fsync,--wrap=close)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sfi_target(storage-file)
foreach(case original_transfer nonblocking readiness native fresh amplification)
 add_test(NAME storage_followthrough.${case} COMMAND storage-file ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SFI_LEARNER)
 set(cases nonblocking nonblocking fresh fresh)
 foreach(i RANGE 1 4)
  sfi_target(storage-file-mutant-${i})
  target_compile_definitions(storage-file-mutant-${i} PRIVATE SFI_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-file-mutant-${i}> ${case})
 endforeach()
endif()
