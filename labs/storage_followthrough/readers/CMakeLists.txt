cmake_minimum_required(VERSION 3.20)
project(storage_readers_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit original storage provider")
set(CMASTERY_CACHE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../cache" CACHE PATH "Explicit cache provider")
set(CMASTERY_LOOKUP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../lookup" CACHE PATH "Explicit copied snapshot provider")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h" OR NOT EXISTS "${CMASTERY_CACHE_ROOT}/cache.h" OR NOT EXISTS "${CMASTERY_LOOKUP_ROOT}/lookup.h")
 message(FATAL_ERROR "Supply completed storage, cache and lookup providers")
endif()
set(SR_SOURCE "${CMAKE_CURRENT_LIST_DIR}/readers.c")
set(SR_IMPORTS "${CMASTERY_STORAGE_ROOT}/src/store.c" "${CMASTERY_CACHE_ROOT}/cache.c" "${CMASTERY_LOOKUP_ROOT}/lookup.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(SR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SR_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SR_IMPORTS "${SR_ASSESSOR}/storage/store.c" "${SR_ASSESSOR}/storage_followthrough/cache/cache.c" "${SR_ASSESSOR}/storage_followthrough/lookup/lookup.c")
 if(NOT CMASTERY_SR_LEARNER)
  set(SR_SOURCE "${SR_ASSESSOR}/storage_followthrough/readers/readers.c")
 endif()
endif()
function(sr_target name)
 add_executable(${name} tests.c "${SR_SOURCE}" ${SR_IMPORTS})
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include" "${CMASTERY_CACHE_ROOT}" "${CMASTERY_LOOKUP_ROOT}")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=thread -no-pie)
 endif()
endfunction()
sr_target(storage-readers)
foreach(case snapshots shared limits failure threaded)
 add_test(NAME storage_followthrough.${case} COMMAND storage-readers ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SR_LEARNER)
 set(SR_CONTROLS shared shared snapshots snapshots)
 foreach(i RANGE 1 4)
  sr_target(storage-readers-mutant-${i})
  target_compile_definitions(storage-readers-mutant-${i} PRIVATE SR_MUTANT=${i})
  math(EXPR index "${i}-1")
  list(GET SR_CONTROLS ${index} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-readers-mutant-${i}> ${case})
 endforeach()
endif()
