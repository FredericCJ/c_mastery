cmake_minimum_required(VERSION 3.20)
project(storage_lookup_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit learner storage provider directory")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h")
 message(FATAL_ERROR "Supply the completed storage provider using CMASTERY_STORAGE_ROOT")
endif()
set(SL_SOURCE "${CMAKE_CURRENT_LIST_DIR}/lookup.c")
set(SL_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(SL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SL_STORE "${SL_ASSESSOR}/storage/store.c")
 if(NOT CMASTERY_SL_LEARNER)
  set(SL_SOURCE "${SL_ASSESSOR}/storage_followthrough/lookup/lookup.c")
 endif()
endif()
function(sl_target name)
 add_executable(${name} tests.c "${SL_SOURCE}" "${SL_STORE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sl_target(storage-lookup)
foreach(case binary normalize scans snapshot lower structural)
 add_test(NAME storage_followthrough.${case} COMMAND storage-lookup ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SL_LEARNER)
 set(SL_CONTROLS normalize normalize structural lower binary)
 foreach(i RANGE 1 5)
  sl_target(storage-lookup-mutant-${i})
  target_compile_definitions(storage-lookup-mutant-${i} PRIVATE SL_MUTANT=${i})
  if(i EQUAL 5)
   target_compile_definitions(storage-lookup-mutant-${i} PRIVATE STORAGE_MUTANT_BAD_SEPARATOR=1)
  endif()
  math(EXPR index "${i}-1")
  list(GET SL_CONTROLS ${index} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-lookup-mutant-${i}> ${case})
 endforeach()
endif()
