cmake_minimum_required(VERSION 3.20)
project(storage_codec_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit learner storage provider directory")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h")
 message(FATAL_ERROR "Supply the completed storage provider using CMASTERY_STORAGE_ROOT")
endif()
set(SFC_SOURCE "${CMAKE_CURRENT_LIST_DIR}/codec.c")
set(SFC_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(SFC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SFC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SFC_STORE "${SFC_ASSESSOR}/storage/store.c")
 if(NOT CMASTERY_SFC_LEARNER)
  set(SFC_SOURCE "${SFC_ASSESSOR}/storage_followthrough/codec/codec.c")
 endif()
endif()
function(sfc_target name)
 add_executable(${name} tests.c "${SFC_SOURCE}" "${SFC_STORE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sfc_target(storage-codec)
foreach(case original migration variable atomic fresh reopen)
 add_test(NAME storage_followthrough.${case} COMMAND storage-codec ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SFC_LEARNER)
 set(SFC_CONTROLS variable migration atomic migration fresh)
 foreach(i RANGE 1 5)
  sfc_target(storage-codec-mutant-${i})
  target_compile_definitions(storage-codec-mutant-${i} PRIVATE SFC_MUTANT=${i})
  math(EXPR index "${i}-1")
  list(GET SFC_CONTROLS ${index} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-codec-mutant-${i}> ${case})
 endforeach()
endif()
