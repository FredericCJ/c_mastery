cmake_minimum_required(VERSION 3.20)
project(storage_delta_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit storage public header")
set(CMASTERY_MEDIUM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../durability" CACHE PATH "Explicit completed failure medium")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h" OR NOT EXISTS "${CMASTERY_MEDIUM_ROOT}/medium.h" OR NOT EXISTS "${CMASTERY_MEDIUM_ROOT}/medium.c")
 message(FATAL_ERROR "Supply storage header and completed medium using explicit provider paths")
endif()
set(SD_SOURCE "${CMAKE_CURRENT_LIST_DIR}/delta.c")
set(SD_MEDIUM "${CMASTERY_MEDIUM_ROOT}/medium.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/delta/delta.c")
  set(SD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SD_MEDIUM "${SD_ASSESSOR}/storage_followthrough/durability/medium.c")
 if(NOT CMASTERY_SD_LEARNER)
  set(SD_SOURCE "${SD_ASSESSOR}/storage_followthrough/delta/delta.c")
 endif()
endif()
function(sd_target name)
 add_executable(${name} tests.c "${SD_SOURCE}" "${SD_MEDIUM}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include" "${CMASTERY_MEDIUM_ROOT}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sd_target(storage-delta)
foreach(case boundary commit checkpoint repeated corrupt cost)
 add_test(NAME storage_followthrough.${case} COMMAND storage-delta ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SD_LEARNER)
 set(cases commit checkpoint corrupt boundary)
 foreach(i RANGE 1 4)
  sd_target(storage-delta-mutant-${i})
  target_compile_definitions(storage-delta-mutant-${i} PRIVATE SD_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-delta-mutant-${i}> ${case})
 endforeach()
endif()
