cmake_minimum_required(VERSION 3.20)
project(storage_service_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_LABS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Explicit completed public components")
foreach(path storage/include/store.h runtime/include/cvm.h runtime/adapters/store_host.c runtime_followthrough/services/services.h storage_followthrough/durability/medium.h)
 if(NOT EXISTS "${CMASTERY_LABS_ROOT}/${path}")
  message(FATAL_ERROR "Supply explicit storage runtime services and medium public components")
 endif()
endforeach()
set(SW_SOURCE "${CMAKE_CURRENT_LIST_DIR}/workflow.c")
set(SW_STORE "${CMASTERY_LABS_ROOT}/storage/src/store.c")
set(SW_CVM "${CMASTERY_LABS_ROOT}/runtime/src/cvm.c")
set(SW_VH "${CMASTERY_LABS_ROOT}/runtime_followthrough/services/learner.c")
set(SW_MEDIUM "${CMASTERY_LABS_ROOT}/storage_followthrough/durability/medium.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/service/workflow.c")
  set(SW_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SW_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SW_STORE "${SW_ASSESSOR}/storage/store.c")
 set(SW_CVM "${SW_ASSESSOR}/runtime/cvm.c")
 set(SW_VH "${SW_ASSESSOR}/runtime_followthrough/services/services.c")
 set(SW_MEDIUM "${SW_ASSESSOR}/storage_followthrough/durability/medium.c")
 if(NOT CMASTERY_SW_LEARNER)
  set(SW_SOURCE "${SW_ASSESSOR}/storage_followthrough/service/workflow.c")
 endif()
endif()
function(sw_target name)
 add_executable(${name} tests.c "${SW_SOURCE}" "${SW_STORE}" "${SW_CVM}" "${SW_VH}" "${SW_MEDIUM}" "${CMASTERY_LABS_ROOT}/runtime/adapters/store_host.c")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_LABS_ROOT}/storage/include" "${CMASTERY_LABS_ROOT}/runtime/include" "${CMASTERY_LABS_ROOT}/runtime/adapters" "${CMASTERY_LABS_ROOT}/storage_followthrough/durability" "${CMASTERY_LABS_ROOT}/runtime_followthrough/services")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sw_target(storage-service)
foreach(case ordinary cuts create uncertain capacity)
 add_test(NAME storage_followthrough.${case} COMMAND storage-service ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SW_LEARNER)
 set(cases cuts ordinary uncertain cuts)
 foreach(i RANGE 1 4)
  sw_target(storage-service-mutant-${i})
  target_compile_definitions(storage-service-mutant-${i} PRIVATE SW_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-service-mutant-${i}> ${case})
 endforeach()
endif()
