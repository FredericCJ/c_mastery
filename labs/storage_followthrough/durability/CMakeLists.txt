cmake_minimum_required(VERSION 3.20)
project(storage_durability_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit completed storage provider")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h")
 message(FATAL_ERROR "Supply completed storage provider using CMASTERY_STORAGE_ROOT")
endif()
set(DM_SOURCE "${CMAKE_CURRENT_LIST_DIR}/medium.c")
set(DG_SOURCE "${CMAKE_CURRENT_LIST_DIR}/guard.c")
set(DU_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
set(DU_THIRD "${CMAKE_CURRENT_LIST_DIR}/third/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage/store.c")
  set(DU_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(DU_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(DU_ORIGINAL "${DU_ASSESSOR}/storage/store.c")
 set(DU_STORE "${CMAKE_CURRENT_BINARY_DIR}/store-guarded.c")
 set(DU_THIRD "${CMAKE_CURRENT_BINARY_DIR}/store-third.c")
 foreach(slots RANGE 2 3)
  if(slots EQUAL 2)
   set(output "${DU_STORE}")
  else()
   set(output "${DU_THIRD}")
  endif()
  add_custom_command(OUTPUT "${output}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/instrument_store.py" "${DU_ORIGINAL}" "${output}" --slots ${slots} --sha256 DEPENDS "${DU_ORIGINAL}" "${CMAKE_CURRENT_LIST_DIR}/instrument_store.py" VERBATIM)
 endforeach()
 if(NOT CMASTERY_DUR_LEARNER)
  set(DM_SOURCE "${DU_ASSESSOR}/storage_followthrough/durability/medium.c")
  set(DG_SOURCE "${DU_ASSESSOR}/storage_followthrough/durability/guard.c")
 endif()
endif()
function(du_target name variant)
 if(variant STREQUAL "third")
  add_executable(${name} third/tests.c "${DM_SOURCE}" "${DG_SOURCE}" "${DU_THIRD}")
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/third/include")
  target_compile_definitions(${name} PRIVATE DM_PAGES=68 DG_MAGIC=0x43535436u)
 elseif(variant STREQUAL "original")
  add_executable(${name} tests.c "${DM_SOURCE}" "${DU_ORIGINAL}")
  target_compile_definitions(${name} PRIVATE DUR_ORIGINAL=1)
 else()
  add_executable(${name} tests.c "${DM_SOURCE}" "${DG_SOURCE}" "${DU_STORE}")
 endif()
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
 if(CMASTERY_TSAN)
  target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=thread -no-pie)
 endif()
endfunction()
du_target(storage-durability main)
foreach(case commit checkpoint repeated lies invalid successive metadata late_ack)
 add_test(NAME storage_followthrough.${case} COMMAND storage-durability ${case})
endforeach()
du_target(storage-durability-third third)
foreach(case selection cuts maximum)
 add_test(NAME storage_followthrough.third_${case} COMMAND storage-durability-third ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_DUR_LEARNER)
 du_target(storage-durability-original original)
 add_test(NAME storage_followthrough.original_structural COMMAND storage-durability-original invalid)
 set(definitions STORAGE_MUTANT_NO_COMMIT_FLUSH STORAGE_MUTANT_EARLY_RECLAIM DG_MUTANT DM_MUTANT STORAGE_MUTANT_THIRD_LOWEST)
 set(cases commit checkpoint invalid commit selection)
 foreach(i RANGE 1 5)
  math(EXPR ix "${i}-1")
  list(GET definitions ${ix} definition)
  list(GET cases ${ix} case)
  if(i EQUAL 5)
   du_target(storage-durability-mutant-${i} third)
  else()
   du_target(storage-durability-mutant-${i} main)
  endif()
  target_compile_definitions(storage-durability-mutant-${i} PRIVATE ${definition}=1)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-durability-mutant-${i}> ${case})
 endforeach()
endif()
