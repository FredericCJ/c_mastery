cmake_minimum_required(VERSION 3.20)
project(storage_tree_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(TREE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/tree.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_TREE_LEARNER)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/tree/tree.c")
  set(TREE_SOURCE "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/tree/tree.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/tree.c")
  set(TREE_SOURCE "${CMASTERY_ASSESSOR_ROOT}/tree.c")
 else()
  message(FATAL_ERROR "Supply the explicit tree fork assessor source")
 endif()
endif()
function(tree_target name capacity full)
 add_executable(${name} tests.c "${TREE_SOURCE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
 target_compile_definitions(${name} PRIVATE TREE_CAPACITY=${capacity} TREE_FULL_REFRESH=${full})
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
foreach(capacity 7 3)
 foreach(policy path full)
  if(policy STREQUAL "full")
   set(full 1)
  else()
   set(full 0)
  endif()
  tree_target(storage-tree-${capacity}-${policy} ${capacity} ${full})
  foreach(case exercise minimum separator shape mixed persistence)
   add_test(NAME storage_followthrough.cap${capacity}.${policy}.${case} COMMAND storage-tree-${capacity}-${policy} ${case})
  endforeach()
 endforeach()
 if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_TREE_LEARNER)
  set(TREE_CONTROLS exercise minimum exercise separator shape)
  foreach(i RANGE 1 5)
   tree_target(storage-tree-mutant-${capacity}-${i} ${capacity} 0)
   target_compile_definitions(storage-tree-mutant-${capacity}-${i} PRIVATE TREE_MUTANT=${i})
   if(i EQUAL 4)
    target_compile_definitions(storage-tree-mutant-${capacity}-${i} PRIVATE STORAGE_MUTANT_BAD_SEPARATOR=1)
   endif()
   math(EXPR index "${i}-1")
   list(GET TREE_CONTROLS ${index} case)
   add_test(NAME storage_followthrough.mutation.cap${capacity}.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-tree-mutant-${capacity}-${i}> ${case})
  endforeach()
 endif()
endforeach()
