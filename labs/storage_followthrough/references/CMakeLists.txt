cmake_minimum_required(VERSION 3.20)
project(storage_reference_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(REF_SOURCE "${CMAKE_CURRENT_LIST_DIR}/refs.c")
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_REF_LEARNER)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/references/refs.c")
  set(REF_SOURCE "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/references/refs.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/refs.c")
  set(REF_SOURCE "${CMASTERY_ASSESSOR_ROOT}/refs.c")
 else()
  message(FATAL_ERROR "Supply the explicit reference fork assessor source")
 endif()
endif()
function(ref_target name)
 add_executable(${name} tests.c "${REF_SOURCE}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ref_target(storage-references)
foreach(case exercise persistence reuse retirement stale rollback)
 add_test(NAME storage_followthrough.${case} COMMAND storage-references ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_REF_LEARNER)
 set(REF_CONTROLS stale retirement rollback)
 foreach(i RANGE 1 3)
  ref_target(storage-references-mutant-${i})
  target_compile_definitions(storage-references-mutant-${i} PRIVATE REF_MUTANT=${i})
  math(EXPR index "${i}-1")
  list(GET REF_CONTROLS ${index} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-references-mutant-${i}> ${case})
 endforeach()
endif()
