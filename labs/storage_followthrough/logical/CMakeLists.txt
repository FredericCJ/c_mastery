cmake_minimum_required(VERSION 3.20)
project(storage_logical_followthrough C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_STORAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../storage" CACHE PATH "Explicit storage public header")
set(CMASTERY_MEDIUM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../durability" CACHE PATH "Explicit completed failure medium")
if(NOT EXISTS "${CMASTERY_STORAGE_ROOT}/include/store.h" OR NOT EXISTS "${CMASTERY_MEDIUM_ROOT}/medium.h" OR NOT EXISTS "${CMASTERY_MEDIUM_ROOT}/medium.c")
 message(FATAL_ERROR "Supply storage header and completed medium using explicit provider paths")
endif()
set(SL_SOURCE "${CMAKE_CURRENT_LIST_DIR}/logical.c")
set(SL_MEDIUM "${CMASTERY_MEDIUM_ROOT}/medium.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/storage_followthrough/logical/logical.c")
  set(SL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 else()
  get_filename_component(SL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}" DIRECTORY)
 endif()
 set(SL_MEDIUM "${SL_ASSESSOR}/storage_followthrough/durability/medium.c")
 if(NOT CMASTERY_SL_LEARNER)
  set(SL_SOURCE "${SL_ASSESSOR}/storage_followthrough/logical/logical.c")
 endif()
endif()
function(sd_target name)
 add_executable(${name} tests.c "${SL_SOURCE}" "${SL_MEDIUM}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMASTERY_STORAGE_ROOT}/include" "${CMASTERY_MEDIUM_ROOT}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
sd_target(storage-logical)
foreach(case commit checkpoint repeated identity limits corruption histories)
 add_test(NAME storage_followthrough.${case} COMMAND storage-logical ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT AND NOT CMASTERY_SL_LEARNER)
 set(cases commit repeated commit identity limits)
 foreach(i RANGE 1 5)
  sd_target(storage-logical-mutant-${i})
  target_compile_definitions(storage-logical-mutant-${i} PRIVATE SL_MUTANT=${i})
  math(EXPR ix "${i}-1")
  list(GET cases ${ix} case)
  add_test(NAME storage_followthrough.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/expect_failure.py" $<TARGET_FILE:storage-logical-mutant-${i}> ${case})
 endforeach()
endif()
