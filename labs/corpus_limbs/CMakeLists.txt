cmake_minimum_required(VERSION 3.20)
project(corpus_limbs C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/limbs.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_limbs/limbs.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/corpus_limbs/limbs.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/limbs.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/limbs.c")
 else()
  message(FATAL_ERROR "Supply the explicit completed corpus_limbs implementation")
 endif()
endif()
function(limb_target name bits mutation)
 add_executable(${name} tests.c "${IMPL}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_definitions(${name} PRIVATE BL_BITS=${bits} BL_MUTANT=${mutation})
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
foreach(bits 16 8)
 limb_target(limbs${bits} ${bits} 0)
 foreach(anchor arithmetic ownership)
  add_test(NAME corpus_limbs.${anchor}${bits} COMMAND limbs${bits} ${anchor})
 endforeach()
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2 3 4 5 6)
   limb_target(limbs${bits}-mutant${mutant} ${bits} ${mutant})
   if(mutant LESS 5)
    set(anchor arithmetic)
   else()
    set(anchor ownership)
   endif()
   add_test(NAME corpus_limbs.mutation.${anchor}${bits}.${mutant}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py"
            $<TARGET_FILE:limbs${bits}-mutant${mutant}> ${anchor})
  endforeach()
 endif()
endforeach()
