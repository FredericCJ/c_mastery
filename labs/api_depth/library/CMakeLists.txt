cmake_minimum_required(VERSION 3.20)
project(Course VERSION 1.0.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(LC_RELEASE 1 CACHE STRING "Semantic release 1 or 2")
set(LC_REPRESENTATION 1 CACHE STRING "Private layout version")
set(LC_IMPLEMENTATION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/learner" CACHE PATH "Learner implementation directory")
set(LC_PROVIDER_ROOT "" CACHE PATH "Private staged earlier ApiBuffer producer")
option(LC_STATIC_HELPER "Private helper dependency for static propagation exercise" OFF)
set(LC_BACKEND "NONE" CACHE STRING "Auxiliary teaching backend: NONE, HOST, MODEL, BOTH")
if(LC_BACKEND STREQUAL "HOST")
 set(backend_sources "${LC_IMPLEMENTATION_DIR}/host.c")
elseif(LC_BACKEND STREQUAL "MODEL")
 set(backend_sources "${LC_IMPLEMENTATION_DIR}/board.c")
elseif(LC_BACKEND STREQUAL "BOTH")
 set(backend_sources "${LC_IMPLEMENTATION_DIR}/host.c" "${LC_IMPLEMENTATION_DIR}/board.c")
elseif(NOT LC_BACKEND STREQUAL "NONE")
 message(FATAL_ERROR "Unknown LC_BACKEND: select NONE, HOST, MODEL or BOTH")
endif()
if(backend_sources)
 add_library(teaching_backend STATIC ${backend_sources})
 target_include_directories(teaching_backend PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(NOT LC_PROVIDER_ROOT)
 message(FATAL_ERROR "LC_PROVIDER_ROOT must name the actual earlier producer source/header directory")
endif()
configure_file(lc_private.h.in lc_private.h @ONLY)
add_library(provider STATIC "${LC_PROVIDER_ROOT}/api_depth.c")
set_target_properties(provider PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(provider PRIVATE "${LC_PROVIDER_ROOT}")
add_library(course "${LC_IMPLEMENTATION_DIR}/lc.c")
set_target_properties(course PROPERTIES OUTPUT_NAME course VERSION ${LC_RELEASE}.0.0 SOVERSION 1 C_VISIBILITY_PRESET hidden)
target_compile_features(course PUBLIC c_std_17)
target_include_directories(course PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>" PRIVATE "${LC_PROVIDER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(course PRIVATE provider)
if(LC_STATIC_HELPER)
 add_library(helper STATIC "${LC_IMPLEMENTATION_DIR}/helper.c")
 target_compile_definitions(course PRIVATE LC_STATIC_HELPER=1)
 target_link_libraries(course PRIVATE helper)
 install(TARGETS helper EXPORT CourseTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(TARGETS course provider EXPORT CourseTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/lc.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CourseTargets NAMESPACE Course:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Course)
configure_package_config_file(CourseConfig.cmake.in CourseConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Course)
write_basic_package_version_file(CourseConfigVersion.cmake VERSION 1.0.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CourseConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CourseConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Course)
