cmake_minimum_required(VERSION 3.20)
project(api_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit own or common assessor root")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/api_depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/api_depth/api_depth.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/api_depth/api_depth.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/api_depth.c")
 endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
function(program target)
 add_executable(${target} tests/tests.c "${IMPL}")
 target_include_directories(${target} PRIVATE include)
endfunction()
program(api-tests)
enable_testing()
foreach(case append registry)
 add_test(NAME api_depth.${case} COMMAND api-tests ${case})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 1 2)
   set(mutant ${variant})
   if(case STREQUAL "registry")
    math(EXPR mutant "${variant}+2")
   endif()
   program(api-mutant-${mutant})
   target_compile_definitions(api-mutant-${mutant} PRIVATE API_MUTANT=${mutant})
   add_test(NAME api_depth.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:api-mutant-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
