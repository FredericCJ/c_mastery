cmake_minimum_required(VERSION 3.20)
project(corpus_events C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/events.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_events/events.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/corpus_events/events.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/events.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/events.c")
 else()
  message(FATAL_ERROR "Supply the explicit completed corpus_events implementation")
 endif()
endif()
function(event_target name bits mutation)
 add_executable(${name} tests.c "${IMPL}")
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_definitions(${name} PRIVATE EV_TEXT=${bits} EV_MUTANT=${mutation})
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
foreach(bits 16 1)
 event_target(events${bits} ${bits} 0)
 foreach(anchor parser consumer)
  add_test(NAME corpus_events.${anchor}${bits} COMMAND events${bits} ${anchor})
 endforeach()
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2 3 4)
   event_target(events${bits}-mutant${mutant} ${bits} ${mutant})
   if(mutant LESS 3)
    set(anchor parser)
   else()
    set(anchor consumer)
   endif()
   add_test(NAME corpus_events.mutation.${anchor}${bits}.${mutant}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py"
            $<TARGET_FILE:events${bits}-mutant${mutant}> ${anchor})
  endforeach()
 endif()
endforeach()
