cmake_minimum_required(VERSION 3.20)
project(c_mastery_foundation_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor or assessor/foundation_depth directory")
option(CMASTERY_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer on hosted binaries" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/depth.c")
if(CMASTERY_ASSESSOR_ROOT)
  if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
    message(FATAL_ERROR "CMASTERY_ASSESSOR_ROOT must be absolute")
  endif()
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/foundation_depth/depth.c")
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}/foundation_depth/depth.c")
  else()
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}/depth.c")
  endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(depth STATIC "${IMPL}")
target_include_directories(depth PUBLIC include)
add_executable(depth-tests tests/tests.c)
target_link_libraries(depth-tests PRIVATE depth)
if(CMASTERY_ASSESSOR_ROOT)
  get_filename_component(ASSESSOR_DIR "${IMPL}" DIRECTORY)
  target_include_directories(depth-tests PRIVATE "${ASSESSOR_DIR}")
  target_compile_definitions(depth-tests PRIVATE FD_ASSESSOR_TESTS)
endif()
add_executable(depth-oracle-driver tests/oracle_driver.c)
target_link_libraries(depth-oracle-driver PRIVATE depth)
add_executable(depth-authorization drills/authorization.c)
target_link_libraries(depth-authorization PRIVATE depth)
enable_testing()
foreach(case semantic extent cursor codec lifecycle overlap handles defense)
  add_test(NAME foundation_depth.${case} COMMAND depth-tests ${case})
  set_tests_properties(foundation_depth.${case} PROPERTIES TIMEOUT 20)
  if(CMASTERY_ASSESSOR_ROOT)
    add_executable(depth-mutant-${case} tests/tests.c "${IMPL}")
    target_include_directories(depth-mutant-${case} PRIVATE include "${ASSESSOR_DIR}")
    string(TOUPPER ${case} upper)
    target_compile_definitions(depth-mutant-${case} PRIVATE FD_MUTATE_${upper} FD_ASSESSOR_TESTS)
    add_test(NAME foundation_depth.mutation.${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:depth-mutant-${case}> ${case})
    set_tests_properties(foundation_depth.mutation.${case} PROPERTIES TIMEOUT 20)
  endif()
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  add_test(NAME foundation_depth.authorization_observation COMMAND depth-authorization)
endif()
