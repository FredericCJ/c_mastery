cmake_minimum_required(VERSION 3.20)
project(resilience C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or resilience assessor root")
set(CMASTERY_MESSAGING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../messaging" CACHE PATH "Accepted messaging workspace")
set(CMASTERY_MESSAGING_ASSESSOR_ROOT "" CACHE PATH "Explicit messaging assessor root")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
option(CMASTERY_TSAN "Separate hosted ThreadSanitizer" OFF)
set(IMPL src/resilience.c)
set(CHANNEL "${CMASTERY_MESSAGING_ROOT}/src/messaging.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/resilience/resilience.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/resilience/resilience.c")
  if(NOT CMASTERY_MESSAGING_ASSESSOR_ROOT)
   set(CMASTERY_MESSAGING_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/messaging")
  endif()
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/resilience.c")
 endif()
endif()
if(CMASTERY_MESSAGING_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_MESSAGING_ASSESSOR_ROOT}/messaging/messaging.c")
  set(CHANNEL "${CMASTERY_MESSAGING_ASSESSOR_ROOT}/messaging/messaging.c")
 else()
  set(CHANNEL "${CMASTERY_MESSAGING_ASSESSOR_ROOT}/messaging.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "ASan and TSan require separate executions")
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(rz_program target)
 add_executable(${target} tests/tests.c ${IMPL} ${CHANNEL})
 target_include_directories(${target} PRIVATE include "${CMASTERY_MESSAGING_ROOT}/include")
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
rz_program(resilience-tests)
enable_testing()
foreach(case deadline pressure wheel bucket retry gate hedge receiver service)
 add_test(NAME resilience.${case} COMMAND resilience-tests ${case})
 set_tests_properties(resilience.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2)
   rz_program(resilience-mutant-${case}-${mutant})
   string(TOUPPER ${case} CASE_UPPER)
   target_compile_definitions(resilience-mutant-${case}-${mutant} PRIVATE RZ_MUTATE_${CASE_UPPER}${mutant})
   add_test(NAME resilience.mutation.${case}.${mutant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:resilience-mutant-${case}-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
