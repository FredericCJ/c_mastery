function(rz_program target)
 add_executable(${target} tests/tests.c ${IMPL} ${CHANNEL})
 target_include_directories(${target} PRIVATE include "${CMASTERY_MESSAGING_ROOT}/include")
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
rz_program(resilience-tests)
enable_testing()
foreach(case deadline pressure wheel bucket retry gate hedge receiver service)
 add_test(NAME resilience.${case} COMMAND resilience-tests ${case})
 set_tests_properties(resilience.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutant 1 2)
   rz_program(resilience-mutant-${case}-${mutant})
   string(TOUPPER ${case} CASE_UPPER)
   target_compile_definitions(resilience-mutant-${case}-${mutant} PRIVATE RZ_MUTATE_${CASE_UPPER}${mutant})
   add_test(NAME resilience.mutation.${case}.${mutant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:resilience-mutant-${case}-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
