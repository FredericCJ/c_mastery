cmake_minimum_required(VERSION 3.16)
project(corpus_ring C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CR_PROVIDER "${CMAKE_CURRENT_SOURCE_DIR}/src/ring.c" CACHE FILEPATH "Explicit completed component")
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit private provider root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_ring/ring.c")
    set(CR_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/corpus_ring")
  else()
    set(CR_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
  endif()
  set(CR_PROVIDER "${CR_PRIVATE}/ring.c")
endif()
if(NOT EXISTS "${CR_PROVIDER}")
  message(FATAL_ERROR "Supply explicit completed ring provider")
endif()
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic-errors)
add_library(ring STATIC "${CR_PROVIDER}")
target_include_directories(ring PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(ring_tests tests.c)
target_link_libraries(ring_tests PRIVATE ring)
add_executable(ring_driver driver.c)
target_link_libraries(ring_driver PRIVATE ring)
enable_testing()
foreach(anchor lookup membership replicas routing)
  add_test(NAME corpus_ring.${anchor} COMMAND ring_tests ${anchor})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME corpus_ring.oracle COMMAND "${Python3_EXECUTABLE}" "${CR_PRIVATE}/oracle.py" $<TARGET_FILE:ring_driver>)
  set_tests_properties(corpus_ring.oracle PROPERTIES TIMEOUT 180)
  foreach(m RANGE 1 9)
    add_executable(ring_mutant_${m} tests.c "${CR_PROVIDER}")
    target_include_directories(ring_mutant_${m} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(ring_mutant_${m} PRIVATE CR_MUTANT=${m})
    if(m EQUAL 9)
      set(anchor membership)
    elseif(m LESS 3)
      set(anchor lookup)
    elseif(m LESS 5)
      set(anchor membership)
    elseif(m LESS 7)
      set(anchor replicas)
    else()
      set(anchor routing)
    endif()
    add_test(NAME corpus_ring.mutation.${anchor}.${m} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject.py" $<TARGET_FILE:ring_mutant_${m}> ${anchor})
  endforeach()
endif()
