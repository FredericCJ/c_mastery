cmake_minimum_required(VERSION 3.20)
project(higher_order C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/higher_order/free.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/higher_order")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(program target)
 add_executable(${target} tests/tests.c "${IMPL}/free.c" "${IMPL}/final.c")
 target_include_directories(${target} PRIVATE include)
endfunction()
program(higher-order-tests)
enable_testing()
set(index 0)
foreach(name free final)
 add_test(NAME higher_order.${name} COMMAND higher-order-tests ${name})
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 1 2)
   math(EXPR mutation "${index}*2+${variant}")
   program(higher-order-mutant-${mutation})
   target_compile_definitions(higher-order-mutant-${mutation} PRIVATE HO_MUTANT=${mutation})
   add_test(NAME higher_order.mutation.${name}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:higher-order-mutant-${mutation}> ${name})
  endforeach()
 endif()
 math(EXPR index "${index}+1")
endforeach()
foreach(probe type function)
 add_test(NAME higher_order.constraint.${probe} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_reject.py" "${CMAKE_C_COMPILER}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/${probe}_reject.c" "${CMAKE_CURRENT_SOURCE_DIR}/include")
endforeach()
