cmake_minimum_required(VERSION 3.20)
project(numeric_interval C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Common or family assessor root")
option(CMASTERY_SANITIZE "Address/undefined sanitizer campaign" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL src/interval.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/numeric_interval/interval.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/numeric_interval/interval.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/interval.c")
 endif()
endif()
function(ni_program name driver)
 add_executable(${name} ${driver} ${IMPL})
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
ni_program(numeric-interval-tests tests.c)
ni_program(numeric-interval-driver driver.c)
enable_testing()
add_test(NAME numeric_interval.enclosure COMMAND numeric-interval-tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutation 1 2)
  ni_program(numeric-interval-mutant-${mutation} tests.c)
  target_compile_definitions(numeric-interval-mutant-${mutation} PRIVATE NI_MUTANT=${mutation})
  add_test(NAME numeric_interval.mutation.enclosure.${mutation} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject_mutant.py" $<TARGET_FILE:numeric-interval-mutant-${mutation}>)
 endforeach()
endif()
