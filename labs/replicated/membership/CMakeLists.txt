cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_membership C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPM_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPM_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPM_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPM_HASH "${CMASTERY_REPLICATED_ROOT}/hashing/hashing.c")
set(RPM_OWN "${RPM_DIR}/membership.c")
set(RPM_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPM_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPM_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPM_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPM_HASH "${RPM_PRIVATE}/hashing/hashing.c")
 set(RPM_OWN "${RPM_PRIVATE}/membership/membership.c")
 set(RPM_BASE "${RPM_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPM_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_MEMBERSHIP_LEARNER)
 set(RPM_OWN "${RPM_DIR}/membership.c")
endif()
function(rpm_target name)
 add_executable(${name} ${RPM_OWN} ${RPM_HASH} ${RPM_BASE} ${RPM_DOD} "${RPM_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPM_DIR}" "${CMASTERY_REPLICATED_ROOT}/hashing" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpm_target(replicated-membership)
foreach(case membership_join membership_wire membership_auth membership_replay)
 add_test(NAME replicated.${case} COMMAND replicated-membership ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 9)
  rpm_target(replicated-membership-mutant-${number})
  target_compile_definitions(replicated-membership-mutant-${number} PRIVATE RPM_MUTATION=${number})
  if(number EQUAL 9)
   set(case membership_wire)
  elseif(number LESS_EQUAL 2)
   set(case membership_join)
  elseif(number LESS_EQUAL 3)
   set(case membership_wire)
  elseif(number LESS_EQUAL 6)
   set(case membership_auth)
  else()
   set(case membership_replay)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-membership-mutant-${number}> ${case})
 endforeach()
endif()
