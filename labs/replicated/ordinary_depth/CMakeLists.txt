set(RPO_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPO_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPO_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPO_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPO_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPO_BASE "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPO_BASE "${CMASTERY_ASSESSOR_ROOT}/replicated.c")
 endif()
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPO_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_ORDINARY_DEPTH_LEARNER)
 set(RPO_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
endif()
function(rpo_target name driver)
 add_executable(${name} ${RPO_BASE} ${RPO_DOD})
 if(driver)
  target_sources(${name} PRIVATE "${CMASTERY_REPLICATED_ROOT}/tests/oracle_driver.c")
 else()
  target_sources(${name} PRIVATE "${RPO_DIR}/tests.c")
 endif()
 target_include_directories(${name} PRIVATE "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpo_target(replicated-ordinary_depth FALSE)
rpo_target(replicated-original-oracle TRUE)
set(RPO_CASES log projection clocks counters sets sequence merkle wire replay)
foreach(case ${RPO_CASES} laws)
 add_test(NAME replicated.ordinary_${case} COMMAND replicated-ordinary_depth ordinary_${case})
endforeach()

 endforeach()
