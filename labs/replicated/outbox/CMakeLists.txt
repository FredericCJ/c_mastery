set(RPO_DIR "${CMAKE_CURRENT_LIST_DIR}")

set(CMASTERY_DOD_ROOT "${RPO_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(CMASTERY_STORAGE_ROOT "${RPO_DIR}/../../storage" CACHE PATH "Completed storage workspace")
set(CMASTERY_API_DEPTH_ROOT "${RPO_DIR}/../../api_depth" CACHE PATH "Completed API architecture workspace")
set(CMASTERY_HOSTED_SYSTEMS_ROOT "${RPO_DIR}/../../hosted_systems" CACHE PATH "Completed hosted systems stream workspace")
set(RPO_OWN "${RPO_DIR}/outbox.c" "${RPO_DIR}/transport.c")
set(RPO_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c" "${CMASTERY_REPLICATED_ROOT}/followthrough/durable.c")
set(RPO_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
set(RPO_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
set(RPO_CODEC "${CMASTERY_API_DEPTH_ROOT}/architecture/learner.c")
set(RPO_STREAM "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth/stream.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPO_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  foreach(pair "DOD;dod" "STORAGE;storage" "API_DEPTH;api_depth" "HOSTED_SYSTEMS;hosted_systems")
   list(GET pair 0 var)
   list(GET pair 1 dir)
   if(NOT CMASTERY_${var}_ASSESSOR_ROOT)
    set(CMASTERY_${var}_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/${dir}")
   endif()
  endforeach()
 else()
  set(RPO_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPO_OWN "${RPO_PRIVATE}/outbox/outbox.c" "${RPO_PRIVATE}/outbox/transport.c")
 set(RPO_BASE "${RPO_PRIVATE}/replicated.c" "${RPO_PRIVATE}/followthrough/durable.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPO_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_STORAGE_ASSESSOR_ROOT)
 set(RPO_STORE "${CMASTERY_STORAGE_ASSESSOR_ROOT}/store.c")
endif()
if(CMASTERY_API_DEPTH_ASSESSOR_ROOT)
 set(RPO_CODEC "${CMASTERY_API_DEPTH_ASSESSOR_ROOT}/architecture/core.c")
endif()
if(CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT)
 set(RPO_STREAM "${CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT}/stream_depth/stream.c")
endif()
if(CMASTERY_OUTBOX_LEARNER)
 set(RPO_OWN "${RPO_DIR}/outbox.c" "${RPO_DIR}/transport.c")
endif()
function(rpo_target name harness)
 add_executable(${name} ${RPO_OWN} ${RPO_BASE} ${RPO_DOD} ${RPO_STORE} ${RPO_CODEC} ${RPO_STREAM} "${RPO_DIR}/${harness}" "${RPO_DIR}/test_support.c")
 target_include_directories(${name} PRIVATE "${RPO_DIR}" "${CMASTERY_REPLICATED_ROOT}/followthrough" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include" "${CMASTERY_STORAGE_ROOT}/include" "${CMASTERY_API_DEPTH_ROOT}/architecture" "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpo_target(replicated-outbox tests.c)
foreach(case outbox_receive outbox_ack outbox_transport outbox_faults)
 add_test(NAME replicated.${case} COMMAND replicated-outbox ${case})
endforeach()
if(UNIX)
 rpo_target(replicated-outbox-native native.c)
 add_test(NAME replicated.outbox_native COMMAND replicated-outbox-native)
endif()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 11 20)
  if(number GREATER_EQUAL 19)
   rpo_target(replicated-outbox-mutant-${number} native.c)
  else()
   rpo_target(replicated-outbox-mutant-${number} tests.c)
  endif()
  target_compile_definitions(replicated-outbox-mutant-${number} PRIVATE RPO_MUTATION=${number})
  if(number EQUAL 11 OR number EQUAL 13)
   set(case outbox_receive)
  elseif(number EQUAL 12 OR number EQUAL 14)
   set(case outbox_ack)
  elseif(number LESS_EQUAL 16)
   set(case outbox_transport)
  elseif(number LESS_EQUAL 18)
   set(case outbox_faults)
  else()
   set(case outbox_native)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-outbox-mutant-${number}> ${case})
 endforeach()
endif()
