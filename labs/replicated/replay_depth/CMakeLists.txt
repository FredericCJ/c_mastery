function(rpr_target name native)
 add_executable(${name} "${RPR_OWN_ROOT}/replay_core.c" "${RPR_OWN_ROOT}/replay_wire.c" "${RPR_OWN_ROOT}/replay_store.c" "${RPR_OWN_ROOT}/replay_transport.c" ${RPR_BASE} ${RPR_HASH} ${RPR_DOD} ${RPR_STORE} ${RPR_ARCH_CORE} ${RPR_ARCH_EFFECTS} ${RPR_STREAM})
 if(native)
  target_sources(${name} PRIVATE "${RPR_OWN_ROOT}/replay_sink.c" ${RPR_LOG} "${RPR_DIR}/native.c")
 else()
  target_sources(${name} PRIVATE "${RPR_DIR}/tests.c" "${RPR_DIR}/oracle_driver.c")
 endif()
 target_include_directories(${name} PRIVATE "${RPR_DIR}" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_REPLICATED_ROOT}/hashing" "${CMASTERY_REPLICATED_ROOT}/followthrough" "${CMASTERY_DOD_ROOT}/include" "${CMASTERY_STORAGE_ROOT}/include" "${CMASTERY_API_DEPTH_ROOT}/architecture" "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
