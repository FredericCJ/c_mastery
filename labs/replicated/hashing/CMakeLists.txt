cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_hashing C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPH_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPH_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPH_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPH_OWN "${RPH_DIR}/hashing.c")
set(RPH_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPH_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPH_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPH_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPH_OWN "${RPH_PRIVATE}/hashing/hashing.c")
 set(RPH_BASE "${RPH_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPH_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_HASHING_LEARNER)
 set(RPH_OWN "${RPH_DIR}/hashing.c")
endif()
function(rph_target name)
 add_executable(${name} ${RPH_OWN} ${RPH_BASE} ${RPH_DOD} "${RPH_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPH_DIR}" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rph_target(replicated-hashing)
rph_target(replicated-hashing-schedule64)
target_compile_definitions(replicated-hashing-schedule64 PRIVATE RPH_SCHEDULE64=1)
foreach(case sha_stream hash_hmac hash_lifecycle hash_security)
 add_test(NAME replicated.${case} COMMAND replicated-hashing ${case})
 add_test(NAME replicated.schedule64.${case} COMMAND replicated-hashing-schedule64 ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 8)
  rph_target(replicated-hash-mutant-${number})
  target_compile_definitions(replicated-hash-mutant-${number} PRIVATE RPH_MUTATION=${number})
  if(number LESS_EQUAL 2)
   set(case sha_stream)
  elseif(number LESS_EQUAL 4)
   set(case hash_hmac)
  elseif(number LESS_EQUAL 6)
   set(case hash_lifecycle)
  else()
   set(case hash_security)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-hash-mutant-${number}> ${case})
 endforeach()
endif()
