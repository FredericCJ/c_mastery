cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_wire_depth C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPW_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPW_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPW_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPW_OWN "${RPW_DIR}/wire_depth.c")
set(RPW_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPW_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPW_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPW_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPW_OWN "${RPW_PRIVATE}/wire_depth/wire_depth.c")
 set(RPW_BASE "${RPW_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPW_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_WIRE_DEPTH_LEARNER)
 set(RPW_OWN "${RPW_DIR}/wire_depth.c")
endif()
set(CMASTERY_HOSTED_SYSTEMS_ROOT "${RPW_DIR}/../../hosted_systems" CACHE PATH "Completed hosted systems workspace")
set(RPW_STREAM "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth/stream.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/hosted_systems/stream_depth/stream.c")
  set(RPW_STREAM "${CMASTERY_ASSESSOR_ROOT}/hosted_systems/stream_depth/stream.c")
 elseif(CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT)
  set(RPW_STREAM "${CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT}/stream_depth/stream.c")
 endif()
endif()
function(rpw_target name)
 add_executable(${name} ${RPW_OWN} ${RPW_BASE} ${RPW_DOD} ${RPW_STREAM} "${RPW_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPW_DIR}" "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpw_target(replicated-wire_depth)
foreach(case wire_nested wire_limits wire_stream wire_evolution)
 add_test(NAME replicated.${case} COMMAND replicated-wire_depth ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 10)
  rpw_target(replicated-wire-depth-mutant-${number})
  target_compile_definitions(replicated-wire-depth-mutant-${number} PRIVATE RPW_MUTATION=${number})
  if(number LESS_EQUAL 2)
   set(case wire_nested)
  elseif(number LESS_EQUAL 6)
   set(case wire_limits)
  elseif(number EQUAL 7)
   set(case wire_stream)
  else()
   set(case wire_evolution)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-wire-depth-mutant-${number}> ${case})
 endforeach()
endif()
