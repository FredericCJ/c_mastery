cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_merkle_repair C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPK_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPK_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPK_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPK_HASH "${CMASTERY_REPLICATED_ROOT}/hashing/hashing.c")
set(RPK_OWN "${RPK_DIR}/merkle_repair.c")
set(RPK_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPK_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPK_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPK_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPK_HASH "${RPK_PRIVATE}/hashing/hashing.c")
 set(RPK_OWN "${RPK_PRIVATE}/merkle_repair/merkle_repair.c")
 set(RPK_BASE "${RPK_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPK_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_MERKLE_REPAIR_LEARNER)
 set(RPK_OWN "${RPK_DIR}/merkle_repair.c")
endif()
set(CMASTERY_HOSTED_SYSTEMS_ROOT "${RPK_DIR}/../../hosted_systems" CACHE PATH "Completed hosted systems workspace")
set(RPK_STREAM "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth/stream.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/hosted_systems/stream_depth/stream.c")
  set(RPK_STREAM "${CMASTERY_ASSESSOR_ROOT}/hosted_systems/stream_depth/stream.c")
 elseif(CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT)
  set(RPK_STREAM "${CMASTERY_HOSTED_SYSTEMS_ASSESSOR_ROOT}/stream_depth/stream.c")
 endif()
endif()
function(rpk_target name)
 add_executable(${name} ${RPK_OWN} ${RPK_HASH} ${RPK_BASE} ${RPK_DOD} ${RPK_STREAM} "${RPK_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPK_DIR}" "${CMASTERY_HOSTED_SYSTEMS_ROOT}/stream_depth" "${CMASTERY_REPLICATED_ROOT}/hashing" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpk_target(replicated-merkle_repair)
foreach(case merkle_proof merkle_wire merkle_repair merkle_cost)
 add_test(NAME replicated.${case} COMMAND replicated-merkle_repair ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 10)
  rpk_target(replicated-merkle-repair-mutant-${number})
  target_compile_definitions(replicated-merkle-repair-mutant-${number} PRIVATE RPK_MUTATION=${number})
  if(number LESS_EQUAL 4)
   set(case merkle_proof)
  elseif(number LESS_EQUAL 6)
   set(case merkle_wire)
  elseif(number LESS_EQUAL 8)
   set(case merkle_repair)
  else()
   set(case merkle_cost)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-merkle-repair-mutant-${number}> ${case})
 endforeach()
endif()
