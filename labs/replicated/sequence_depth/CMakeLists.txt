cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_sequence_depth C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPS_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPS_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPS_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(RPS_HASH "${CMASTERY_REPLICATED_ROOT}/hashing/hashing.c")
set(RPS_OWN "${RPS_DIR}/sequence_depth.c")
set(RPS_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPS_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPS_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
 else()
  set(RPS_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPS_HASH "${RPS_PRIVATE}/hashing/hashing.c")
 set(RPS_OWN "${RPS_PRIVATE}/sequence_depth/sequence_depth.c")
 set(RPS_BASE "${RPS_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPS_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_SEQUENCE_DEPTH_LEARNER)
 set(RPS_OWN "${RPS_DIR}/sequence_depth.c")
endif()
function(rps_target name)
 add_executable(${name} ${RPS_OWN} ${RPS_HASH} ${RPS_BASE} ${RPS_DOD} "${RPS_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPS_DIR}" "${CMASTERY_REPLICATED_ROOT}/hashing" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rps_target(replicated-sequence_depth)
foreach(case seq_tree seq_snapshot ot_context ot_delivery)
 add_test(NAME replicated.${case} COMMAND replicated-sequence_depth ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 10)
  rps_target(replicated-sequence-depth-mutant-${number})
  target_compile_definitions(replicated-sequence-depth-mutant-${number} PRIVATE RPS_MUTATION=${number})
  if(number LESS_EQUAL 2)
   set(case seq_tree)
  elseif(number LESS_EQUAL 4)
   set(case seq_snapshot)
  elseif(number EQUAL 5 OR number EQUAL 6 OR number EQUAL 9)
   set(case ot_context)
  else()
   set(case ot_delivery)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-sequence-depth-mutant-${number}> ${case})
 endforeach()
endif()
