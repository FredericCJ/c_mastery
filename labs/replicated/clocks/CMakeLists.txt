cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_clocks C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPC_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPC_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPC_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(CMASTERY_STORAGE_ROOT "${RPC_DIR}/../../storage" CACHE PATH "Completed storage workspace")
set(RPC_OWN "${RPC_DIR}/clocks.c")
set(RPC_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPC_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
set(RPC_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPC_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
  if(NOT CMASTERY_STORAGE_ASSESSOR_ROOT)
   set(CMASTERY_STORAGE_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/storage")
  endif()
 else()
  set(RPC_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPC_OWN "${RPC_PRIVATE}/clocks/clocks.c")
 set(RPC_BASE "${RPC_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPC_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_STORAGE_ASSESSOR_ROOT)
 set(RPC_STORE "${CMASTERY_STORAGE_ASSESSOR_ROOT}/store.c")
endif()
if(CMASTERY_CLOCKS_LEARNER)
 set(RPC_OWN "${RPC_DIR}/clocks.c")
endif()
function(rpc_target name)
 add_executable(${name} ${RPC_OWN} ${RPC_BASE} ${RPC_DOD} ${RPC_STORE} "${RPC_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPC_DIR}" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include" "${CMASTERY_STORAGE_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpc_target(replicated-clocks)
foreach(case clock_identity clock_store clock_policy)
 add_test(NAME replicated.${case} COMMAND replicated-clocks ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 6)
  rpc_target(replicated-clock-mutant-${number})
  target_compile_definitions(replicated-clock-mutant-${number} PRIVATE RPC_MUTATION=${number})
  if(number LESS_EQUAL 2)
   set(case clock_identity)
  elseif(number LESS_EQUAL 4)
   set(case clock_store)
  else()
   set(case clock_policy)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-clock-mutant-${number}> ${case})
 endforeach()
endif()
