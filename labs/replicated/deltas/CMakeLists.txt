cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(replicated_deltas C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RPD_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMASTERY_REPLICATED_ROOT "${RPD_DIR}/.." CACHE PATH "Completed replicated workspace")
set(CMASTERY_DOD_ROOT "${RPD_DIR}/../../dod" CACHE PATH "Completed DOD workspace")
set(CMASTERY_STORAGE_ROOT "${RPD_DIR}/../../storage" CACHE PATH "Completed storage workspace")
set(RPD_HASH "${CMASTERY_REPLICATED_ROOT}/hashing/hashing.c")
set(RPD_OWN "${RPD_DIR}/deltas.c")
set(RPD_BASE "${CMASTERY_REPLICATED_ROOT}/src/replicated.c")
set(RPD_DOD "${CMASTERY_DOD_ROOT}/src/dod.c")
set(RPD_STORE "${CMASTERY_STORAGE_ROOT}/src/store.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/replicated/replicated.c")
  set(RPD_PRIVATE "${CMASTERY_ASSESSOR_ROOT}/replicated")
  if(NOT CMASTERY_DOD_ASSESSOR_ROOT)
   set(CMASTERY_DOD_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/dod")
  endif()
  if(NOT CMASTERY_STORAGE_ASSESSOR_ROOT)
   set(CMASTERY_STORAGE_ASSESSOR_ROOT "${CMASTERY_ASSESSOR_ROOT}/storage")
  endif()
 else()
  set(RPD_PRIVATE "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(RPD_HASH "${RPD_PRIVATE}/hashing/hashing.c")
 set(RPD_OWN "${RPD_PRIVATE}/deltas/deltas.c")
 set(RPD_BASE "${RPD_PRIVATE}/replicated.c")
endif()
if(CMASTERY_DOD_ASSESSOR_ROOT)
 set(RPD_DOD "${CMASTERY_DOD_ASSESSOR_ROOT}/dod.c")
endif()
if(CMASTERY_STORAGE_ASSESSOR_ROOT)
 set(RPD_STORE "${CMASTERY_STORAGE_ASSESSOR_ROOT}/store.c")
endif()
if(CMASTERY_DELTAS_LEARNER)
 set(RPD_OWN "${RPD_DIR}/deltas.c")
endif()
function(rpd_target name)
 add_executable(${name} ${RPD_OWN} ${RPD_HASH} ${RPD_BASE} ${RPD_DOD} ${RPD_STORE} "${RPD_DIR}/tests.c")
 target_include_directories(${name} PRIVATE "${RPD_DIR}" "${CMASTERY_REPLICATED_ROOT}/hashing" "${CMASTERY_REPLICATED_ROOT}/include" "${CMASTERY_DOD_ROOT}/include" "${CMASTERY_STORAGE_ROOT}/include")
 if(NOT MSVC)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMASTERY_SANITIZE)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
  endif()
  if(CMASTERY_TSAN)
   target_compile_options(${name} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=thread)
  endif()
 endif()
endfunction()
rpd_target(replicated-deltas)
foreach(case delta_sets delta_wire delta_durable delta_protocol)
 add_test(NAME replicated.${case} COMMAND replicated-deltas ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(number RANGE 1 10)
  rpd_target(replicated-delta-mutant-${number})
  target_compile_definitions(replicated-delta-mutant-${number} PRIVATE RPD_MUTATION=${number})
  if(number LESS_EQUAL 2)
   set(case delta_sets)
  elseif(number EQUAL 3 OR number EQUAL 10)
   set(case delta_wire)
  elseif(number EQUAL 4 OR number EQUAL 6 OR number EQUAL 7)
   set(case delta_durable)
  else()
   set(case delta_protocol)
  endif()
  add_test(NAME replicated.mutation.${case}.${number}
   COMMAND "${Python3_EXECUTABLE}" "${CMASTERY_REPLICATED_ROOT}/followthrough/expect_failure.py" $<TARGET_FILE:replicated-delta-mutant-${number}> ${case})
 endforeach()
endif()
