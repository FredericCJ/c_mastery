cmake_minimum_required(VERSION 3.20)
project(b03_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root or b03_depth directory")
option(CMASTERY_SANITIZE "Hosted ASan and UBSan" OFF)
set(IMPL src/depth.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/b03_depth/depth.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/b03_depth/depth.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/depth.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(b03-tests tests/tests.c ${IMPL})
target_include_directories(b03-tests PRIVATE include)
enable_testing()
find_package(Threads REQUIRED)
add_executable(b03-concurrent tests/concurrent.c ${IMPL})
target_include_directories(b03-concurrent PRIVATE include)
target_link_libraries(b03-concurrent PRIVATE Threads::Threads)
if(CMASTERY_ASSESSOR_ROOT)
 add_test(NAME b03_depth.concurrent_smoke COMMAND b03-concurrent)
 set_tests_properties(b03_depth.concurrent_smoke PROPERTIES TIMEOUT 30)
endif()
foreach(case io cancel mmio deadline publication)
 add_test(NAME b03_depth.${case} COMMAND b03-tests ${case})
 set_tests_properties(b03_depth.${case} PROPERTIES TIMEOUT 20)
 if(CMASTERY_ASSESSOR_ROOT)
  add_executable(b03-mutant-${case} tests/tests.c ${IMPL})
  target_include_directories(b03-mutant-${case} PRIVATE include)
  string(TOUPPER ${case} CASE_UPPER)
  target_compile_definitions(b03-mutant-${case} PRIVATE B3_MUTATE_${CASE_UPPER})
  add_test(NAME b03_depth.mutation.${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:b03-mutant-${case}> ${case})
  set_tests_properties(b03_depth.mutation.${case} PROPERTIES TIMEOUT 20)
 endif()
endforeach()
