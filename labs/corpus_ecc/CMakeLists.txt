cmake_minimum_required(VERSION 3.16)
project(corpus_ecc C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CMASTERY_SANITIZE "Address/undefined sanitizers" OFF)
if(CMASTERY_SANITIZE)
  add_compile_options(-O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit completed reference implementation")
set(source "${CMAKE_CURRENT_SOURCE_DIR}/src/ecc.c")
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/corpus_ecc/ecc.c")
    set(source "${CMASTERY_ASSESSOR_ROOT}/corpus_ecc/ecc.c")
  elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ecc.c")
    set(source "${CMASTERY_ASSESSOR_ROOT}/ecc.c")
  else()
    message(FATAL_ERROR "Supply the explicit completed corpus_ecc implementation")
  endif()
endif()
add_executable(ecc_tests tests.c "${source}")
target_include_directories(ecc_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(ecc_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
foreach(case IN ITEMS 8 16 erasures burst)
  add_test(NAME corpus_ecc.${case} COMMAND ecc_tests ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(number RANGE 1 6)
    add_executable(ecc_mutant${number} tests.c "${source}")
    target_include_directories(ecc_mutant${number} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(ecc_mutant${number} PRIVATE EC_MUTATION=${number})
    target_compile_options(ecc_mutant${number} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
  foreach(case IN ITEMS 8 16)
    foreach(number RANGE 1 2)
      add_test(NAME corpus_ecc.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:ecc_mutant${number}> ${case})
    endforeach()
  endforeach()
  add_test(NAME corpus_ecc.mutation.erasures.1 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:ecc_mutant3> erasures)
  add_test(NAME corpus_ecc.mutation.erasures.2 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:ecc_mutant4> erasures)
  add_test(NAME corpus_ecc.mutation.burst.1 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:ecc_mutant5> burst)
  add_test(NAME corpus_ecc.mutation.burst.2 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py" $<TARGET_FILE:ecc_mutant6> burst)
endif()
