cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_allocator_audit C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OAA_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OAA_PRIVATE "${OAA_DIR}/src")
set(OAA_OWNERSHIP "${OAA_DIR}/../src/ownership.c")
set(CMASTERY_ALGORITHMS_ROOT "${OAA_DIR}/../../algorithms" CACHE PATH "Completed containers.array workspace")
set(CMASTERY_ERRORS_ROOT "${OAA_DIR}/../../errors" CACHE PATH "Completed errors.result workspace")
set(OAA_ARRAY "${CMASTERY_ALGORITHMS_ROOT}/src/v4_algo.c")
set(OAA_ERRORS "${CMASTERY_ERRORS_ROOT}/src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OAA_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OAA_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OAA_PRIVATE "${OAA_ASSESSOR}/allocator_audit")
 set(OAA_OWNERSHIP "${OAA_ASSESSOR}/ownership.c")
 set(OAA_ARRAY "${OAA_ASSESSOR}/../algorithms/v4_algo.c")
 set(OAA_ERRORS "${OAA_ASSESSOR}/../errors/errors.c")
endif()
if(CMASTERY_ALGORITHMS_ASSESSOR_ROOT)
 set(OAA_ARRAY "${CMASTERY_ALGORITHMS_ASSESSOR_ROOT}/v4_algo.c")
endif()
if(CMASTERY_ERRORS_ASSESSOR_ROOT)
 set(OAA_ERRORS "${CMASTERY_ERRORS_ASSESSOR_ROOT}/errors.c")
endif()
if(NOT EXISTS "${CMASTERY_ERRORS_ROOT}/include/errors.h" OR NOT EXISTS "${CMASTERY_ALGORITHMS_ROOT}/include/v4_algo.h")
 message(STATUS "Allocator audit requires explicit containers.array and errors.result workspaces")
 return()
endif()
if(CMASTERY_OAA_LEARNER)
 set(OAA_PRIVATE "${OAA_DIR}/src")
 set(OAA_OWNERSHIP "${OAA_DIR}/../src/ownership.c")
endif()
function(oaa_target name)
 add_executable(${name} "${OAA_DIR}/tests/tests.c" "${OAA_PRIVATE}/allocator_audit.c" "${OAA_PRIVATE}/consumers.c" "${OAA_OWNERSHIP}" "${OAA_ARRAY}" "${OAA_ERRORS}")
 target_include_directories(${name} PRIVATE "${OAA_DIR}/include" "${OAA_DIR}/../include" "${CMASTERY_ALGORITHMS_ROOT}/include" "${CMASTERY_ERRORS_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
oaa_target(ownership-allocator_audit)
foreach(case allocator_audit_base allocator_audit_resize allocator_audit_rollback allocator_audit_poison allocator_audit_counters allocator_audit_consumers allocator_audit_recursive)
 add_test(NAME ownership.${case} COMMAND ownership-allocator_audit ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OAA_CASES allocator_audit_rollback allocator_audit_resize allocator_audit_poison allocator_audit_recursive allocator_audit_base allocator_audit_base)
 foreach(i RANGE 1 6)
  oaa_target(ownership-allocator_audit-mutant-${i})
  if(i EQUAL 5)
   target_compile_definitions(ownership-allocator_audit-mutant-${i} PRIVATE OW_MUTANT=9)
  elseif(i EQUAL 6)
   target_compile_definitions(ownership-allocator_audit-mutant-${i} PRIVATE OW_MUTANT=19)
  else()
   target_compile_definitions(ownership-allocator_audit-mutant-${i} PRIVATE OAA_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OAA_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OAA_DIR}/expect_failure.py" $<TARGET_FILE:ownership-allocator_audit-mutant-${i}> ${case})
 endforeach()
endif()
