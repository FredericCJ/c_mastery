cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_authority_boundary C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OAF_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OAF_PRIVATE "${OAF_DIR}/src")
set(OAF_OWNERSHIP "${OAF_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OAF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OAF_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OAF_PRIVATE "${OAF_ASSESSOR}/authority_boundary")
 set(OAF_OWNERSHIP "${OAF_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OAF_LEARNER)
 set(OAF_PRIVATE "${OAF_DIR}/src")
 set(OAF_OWNERSHIP "${OAF_DIR}/../src/ownership.c")
endif()
function(oaf_target name)
 add_executable(${name} "${OAF_DIR}/tests/tests.c" "${OAF_PRIVATE}/authority_boundary.c" "${OAF_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OAF_DIR}/include" "${OAF_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
oaf_target(ownership-authority_boundary)
foreach(case authority_defect authority_bounds authority_wire authority_rights authority_revoke authority_atomic)
 add_test(NAME ownership.${case} COMMAND ownership-authority_boundary ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OAF_CASES authority_defect authority_bounds authority_rights authority_revoke authority_rights authority_wire authority_rights authority_revoke)
 foreach(i RANGE 1 8)
  oaf_target(ownership-authority_boundary-mutant-${i})
  if(i EQUAL 1)
   target_compile_definitions(ownership-authority_boundary-mutant-${i} PRIVATE OW_MUTANT=10)
  elseif(i EQUAL 2)
   target_compile_definitions(ownership-authority_boundary-mutant-${i} PRIVATE OW_MUTANT=20)
  else()
   target_compile_definitions(ownership-authority_boundary-mutant-${i} PRIVATE OAF_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OAF_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OAF_DIR}/expect_failure.py" $<TARGET_FILE:ownership-authority_boundary-mutant-${i}> ${case})
 endforeach()
endif()
