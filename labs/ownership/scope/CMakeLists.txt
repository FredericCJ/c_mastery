cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_scope C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OCS_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OCS_PRIVATE "${OCS_DIR}/src")
set(OCS_OWNERSHIP "${OCS_DIR}/../src/ownership.c")
set(CMASTERY_ERRORS_ROOT "${OCS_DIR}/../../errors" CACHE PATH "Completed early T65 errors workspace")
set(OCS_ERRORS "${CMASTERY_ERRORS_ROOT}/src/errors.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OCS_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
  set(OCS_ERRORS "${CMASTERY_ASSESSOR_ROOT}/errors/errors.c")
 else()
  set(OCS_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OCS_PRIVATE "${OCS_ASSESSOR}/scope")
 set(OCS_OWNERSHIP "${OCS_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_ERRORS_ASSESSOR_ROOT)
 set(OCS_ERRORS "${CMASTERY_ERRORS_ASSESSOR_ROOT}/errors.c")
endif()
if(NOT EXISTS "${OCS_ERRORS}")
 message(STATUS "Scope composition requires explicit completed early T65 errors workspace")
 return()
endif()
if(CMASTERY_OCS_LEARNER)
 set(OCS_PRIVATE "${OCS_DIR}/src")
 set(OCS_OWNERSHIP "${OCS_DIR}/../src/ownership.c")
endif()
function(ocs_target name)
 add_executable(${name} "${OCS_DIR}/tests/tests.c" "${OCS_PRIVATE}/scope.c" "${OCS_OWNERSHIP}" "${OCS_ERRORS}")
 target_include_directories(${name} PRIVATE "${OCS_DIR}/include" "${OCS_DIR}/../include" "${CMASTERY_ERRORS_ROOT}/include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
ocs_target(ownership-scope)
foreach(case scope_policy scope_jump scope_capacity scope_publish scope_recovery scope_identity)
 add_test(NAME ownership.${case} COMMAND ownership-scope ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OCS_CASES scope_jump scope_recovery scope_recovery scope_identity scope_recovery scope_recovery scope_policy scope_policy)
 foreach(i RANGE 1 8)
  ocs_target(ownership-scope-mutant-${i})
  if(i EQUAL 7)
   target_compile_definitions(ownership-scope-mutant-${i} PRIVATE OW_MUTANT=2)
  elseif(i EQUAL 8)
   target_compile_definitions(ownership-scope-mutant-${i} PRIVATE OW_MUTANT=12)
  else()
   target_compile_definitions(ownership-scope-mutant-${i} PRIVATE OCS_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OCS_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OCS_DIR}/expect_failure.py" $<TARGET_FILE:ownership-scope-mutant-${i}> ${case})
 endforeach()
endif()
