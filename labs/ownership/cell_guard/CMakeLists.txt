cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_cell_guard C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(OCG_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OCG_PRIVATE "${OCG_DIR}/src")
set(OCG_OWNERSHIP "${OCG_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OCG_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OCG_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OCG_PRIVATE "${OCG_ASSESSOR}/cell_guard")
 set(OCG_OWNERSHIP "${OCG_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OCG_LEARNER)
 set(OCG_PRIVATE "${OCG_DIR}/src")
 set(OCG_OWNERSHIP "${OCG_DIR}/../src/ownership.c")
endif()
function(omm_target name)
 add_executable(${name} "${OCG_DIR}/tests/tests.c" "${OCG_PRIVATE}/cell_guard.c" "${OCG_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OCG_DIR}/include" "${OCG_DIR}/../include")
 target_link_libraries(${name} PRIVATE Threads::Threads)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
omm_target(ownership-cell_guard)
foreach(case cell_guard_base cell_guard_orders cell_guard_callbacks cell_guard_close cell_guard_identity cell_guard_threads cell_guard_async_close)
 add_test(NAME ownership.${case} COMMAND ownership-cell_guard ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OCG_CASES cell_guard_callbacks cell_guard_callbacks cell_guard_close cell_guard_identity cell_guard_base cell_guard_base)
 foreach(i RANGE 1 6)
  omm_target(ownership-cell_guard-mutant-${i})
  if(i EQUAL 5)
   target_compile_definitions(ownership-cell_guard-mutant-${i} PRIVATE OW_MUTANT=5)
  elseif(i EQUAL 6)
   target_compile_definitions(ownership-cell_guard-mutant-${i} PRIVATE OW_MUTANT=15)
  else()
   target_compile_definitions(ownership-cell_guard-mutant-${i} PRIVATE OCG_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OCG_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OCG_DIR}/expect_failure.py" $<TARGET_FILE:ownership-cell_guard-mutant-${i}> ${case})
 endforeach()
endif()
