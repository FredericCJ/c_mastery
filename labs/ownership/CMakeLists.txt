cmake_minimum_required(VERSION 3.20)
project(c_mastery_ownership C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
option(CMASTERY_TSAN "Separate hosted thread sanitizer" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/ownership.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "ASan and TSan are distinct campaigns")
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
add_executable(ownership-tests tests/tests.c "${IMPL}")
target_include_directories(ownership-tests PRIVATE include)
target_link_libraries(ownership-tests PRIVATE Threads::Threads)
add_executable(ownership-oracle-driver tests/oracle_driver.c "${IMPL}")
target_include_directories(ownership-oracle-driver PRIVATE include)
enable_testing()
set(index 0)
foreach(case frontier scope move borrows cell shared cow pin allocator facade)
 add_test(NAME ownership.${case} COMMAND ownership-tests ${case})
 set_tests_properties(ownership.${case} PROPERTIES TIMEOUT 30)
 math(EXPR index "${index}+1")
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 0 1)
   math(EXPR mutant "${index}+10*${variant}")
   add_executable(ownership-mutant-${mutant} tests/tests.c "${IMPL}")
   target_include_directories(ownership-mutant-${mutant} PRIVATE include)
   target_link_libraries(ownership-mutant-${mutant} PRIVATE Threads::Threads)
   target_compile_definitions(ownership-mutant-${mutant} PRIVATE OW_MUTANT=${mutant})
   add_test(NAME ownership.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:ownership-mutant-${mutant}> ${case})
  endforeach()
 endif()
endforeach()
include("${CMAKE_CURRENT_SOURCE_DIR}/frontier/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/scope/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/move/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/strides/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/cell_guard/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/references/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/cow_resources/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/pin_async/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/allocator_audit/CMakeLists.txt")
include("${CMAKE_CURRENT_SOURCE_DIR}/authority_boundary/CMakeLists.txt")
