cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_strides C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OSV_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OSV_PRIVATE "${OSV_DIR}/src")
set(OSV_OWNERSHIP "${OSV_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OSV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OSV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OSV_PRIVATE "${OSV_ASSESSOR}/strides")
 set(OSV_OWNERSHIP "${OSV_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OSV_LEARNER)
 set(OSV_PRIVATE "${OSV_DIR}/src")
 set(OSV_OWNERSHIP "${OSV_DIR}/../src/ownership.c")
endif()
function(omm_target name)
 add_executable(${name} "${OSV_DIR}/tests/tests.c" "${OSV_PRIVATE}/strides.c" "${OSV_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OSV_DIR}/include" "${OSV_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
omm_target(ownership-strides)
foreach(case strides_base strides_sets strides_tile strides_transaction strides_identity strides_bounds strides_access)
 add_test(NAME ownership.${case} COMMAND ownership-strides ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OSV_CASES strides_sets strides_sets strides_bounds strides_transaction strides_identity strides_tile strides_base strides_base)
 foreach(i RANGE 1 8)
  omm_target(ownership-strides-mutant-${i})
  if(i EQUAL 7)
   target_compile_definitions(ownership-strides-mutant-${i} PRIVATE OW_MUTANT=4)
  elseif(i EQUAL 8)
   target_compile_definitions(ownership-strides-mutant-${i} PRIVATE OW_MUTANT=14)
  else()
   target_compile_definitions(ownership-strides-mutant-${i} PRIVATE OSV_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OSV_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OSV_DIR}/expect_failure.py" $<TARGET_FILE:ownership-strides-mutant-${i}> ${case})
 endforeach()
endif()
