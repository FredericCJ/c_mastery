cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ownership_move C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OMM_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(OMM_PRIVATE "${OMM_DIR}/src")
set(OMM_OWNERSHIP "${OMM_DIR}/../src/ownership.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/ownership/ownership.c")
  set(OMM_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/ownership")
 else()
  set(OMM_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(OMM_PRIVATE "${OMM_ASSESSOR}/move")
 set(OMM_OWNERSHIP "${OMM_ASSESSOR}/ownership.c")
endif()
if(CMASTERY_OMM_LEARNER)
 set(OMM_PRIVATE "${OMM_DIR}/src")
 set(OMM_OWNERSHIP "${OMM_DIR}/../src/ownership.c")
endif()
function(omm_target name)
 add_executable(${name} "${OMM_DIR}/tests/tests.c" "${OMM_PRIVATE}/move.c" "${OMM_PRIVATE}/registry.c" "${OMM_OWNERSHIP}")
 target_include_directories(${name} PRIVATE "${OMM_DIR}/include" "${OMM_DIR}/../include")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
omm_target(ownership-move)
foreach(case move_base move_inline move_admission move_registry move_theft move_retirement)
 add_test(NAME ownership.${case} COMMAND ownership-move ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(OMM_CASES move_inline move_inline move_registry move_registry move_base move_base)
 foreach(i RANGE 1 6)
  omm_target(ownership-move-mutant-${i})
  if(i EQUAL 5)
   target_compile_definitions(ownership-move-mutant-${i} PRIVATE OW_MUTANT=3)
  elseif(i EQUAL 6)
   target_compile_definitions(ownership-move-mutant-${i} PRIVATE OW_MUTANT=13)
  else()
   target_compile_definitions(ownership-move-mutant-${i} PRIVATE OMM_MUTATION=${i})
  endif()
  math(EXPR index "${i}-1")
  list(GET OMM_CASES ${index} case)
  add_test(NAME ownership.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${OMM_DIR}/expect_failure.py" $<TARGET_FILE:ownership-move-mutant-${i}> ${case})
 endforeach()
endif()
