cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_receivers C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPV_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPV_CORE "${MPV_DIR}/src")
set(MPV_BASE "${MPV_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPV_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPV_CORE "${MPV_ASSESSOR}/receivers")
 set(MPV_BASE "${MPV_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPV_LEARNER)
 set(MPV_CORE "${MPV_DIR}/src")
 set(MPV_BASE "${MPV_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPV_DIR}/tests/tests.c" "${MPV_CORE}/receivers.c" ${MPV_BASE})
 target_include_directories(${name} PRIVATE "${MPV_DIR}/include" "${MPV_DIR}/../include" "${MPV_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-receivers)
foreach(case receiver_dispatch receiver_effects receiver_device receiver_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-receivers ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPV_CASES receiver_dispatch receiver_dispatch receiver_effects receiver_device receiver_device receiver_device receiver_base receiver_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-receivers-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-receivers-mutant-${i} PRIVATE MPV_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+2")
   target_compile_definitions(metaprog-receivers-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPV_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPV_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-receivers-mutant-${i}> ${case})
 endforeach()
endif()
