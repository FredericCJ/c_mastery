cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_logging C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPL_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPL_CORE "${MPL_DIR}/src")
set(MPL_BASE "${MPL_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPL_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPL_CORE "${MPL_ASSESSOR}/logging")
 set(MPL_BASE "${MPL_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPL_LEARNER)
 set(MPL_CORE "${MPL_DIR}/src")
 set(MPL_BASE "${MPL_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPL_DIR}/tests/tests.c" "${MPL_CORE}/logging.c" ${MPL_BASE})
 target_include_directories(${name} PRIVATE "${MPL_DIR}/include" "${MPL_DIR}/../include" "${MPL_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-logging)
foreach(case logging_eval logging_copy logging_atomic logging_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-logging ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPL_CASES logging_eval logging_atomic logging_atomic logging_copy logging_copy logging_atomic logging_base logging_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-logging-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-logging-mutant-${i} PRIVATE MPL_MUTATION=${i})
  else()
   math(EXPR baseline "${i}")
   target_compile_definitions(metaprog-logging-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPL_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPL_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-logging-mutant-${i}> ${case})
 endforeach()
endif()
