cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_policy_effects C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPP_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPP_CORE "${MPP_DIR}/src")
set(MPP_BASE "${MPP_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPP_CORE "${MPP_ASSESSOR}/policy_effects")
 set(MPP_BASE "${MPP_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPP_LEARNER)
 set(MPP_CORE "${MPP_DIR}/src")
 set(MPP_BASE "${MPP_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPP_DIR}/tests/tests.c" "${MPP_CORE}/policy_effects.c" ${MPP_BASE})
 target_include_directories(${name} PRIVATE "${MPP_DIR}/include" "${MPP_DIR}/../include" "${MPP_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-policy_effects)
foreach(case policy_order policy_saturation policy_failure policy_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-policy_effects ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPP_CASES policy_failure policy_failure policy_saturation policy_order policy_saturation policy_order policy_base policy_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-policy_effects-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-policy_effects-mutant-${i} PRIVATE MPP_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-2")
   target_compile_definitions(metaprog-policy_effects-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPP_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPP_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-policy_effects-mutant-${i}> ${case})
 endforeach()
endif()
