cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_capacity C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPC_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPC_CORE "${MPC_DIR}/src")
set(MPC_BASE "${MPC_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPC_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPC_CORE "${MPC_ASSESSOR}/capacity")
 set(MPC_BASE "${MPC_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPC_LEARNER)
 set(MPC_CORE "${MPC_DIR}/src")
 set(MPC_BASE "${MPC_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPC_DIR}/tests/tests.c" "${MPC_CORE}/capacity.c" ${MPC_BASE})
 target_include_directories(${name} PRIVATE "${MPC_DIR}/include" "${MPC_DIR}/../include" "${MPC_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-capacity)
foreach(case capacity_fixed capacity_runtime capacity_refusal capacity_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-capacity ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPC_CASES capacity_fixed capacity_fixed capacity_runtime capacity_refusal capacity_refusal capacity_runtime capacity_base capacity_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-capacity-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-capacity-mutant-${i} PRIVATE MPC_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-4")
   target_compile_definitions(metaprog-capacity-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPC_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPC_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-capacity-mutant-${i}> ${case})
 endforeach()
endif()
