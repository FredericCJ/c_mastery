cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_format_stream C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MFP_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MFP_CORE "${MFP_DIR}/src")
set(MFP_BASE "${MFP_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MFP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MFP_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MFP_CORE "${MFP_ASSESSOR}/format_stream")
 set(MFP_BASE "${MFP_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MFP_LEARNER)
 set(MFP_CORE "${MFP_DIR}/src")
 set(MFP_BASE "${MFP_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MFP_DIR}/tests/tests.c" "${MFP_CORE}/format_stream.c" ${MFP_BASE})
 target_include_directories(${name} PRIVATE "${MFP_DIR}/include" "${MFP_DIR}/../include" "${MFP_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-format_stream)
foreach(case format_parse format_atomic format_stream format_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-format_stream ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MFP_CASES format_parse format_atomic format_atomic format_stream format_stream format_stream format_base format_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-format_stream-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-format_stream-mutant-${i} PRIVATE MFP_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+8")
   target_compile_definitions(metaprog-format_stream-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MFP_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MFP_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-format_stream-mutant-${i}> ${case})
 endforeach()
endif()
