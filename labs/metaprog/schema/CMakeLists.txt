cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_schema C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPS_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPS_CORE "${MPS_DIR}/src")
set(MPS_BASE "${MPS_DIR}/../src/metaprog.c")
set(CMASTERY_API_DEPTH_ROOT "${MPS_DIR}/../../api_depth" CACHE PATH "Completed architecture workspace")
set(MPS_CODEC "${CMASTERY_API_DEPTH_ROOT}/architecture/learner.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPS_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
  set(MPS_CODEC "${CMASTERY_ASSESSOR_ROOT}/api_depth/architecture/core.c")
 else()
  set(MPS_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPS_CORE "${MPS_ASSESSOR}/schema")
 set(MPS_BASE "${MPS_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_API_DEPTH_ASSESSOR_ROOT)
 set(MPS_CODEC "${CMASTERY_API_DEPTH_ASSESSOR_ROOT}/architecture/core.c")
endif()
if(NOT EXISTS "${MPS_CODEC}")
 message(STATUS "schema requires an explicit completed api_depth codec.records workspace")
 return()
endif()
if(CMASTERY_MPS_LEARNER)
 set(MPS_CORE "${MPS_DIR}/src")
 set(MPS_BASE "${MPS_DIR}/../src/metaprog.c")
endif()
set(MPS_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/schema-generated")
add_custom_command(OUTPUT "${MPS_GENERATED}/schema_stamp.h"
 COMMAND "${Python3_EXECUTABLE}" "${MPS_DIR}/schema_stamp.py" "${MPS_CORE}/schema_fields.h" "${MPS_GENERATED}/schema_stamp.h"
 DEPENDS "${MPS_DIR}/schema_stamp.py" "${MPS_CORE}/schema_fields.h" VERBATIM)
add_custom_target(mps-schema-stamp DEPENDS "${MPS_GENERATED}/schema_stamp.h")
function(mpo_target name)
 add_executable(${name} "${MPS_DIR}/tests/tests.c" "${MPS_CORE}/schema.c" ${MPS_BASE} ${MPS_CODEC})
 add_dependencies(${name} mps-schema-stamp)
 target_include_directories(${name} PRIVATE "${MPS_DIR}/include" "${MPS_DIR}/../include" "${MPS_CORE}" "${MPS_GENERATED}" "${CMASTERY_API_DEPTH_ROOT}/architecture")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-schema)
foreach(case schema_wire schema_semantics schema_visit schema_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-schema ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPS_CASES schema_semantics schema_wire schema_visit schema_wire schema_wire schema_semantics schema_base schema_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-schema-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-schema-mutant-${i} PRIVATE MPS_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+4")
   target_compile_definitions(metaprog-schema-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPS_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPS_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-schema-mutant-${i}> ${case})
 endforeach()
endif()
