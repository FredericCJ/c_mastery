cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_build_cache C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPB_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPB_CORE "${MPB_DIR}/src")
set(MPB_BASE "${MPB_DIR}/../src/metaprog.c")
set(CMASTERY_API_DEPTH_ROOT "${MPB_DIR}/../../api_depth" CACHE PATH "Completed architecture workspace")
set(MPB_CODEC "${CMASTERY_API_DEPTH_ROOT}/architecture/learner.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPB_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
  set(MPB_CODEC "${CMASTERY_ASSESSOR_ROOT}/api_depth/architecture/core.c")
 else()
  set(MPB_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPB_CORE "${MPB_ASSESSOR}/build_cache")
 set(MPB_BASE "${MPB_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_API_DEPTH_ASSESSOR_ROOT)
 set(MPB_CODEC "${CMASTERY_API_DEPTH_ASSESSOR_ROOT}/architecture/core.c")
endif()
if(NOT EXISTS "${MPB_CODEC}")
 message(STATUS "build_cache requires an explicit completed api_depth codec.records workspace")
 return()
endif()
if(CMASTERY_MPB_LEARNER)
 set(MPB_CORE "${MPB_DIR}/src")
 set(MPB_BASE "${MPB_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPB_DIR}/tests/tests.c" "${MPB_CORE}/build_cache.c" ${MPB_BASE} ${MPB_CODEC})
 target_include_directories(${name} PRIVATE "${MPB_DIR}/include" "${MPB_DIR}/../include" "${MPB_CORE}" "${CMASTERY_API_DEPTH_ROOT}/architecture")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-build_cache)
find_package(Threads REQUIRED)
function(mpb_host_target name test)
 add_executable(${name} "${MPB_DIR}/tests/${test}.c" "${MPB_CORE}/build_cache.c" "${MPB_CORE}/cache_file.c" ${MPB_BASE} ${MPB_CODEC})
 target_include_directories(${name} PRIVATE "${MPB_DIR}/include" "${MPB_DIR}/../include" "${CMASTERY_API_DEPTH_ROOT}/architecture")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_libraries(${name} PRIVATE Threads::Threads)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()

mpb_host_target(metaprog-build_cache-parallel parallel)
add_test(NAME metaprog.cache_files COMMAND metaprog-build_cache-files)
add_test(NAME metaprog.cache_parallel COMMAND metaprog-build_cache-parallel)
foreach(case cache_inputs cache_schedule cache_publication cache_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-build_cache ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPB_CASES cache_inputs cache_inputs cache_schedule cache_schedule cache_publication cache_schedule cache_base cache_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-build_cache-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-build_cache-mutant-${i} PRIVATE MPB_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+12")
   target_compile_definitions(metaprog-build_cache-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPB_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPB_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-build_cache-mutant-${i}> ${case})
 endforeach()
 foreach(i RANGE 9 10)
  mpb_host_target(metaprog-build_cache-mutant-${i} files)
  math(EXPR defect "${i}-8")

  add_test(NAME metaprog.mutation.cache_files.${i} COMMAND "${Python3_EXECUTABLE}" "${MPB_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-build_cache-mutant-${i}> cache_files)
 endforeach()
endif()
