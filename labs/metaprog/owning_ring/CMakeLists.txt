cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_owning_ring C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPO_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPO_CORE "${MPO_DIR}/src")
set(MPO_BASE "${MPO_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPO_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPO_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPO_CORE "${MPO_ASSESSOR}/owning_ring")
 set(MPO_BASE "${MPO_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPO_LEARNER)
 set(MPO_CORE "${MPO_DIR}/src")
 set(MPO_BASE "${MPO_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPO_DIR}/tests/tests.c" "${MPO_CORE}/owning_ring.c" "${MPO_CORE}/erased.c" "${MPO_CORE}/manual.c" ${MPO_BASE})
 target_include_directories(${name} PRIVATE "${MPO_DIR}/include" "${MPO_DIR}/../include" "${MPO_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-owning_ring)
foreach(case mono_base owning_failure owning_transfer owning_erased)
 add_test(NAME metaprog.${case} COMMAND metaprog-owning_ring ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPO_CASES owning_failure owning_failure owning_transfer owning_erased owning_transfer owning_erased mono_base mono_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-owning-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-owning-mutant-${i} PRIVATE MPO_MUTATION=${i})
  else()
   math(EXPR baseline "${i}-6")
   target_compile_definitions(metaprog-owning-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPO_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPO_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-owning-mutant-${i}> ${case})
 endforeach()
endif()
