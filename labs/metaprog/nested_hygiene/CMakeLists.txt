cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_nested_hygiene C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPH_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPH_CORE "${MPH_DIR}/src")
set(MPH_BASE "${MPH_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPH_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPH_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPH_CORE "${MPH_ASSESSOR}/nested_hygiene")
 set(MPH_BASE "${MPH_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPH_LEARNER)
 set(MPH_CORE "${MPH_DIR}/src")
 set(MPH_BASE "${MPH_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPH_DIR}/tests/tests.c" "${MPH_CORE}/nested_hygiene.c" ${MPH_BASE})
 target_include_directories(${name} PRIVATE "${MPH_DIR}/include" "${MPH_DIR}/../include" "${MPH_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-nested_hygiene)
foreach(case macro_nested macro_budgets macro_capture macro_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-nested_hygiene ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPH_CASES macro_nested macro_nested macro_budgets macro_budgets macro_capture macro_capture macro_base macro_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-nested_hygiene-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-nested_hygiene-mutant-${i} PRIVATE MPH_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+10")
   target_compile_definitions(metaprog-nested_hygiene-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPH_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPH_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-nested_hygiene-mutant-${i}> ${case})
 endforeach()
endif()
