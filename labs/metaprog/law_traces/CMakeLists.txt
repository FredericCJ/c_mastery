cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(metaprog_law_traces C)
 set(CMAKE_C_STANDARD 17)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(CMAKE_C_EXTENSIONS OFF)
 enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MPT_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(MPT_CORE "${MPT_DIR}/src")
set(MPT_BASE "${MPT_DIR}/../src/metaprog.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/metaprog/metaprog.c")
  set(MPT_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/metaprog")
 else()
  set(MPT_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(MPT_CORE "${MPT_ASSESSOR}/law_traces")
 set(MPT_BASE "${MPT_ASSESSOR}/metaprog.c")
endif()
if(CMASTERY_MPT_LEARNER)
 set(MPT_CORE "${MPT_DIR}/src")
 set(MPT_BASE "${MPT_DIR}/../src/metaprog.c")
endif()
function(mpo_target name)
 add_executable(${name} "${MPT_DIR}/tests/tests.c" "${MPT_CORE}/law_traces.c" "${MPT_CORE}/fault_adapter.c" ${MPT_BASE})
 target_include_directories(${name} PRIVATE "${MPT_DIR}/include" "${MPT_DIR}/../include" "${MPT_CORE}")
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(CMASTERY_SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined -no-pie)
 endif()
endfunction()
mpo_target(metaprog-law_traces)
foreach(case law_owner law_canonical law_shrink law_base)
 add_test(NAME metaprog.${case} COMMAND metaprog-law_traces ${case})
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 set(MPT_CASES law_owner law_canonical law_canonical law_canonical law_shrink law_shrink law_base law_base)
 foreach(i RANGE 1 8)
  mpo_target(metaprog-law_traces-mutant-${i})
  if(i LESS 7)
   target_compile_definitions(metaprog-law_traces-mutant-${i} PRIVATE MPT_MUTATION=${i})
  else()
   math(EXPR baseline "${i}+6")
   target_compile_definitions(metaprog-law_traces-mutant-${i} PRIVATE MOD_MUTATION=${baseline})
  endif()
  math(EXPR index "${i}-1")
  list(GET MPT_CASES ${index} case)
  add_test(NAME metaprog.mutation.${case}.${i} COMMAND "${Python3_EXECUTABLE}" "${MPT_DIR}/expect_failure.py" $<TARGET_FILE:metaprog-law_traces-mutant-${i}> ${case})
 endforeach()
endif()
