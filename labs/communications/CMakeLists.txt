cmake_minimum_required(VERSION 3.20)
project(c_mastery_communications C)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit opt-in assessor/communications directory")
if(CMASTERY_ASSESSOR_ROOT)
  if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
    message(FATAL_ERROR "CMASTERY_ASSESSOR_ROOT must be absolute")
  endif()
  set(COM_SOURCE "${CMASTERY_ASSESSOR_ROOT}")
else()
  set(COM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/learner")
endif()
add_library(com_core STATIC "${COM_SOURCE}/http.c" "${COM_SOURCE}/edu7.c" "${COM_SOURCE}/net.c" "${COM_SOURCE}/mcu.c")
target_include_directories(com_core PUBLIC include)
target_compile_options(com_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(http_contracts tests/http.c)
target_link_libraries(http_contracts PRIVATE com_core)
foreach(kind headers framing chunks response io deadlines)
  add_test(NAME http_${kind} COMMAND http_contracts ${kind})
  set_tests_properties(http_${kind} PROPERTIES LABELS "host;communications;http")
endforeach()
if(UNIX)
  add_executable(com_server "${COM_SOURCE}/server.c")
  target_link_libraries(com_server PRIVATE com_core)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME http_socket COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/socket_client.py" $<TARGET_FILE:com_server>)
  set_tests_properties(http_socket PROPERTIES TIMEOUT 30 LABELS "host;communications;http;socket")
endif()

add_executable(edu7_contracts tests/edu7.c)
target_link_libraries(edu7_contracts PRIVATE com_core)
foreach(kind link router transport presentation session physical_model pipeline)
  add_test(NAME edu7_${kind} COMMAND edu7_contracts ${kind})
  set_tests_properties(edu7_${kind} PROPERTIES LABELS "host;communications;edu7")
endforeach()
add_executable(net_contracts tests/net.c)
target_link_libraries(net_contracts PRIVATE com_core)
foreach(kind arp packets sequence timers http zero_window icmp continue)
  add_test(NAME net_${kind} COMMAND net_contracts ${kind})
  set_tests_properties(net_${kind} PROPERTIES LABELS "host;communications;net")
endforeach()

add_executable(mcu_contracts tests/mcu.c)
target_link_libraries(mcu_contracts PRIVATE com_core)
foreach(kind spi dma)
  add_test(NAME mcu_${kind} COMMAND mcu_contracts ${kind})
  set_tests_properties(mcu_${kind} PROPERTIES LABELS "host;communications;hardware-model")
endforeach()

add_executable(com_campaign tests/campaign.c)
target_link_libraries(com_campaign PRIVATE com_core)
add_test(NAME communications_campaign COMMAND com_campaign)
set_tests_properties(communications_campaign PROPERTIES LABELS "host;communications;malformed-input")

# v4 additive implementation units
add_executable(v4_communications_tests "${COM_SOURCE}/v4_com.c" tests/v4_com.c)
target_include_directories(v4_communications_tests PRIVATE include)
target_compile_options(v4_communications_tests PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(v4_communications_tests PRIVATE com_core)
foreach(case reassembly arp tcp_fault canfd)
 add_test(NAME communications.v4_${case} COMMAND v4_communications_tests ${case})
 set_tests_properties(communications.v4_${case} PROPERTIES TIMEOUT 10)
endforeach()
if(CMASTERY_ASSESSOR_ROOT)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 include(tests/v4_mutations.cmake)
endif()
