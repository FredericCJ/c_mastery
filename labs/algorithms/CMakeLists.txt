cmake_minimum_required(VERSION 3.20)
project(cmastery_algorithms C ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/algorithms; empty builds learner source")
option(CMASTERY_SANITIZERS "Address and undefined behavior sanitizers for hosted C" OFF)
set(ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "CMASTERY_ASSESSOR_ROOT must be absolute")
 endif()
 set(ENGINE_DIR "${CMASTERY_ASSESSOR_ROOT}")
endif()
if(NOT EXISTS "${ENGINE_DIR}/algo.c")
 message(FATAL_ERROR "Selected implementation does not exist")
endif()
enable_testing()
function(algo_target name)
 add_executable(${name} "${ENGINE_DIR}/algo.c" tests/test_algo.c)
 target_include_directories(${name} PRIVATE include)
 if(UNIX)
  target_link_libraries(${name} PRIVATE m)
 endif()
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NOT CMASTERY_ASSESSOR_ROOT)
   target_compile_options(${name} PRIVATE -Wno-unused-parameter)
  endif()
  if(CMASTERY_SANITIZERS)
   target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
 endif()
endfunction()
algo_target(algorithms_tests)
if(UNIX)
 add_executable(algorithms_measure "${ENGINE_DIR}/algo.c" tests/measure.c)
 target_include_directories(algorithms_measure PRIVATE include)
 target_link_libraries(algorithms_measure PRIVATE m)
endif()
foreach(case ring hash avl sets graph sort heap match arena pool lifetime sum q7 numeric alu cache models)
 add_test(NAME algorithms.${case} COMMAND algorithms_tests ${case})
 set_tests_properties(algorithms.${case} PROPERTIES TIMEOUT 60)
endforeach()
if(UNIX AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
 target_sources(algorithms_tests PRIVATE "${ENGINE_DIR}/x86_64.S" tests/abi_x86_64.S adapters/checked_asm.c)
 target_compile_definitions(algorithms_tests PRIVATE ALGO_NATIVE_ASM=1)
 add_test(NAME algorithms.x86_64 COMMAND algorithms_tests assembly)
 foreach(case lower insert sift)
  add_test(NAME algorithms.x86_64_${case} COMMAND algorithms_tests assembly_${case})
 endforeach()
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutation SIGNED UNSTABLE ABI)
   algo_target(asm_mutant_${mutation})
   target_sources(asm_mutant_${mutation} PRIVATE "${ENGINE_DIR}/x86_64.S" tests/abi_x86_64.S adapters/checked_asm.c)
   target_compile_definitions(asm_mutant_${mutation} PRIVATE ALGO_NATIVE_ASM=1 ALGO_ASM_${mutation}=1)
   add_test(NAME algorithms.mutation_asm_${mutation} COMMAND asm_mutant_${mutation} assembly)
   set_tests_properties(algorithms.mutation_asm_${mutation} PROPERTIES WILL_FAIL TRUE TIMEOUT 30)
  endforeach()
 endif()
endif()
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutation TOMBSTONE STABILITY GENERATION)
  algo_target(c_mutant_${mutation})
  target_compile_definitions(c_mutant_${mutation} PRIVATE ALGO_MUTANT_${mutation}=1)
 endforeach()
 add_test(NAME algorithms.mutation_tombstone COMMAND c_mutant_TOMBSTONE hash)
 add_test(NAME algorithms.mutation_stability COMMAND c_mutant_STABILITY sort)
 add_test(NAME algorithms.mutation_generation COMMAND c_mutant_GENERATION lifetime)
 set_tests_properties(algorithms.mutation_tombstone algorithms.mutation_stability algorithms.mutation_generation PROPERTIES WILL_FAIL TRUE TIMEOUT 30)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
find_program(ARM_GCC arm-none-eabi-gcc)
find_program(QEMU_ARM qemu-system-arm)
if(Python3_FOUND AND ARM_GCC AND QEMU_ARM)
 set(THUMB_ARGS "")
 if(CMASTERY_ASSESSOR_ROOT)
  list(APPEND THUMB_ARGS --assessor-root "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 add_test(NAME algorithms.thumb COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/thumb/run.py" ${THUMB_ARGS})
 set_tests_properties(algorithms.thumb PROPERTIES TIMEOUT 60)
 foreach(case lower insert sift)
  add_test(NAME algorithms.thumb_${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/thumb/run.py" ${THUMB_ARGS} --case ${case})
  set_tests_properties(algorithms.thumb_${case} PROPERTIES TIMEOUT 60)
 endforeach()
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(mutation SIGNED UNSTABLE ABI)
   add_test(NAME algorithms.mutation_thumb_${mutation} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/thumb/run.py" ${THUMB_ARGS} --mutation ${mutation} --expect-rejection)
   set_tests_properties(algorithms.mutation_thumb_${mutation} PROPERTIES TIMEOUT 60)
  endforeach()
 endif()
endif()
