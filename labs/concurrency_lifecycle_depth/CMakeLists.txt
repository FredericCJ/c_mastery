cmake_minimum_required(VERSION 3.20)
project(concurrency_lifecycle_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit common or family assessor directory")
option(CMASTERY_SANITIZE "ASan and UBSan" OFF)
option(CMASTERY_TSAN "ThreadSanitizer" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
  if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/concurrency_lifecycle_depth/core.c")
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}/concurrency_lifecycle_depth")
  else()
    set(IMPL "${CMASTERY_ASSESSOR_ROOT}")
  endif()
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
  add_compile_options(-fsanitize=thread -g -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
function(program name kind)
  add_executable(${name} tests/${kind}.c "${IMPL}/core.c")
  target_include_directories(${name} PRIVATE include)
  if(kind STREQUAL "posix")
    target_sources(${name} PRIVATE "${IMPL}/posix.c")
    target_link_libraries(${name} PRIVATE Threads::Threads)
  endif()
endfunction()
program(cl-core core)
program(cl-posix posix)
program(cl-trace trace)
enable_testing()
function(anchor case kind first second)
  add_test(NAME concurrency_lifecycle_depth.${case} COMMAND cl-${kind} ${case})
  set_tests_properties(concurrency_lifecycle_depth.${case} PROPERTIES TIMEOUT 20)
  if(CMASTERY_ASSESSOR_ROOT)
    set(index 0)
    foreach(definition IN ITEMS "${first}" "${second}")
      math(EXPR index "${index}+1")
      set(name cl-${case}-${index})
      program(${name} ${kind})
      target_compile_definitions(${name} PRIVATE ${definition})
      add_test(NAME concurrency_lifecycle_depth.mutation.${case}.${index}
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py" $<TARGET_FILE:${name}> ${case})
    endforeach()
  endif()
endfunction()
anchor(queue core CL_MUTANT=1 CL_MUTANT=2)
anchor(pool core CL_MUTANT=3 CL_MUTANT=12)
anchor(life core CL_MUTANT=4 CL_MUTANT=9)
anchor(record core CL_MUTANT=5 CL_MUTANT=10)
anchor(ack core CL_MUTANT=6 CL_MUTANT=11)
anchor(threaded posix CL_MUTANT=1 CLP_MUTANT=1)
anchor(waiting posix CLP_MUTANT=1 CLP_MUTANT=2)
anchor(robust posix CLP_MUTANT=4 CLP_MUTANT=5)
add_test(NAME concurrency_lifecycle_depth.queue_model
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/model.py" $<TARGET_FILE:cl-trace>)
set_tests_properties(concurrency_lifecycle_depth.queue_model PROPERTIES TIMEOUT 30)
if(CMASTERY_ASSESSOR_ROOT)
  add_executable(cl-pointer-negative tests/pointer_copy_negative.c)
  add_test(NAME concurrency_lifecycle_depth.mutation.queue.pointer_copy
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py" $<TARGET_FILE:cl-pointer-negative>)
  program(cl-full core)
  target_compile_definitions(cl-full PRIVATE CL_MUTANT=7)
  add_test(NAME concurrency_lifecycle_depth.mutation.queue.overwrite
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py" $<TARGET_FILE:cl-full> queue)
  program(cl-destroy posix)
  target_compile_definitions(cl-destroy PRIVATE CLP_MUTANT=3)
  add_test(NAME concurrency_lifecycle_depth.mutation.waiting.early_destroy
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py" $<TARGET_FILE:cl-destroy> waiting)
endif()
