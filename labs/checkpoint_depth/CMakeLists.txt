cmake_minimum_required(VERSION 3.20)
project(checkpoint_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit own or common assessor root")
option(CMASTERY_SANITIZE "Host ASan and UBSan" OFF)
option(CMASTERY_TSAN "Host thread sanitizer" OFF)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/checkpoint_depth.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/checkpoint_depth/checkpoint_depth.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/checkpoint_depth/checkpoint_depth.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/checkpoint_depth.c")
 endif()
endif()
get_filename_component(IMPL_DIR "${IMPL}" DIRECTORY)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_compile_options(-ffp-contract=off -fno-fast-math)
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
 add_compile_options(-fno-slp-vectorize)
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -g -O0)
 add_link_options(-fsanitize=thread)
endif()
function(program target)
 add_executable(${target} tests/tests.c tests/memory.c tests/numeric.c tests/vector.c tests/writer.c tests/async.c tests/linux.c tests/operation.c tests/firmware.c "${IMPL}" "${IMPL_DIR}/checkpoint_memory.c" "${IMPL_DIR}/checkpoint_numeric.c" "${IMPL_DIR}/checkpoint_vector.c" "${IMPL_DIR}/checkpoint_writer.c" "${IMPL_DIR}/checkpoint_channel.c" "${IMPL_DIR}/checkpoint_async.c" "${IMPL_DIR}/checkpoint_linux.c" "${IMPL_DIR}/checkpoint_operation.c" "${IMPL_DIR}/checkpoint_firmware.c")
 target_include_directories(${target} PRIVATE include)
 target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
program(checkpoint-tests)
enable_testing()
add_test(NAME checkpoint_depth.codec COMMAND checkpoint-tests codec)
add_test(NAME checkpoint_depth.memory COMMAND checkpoint-tests memory)
add_test(NAME checkpoint_depth.ownership COMMAND checkpoint-tests ownership)
add_test(NAME checkpoint_depth.numeric COMMAND checkpoint-tests numeric)
add_test(NAME checkpoint_depth.numeric_policy COMMAND checkpoint-tests numeric_policy)
add_test(NAME checkpoint_depth.dispatch COMMAND checkpoint-tests dispatch)
add_test(NAME checkpoint_depth.vector COMMAND checkpoint-tests vector)
add_test(NAME checkpoint_depth.numeric_transaction COMMAND checkpoint-tests numeric_transaction)
add_test(NAME checkpoint_depth.writer COMMAND checkpoint-tests writer)
add_test(NAME checkpoint_depth.writer_ownership COMMAND checkpoint-tests writer_ownership)
add_test(NAME checkpoint_depth.async_admission COMMAND checkpoint-tests async_admission)
add_test(NAME checkpoint_depth.async_lifecycle COMMAND checkpoint-tests async_lifecycle)
add_test(NAME checkpoint_depth.async_wait COMMAND checkpoint-tests async_wait)
add_test(NAME checkpoint_depth.linux_reader COMMAND checkpoint-tests linux_reader)
add_test(NAME checkpoint_depth.operation COMMAND checkpoint-tests operation)
add_test(NAME checkpoint_depth.firmware_ring COMMAND checkpoint-tests firmware_ring)
add_test(NAME checkpoint_depth.firmware_tx COMMAND checkpoint-tests firmware_tx)
add_test(NAME checkpoint_depth.firmware_clock COMMAND checkpoint-tests firmware_clock)
add_test(NAME checkpoint_depth.stream COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_stream.py" $<TARGET_FILE:checkpoint-tests>)
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutant RANGE 1 38)
  program(checkpoint-mutant-${mutant})
  target_compile_definitions(checkpoint-mutant-${mutant} PRIVATE CP_MUTANT=${mutant})
  if(mutant LESS 3)
   set(case codec)
   set(variant ${mutant})
  elseif(mutant LESS 5)
   set(case stream)
   math(EXPR variant "${mutant}-2")
  elseif(mutant LESS 7)
   set(case memory)
   math(EXPR variant "${mutant}-4")
  elseif(mutant LESS 9)
   set(case ownership)
   math(EXPR variant "${mutant}-6")
  elseif(mutant LESS 11)
   set(case numeric)
   math(EXPR variant "${mutant}-8")
  elseif(mutant LESS 13)
   set(case numeric_policy)
   math(EXPR variant "${mutant}-10")
  elseif(mutant LESS 15)
   set(case dispatch)
   math(EXPR variant "${mutant}-12")
  elseif(mutant LESS 17)
   set(case vector)
   math(EXPR variant "${mutant}-14")
  elseif(mutant LESS 19)
   set(case numeric_transaction)
   math(EXPR variant "${mutant}-16")
  elseif(mutant LESS 21)
   set(case writer)
   math(EXPR variant "${mutant}-18")
  elseif(mutant LESS 23)
   set(case writer_ownership)
   math(EXPR variant "${mutant}-20")
  elseif(mutant LESS 25)
   set(case async_admission)
   math(EXPR variant "${mutant}-22")
  elseif(mutant LESS 27)
   set(case async_lifecycle)
   math(EXPR variant "${mutant}-24")
  elseif(mutant LESS 29)
   set(case async_wait)
   math(EXPR variant "${mutant}-26")
  elseif(mutant LESS 31)
   set(case linux_reader)
   math(EXPR variant "${mutant}-28")
  elseif(mutant LESS 33)
   set(case operation)
   math(EXPR variant "${mutant}-30")
  elseif(mutant LESS 35)
   set(case firmware_ring)
   math(EXPR variant "${mutant}-32")
  elseif(mutant LESS 37)
   set(case firmware_tx)
   math(EXPR variant "${mutant}-34")
  else()
   set(case firmware_clock)
   math(EXPR variant "${mutant}-36")
  endif()
  add_test(NAME checkpoint_depth.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py" $<TARGET_FILE:checkpoint-mutant-${mutant}> ${case})
 endforeach()
endif()
