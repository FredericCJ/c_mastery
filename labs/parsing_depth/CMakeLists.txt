cmake_minimum_required(VERSION 3.20)
project(parsing_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Common or family assessor root")
option(CMASTERY_SANITIZE "Address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
set(IMPL src/parsing.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/parsing_depth/regex.c")
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}/parsing_depth")
 else()
  set(PD_ASSESSOR "${CMASTERY_ASSESSOR_ROOT}")
 endif()
 set(IMPL "${PD_ASSESSOR}/regex.c" "${PD_ASSESSOR}/lr.c" "${PD_ASSESSOR}/pratt.c" "${PD_ASSESSOR}/scannerless.c")
endif()
function(pd_program target driver)
 add_executable(${target} ${driver} ${IMPL})
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
pd_program(parsing-depth-tests tests.c)
pd_program(parsing-depth-driver driver.c)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
set(PD_ORDINAL 0)
foreach(case regex lr pratt scannerless)
 math(EXPR PD_ORDINAL "${PD_ORDINAL}+1")
 add_test(NAME parsing_depth.${case} COMMAND parsing-depth-tests ${PD_ORDINAL})
 set_tests_properties(parsing_depth.${case} PROPERTIES TIMEOUT 30)
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant 1 2)
   math(EXPR mutation "(${PD_ORDINAL}-1)*2+${variant}")
   pd_program(parsing-depth-mutant-${mutation} tests.c)
   target_compile_definitions(parsing-depth-mutant-${mutation} PRIVATE PD_MUTANT=${mutation})
   add_test(NAME parsing_depth.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reject_mutant.py" $<TARGET_FILE:parsing-depth-mutant-${mutation}> ${PD_ORDINAL})
  endforeach()
 endif()
endforeach()
