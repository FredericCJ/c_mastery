cmake_minimum_required(VERSION 3.20)
project(c_mastery_coroutines C ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor family directory")
option(CMASTERY_SANITIZE "ASan/UBSan with fiber annotations" OFF)
option(CMASTERY_TSAN "TSan with fiber annotations" OFF)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "Separate ASan and TSan campaigns are required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -fcf-protection=none)
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
 add_compile_definitions(CO_ASAN)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=thread -no-pie)
 add_compile_definitions(CO_TSAN)
endif()
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/coroutines.c")
set(FIBER "${CMAKE_CURRENT_SOURCE_DIR}/src/fiber.c")
if(CMASTERY_ASSESSOR_ROOT)
 set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}")
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/coroutines/coroutines.c")
  set(ASSESSOR_FAMILY "${CMASTERY_ASSESSOR_ROOT}/coroutines")
 endif()
 set(IMPL "${ASSESSOR_FAMILY}/coroutines.c")
 set(FIBER "${ASSESSOR_FAMILY}/fiber.c" "${ASSESSOR_FAMILY}/fiber_x86_64.S")
endif()
add_library(coroutines STATIC "${IMPL}")
target_include_directories(coroutines PUBLIC include)
add_executable(coroutines-tests tests/tests.c)
target_link_libraries(coroutines-tests PRIVATE coroutines)
add_executable(fiber-tests tests/fiber.c ${FIBER})
target_include_directories(fiber-tests PRIVATE include)
target_link_libraries(fiber-tests PRIVATE m)
enable_testing()
add_test(NAME coroutines.fiber COMMAND fiber-tests)
foreach(case iterator dataflow generator fiber cps parser packrat async effect)
 if(NOT case STREQUAL "fiber")
  add_test(NAME coroutines.${case} COMMAND coroutines-tests ${case})
 endif()
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(number 1 2)
   set(mutation "CR_MUTATE_")
   string(TOUPPER ${case} upper)
   string(APPEND mutation ${upper})
   if(number EQUAL 2)
    string(APPEND mutation 2)
   endif()
   if(case STREQUAL "fiber")
    add_executable(mutant-${case}-${number} tests/fiber.c ${FIBER})
    target_link_libraries(mutant-${case}-${number} PRIVATE m)
    add_test(NAME coroutines.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}>)
   else()
    add_executable(mutant-${case}-${number} tests/tests.c "${IMPL}")
    add_test(NAME coroutines.mutation.${case}.${number} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject.py $<TARGET_FILE:mutant-${case}-${number}> ${case})
   endif()
   target_include_directories(mutant-${case}-${number} PRIVATE include)
   target_compile_definitions(mutant-${case}-${number} PRIVATE ${mutation})
  endforeach()
 endif()
endforeach()
