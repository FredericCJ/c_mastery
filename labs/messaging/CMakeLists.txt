cmake_minimum_required(VERSION 3.20)
project(c_mastery_messaging C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor root")
option(CMASTERY_SANITIZE "Hosted ASan/UBSan" OFF)
option(CMASTERY_TSAN "Separate hosted thread sanitizer" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/messaging.c")
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/messaging/messaging.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/messaging/messaging.c")
 else()
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/messaging.c")
 endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZE AND CMASTERY_TSAN)
 message(FATAL_ERROR "ASan and TSan are distinct campaigns")
endif()
if(CMASTERY_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(CMASTERY_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
add_executable(messaging-tests tests/tests.c "${IMPL}")
target_include_directories(messaging-tests PRIVATE include)
target_link_libraries(messaging-tests PRIVATE Threads::Threads)
add_executable(messaging-oracle-driver tests/oracle_driver.c "${IMPL}")
target_include_directories(messaging-oracle-driver PRIVATE include)
enable_testing()
set(index 0)
foreach(case nursery channels operations actors supervision stealing transactions snapshots sequences streams)
 add_test(NAME messaging.${case} COMMAND messaging-tests ${case})
 set_tests_properties(messaging.${case} PROPERTIES TIMEOUT 30)
 math(EXPR index "${index}+1")
 if(CMASTERY_ASSESSOR_ROOT)
  foreach(variant RANGE 0 1)
   math(EXPR mutant "${index}+10*${variant}")
   add_executable(messaging-mutant-${mutant} tests/tests.c "${IMPL}")
   target_include_directories(messaging-mutant-${mutant} PRIVATE include)
   target_link_libraries(messaging-mutant-${mutant} PRIVATE Threads::Threads)
   target_compile_definitions(messaging-mutant-${mutant} PRIVATE SC_MUTANT=${mutant})
   add_test(NAME messaging.mutation.${case}.${variant} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_mutant.py" $<TARGET_FILE:messaging-mutant-${mutant}> ${case})
  endforeach()
 endif()
endforeach()

add_test(NAME messaging.threads COMMAND messaging-tests threads)
set_tests_properties(messaging.threads PROPERTIES TIMEOUT 60)
