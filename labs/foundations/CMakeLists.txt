cmake_minimum_required(VERSION 3.20)
project(c_mastery_foundations C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/foundations directory")
option(CMASTERY_MEMORY_DRILLS "Build and execute isolated ASan/LSan fault controls" OFF)
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "CMASTERY_ASSESSOR_ROOT must be absolute")
 endif()
 set(IMPL "${CMASTERY_ASSESSOR_ROOT}/foundation.c")
else()
 set(IMPL "${CMAKE_CURRENT_SOURCE_DIR}/src/foundation.c")
endif()
add_library(foundation STATIC "${IMPL}")
target_include_directories(foundation PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(foundation PRIVATE -Wall -Wextra -Wpedantic)
add_executable(foundation-tests tests/tests.c)
target_link_libraries(foundation-tests PRIVATE foundation)
add_executable(foundation-benchmark src/benchmark.c)
target_link_libraries(foundation-benchmark PRIVATE foundation)
add_executable(foundation-file-io src/file_io.c)
target_link_libraries(foundation-file-io PRIVATE foundation)
add_executable(foundation-instrumentation src/instrumentation.c)
target_compile_options(foundation-instrumentation PRIVATE -finstrument-functions -fno-inline)
add_executable(foundation-language src/language_probe.c)
install(TARGETS foundation ARCHIVE DESTINATION lib)
install(FILES include/foundation.h DESTINATION include)
foreach(flavor faulty repaired)
 add_executable(admission-${flavor} drills/admission.c)
 target_compile_options(admission-${flavor} PRIVATE -O0 -g)
 if(flavor STREQUAL repaired)
  target_compile_definitions(admission-${flavor} PRIVATE REPAIRED=1)
 endif()
endforeach()
enable_testing()
foreach(case prefix size cursor owner parse sort properties io async cleanup packet)
 add_test(NAME foundations.${case} COMMAND foundation-tests ${case})
 set_tests_properties(foundations.${case} PROPERTIES TIMEOUT 5)
endforeach()
add_test(NAME foundations.compile COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/negative_compile.py --cc ${CMAKE_C_COMPILER})
add_test(NAME foundations.admission COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/admission_observation.py --faulty $<TARGET_FILE:admission-faulty> --repaired $<TARGET_FILE:admission-repaired>)
add_test(NAME foundations.reducer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/drills/reduce.py $<TARGET_FILE:admission-faulty> NNANNAN)
add_test(NAME foundations.instrumentation COMMAND foundation-instrumentation)
add_test(NAME foundations.language COMMAND foundation-language)
if(CMASTERY_ASSESSOR_ROOT)
 foreach(mutation size stability)
  add_executable(foundation-mutant-${mutation} tests/tests.c "${IMPL}")
  target_include_directories(foundation-mutant-${mutation} PRIVATE include)
  string(TOUPPER ${mutation} upper)
  target_compile_definitions(foundation-mutant-${mutation} PRIVATE F_MUTATE_${upper})
  if(mutation STREQUAL size)
   set(test_case size)
  else()
   set(test_case sort)
  endif()
  add_test(NAME foundations.mutation.${mutation} COMMAND foundation-mutant-${mutation} ${test_case})
  set_tests_properties(foundations.mutation.${mutation} PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
 endforeach()
endif()
if(CMASTERY_MEMORY_DRILLS)
 foreach(flavor faulty repaired)
  add_executable(memory-${flavor} drills/memory_faults.c)
  target_compile_options(memory-${flavor} PRIVATE -O0 -g -fsanitize=address -fno-omit-frame-pointer)
  target_link_options(memory-${flavor} PRIVATE -fsanitize=address)
  if(flavor STREQUAL repaired)
   target_compile_definitions(memory-${flavor} PRIVATE REPAIRED=1)
  endif()
 endforeach()
 add_test(NAME foundations.memory COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/memory_observation.py --faulty $<TARGET_FILE:memory-faulty> --repaired $<TARGET_FILE:memory-repaired>)
 set_tests_properties(foundations.memory PROPERTIES TIMEOUT 30)
endif()
