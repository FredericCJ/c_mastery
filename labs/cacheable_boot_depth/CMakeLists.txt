cmake_minimum_required(VERSION 3.16)
project(cacheable_boot_depth C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
set(IMPL src/boot.c)
if(CMASTERY_ASSESSOR_ROOT)
 if(EXISTS "${CMASTERY_ASSESSOR_ROOT}/cacheable_boot_depth/boot.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/cacheable_boot_depth/boot.c")
 elseif(EXISTS "${CMASTERY_ASSESSOR_ROOT}/boot.c")
  set(IMPL "${CMASTERY_ASSESSOR_ROOT}/boot.c")
 else()
  message(FATAL_ERROR "Explicit completed cacheable_boot_depth source missing")
 endif()
endif()
if(NOT CB_PROVIDER_SOURCE)
 set(CB_PROVIDER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/imports")
endif()
foreach(NAME selected_startup h745_cache)
 if(NOT EXISTS "${CB_PROVIDER_SOURCE}/${NAME}.c")
  message(FATAL_ERROR "Explicit prerequisite source missing: ${NAME}")
 endif()
endforeach()
add_library(cold_provider OBJECT "${CB_PROVIDER_SOURCE}/selected_startup.c" "${CB_PROVIDER_SOURCE}/h745_cache.c")
target_include_directories(cold_provider PRIVATE imports)
add_executable(cacheable_boot ${IMPL} model.c tests.c $<TARGET_OBJECTS:cold_provider>)
target_include_directories(cacheable_boot PRIVATE . imports)
if(NOT MSVC)
 target_compile_options(cacheable_boot PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_compile_options(cold_provider PRIVATE -Wall -Wextra -Wpedantic -Werror)
 # The retained provider already documents this exact diagnostic exception.
 set_source_files_properties("${CB_PROVIDER_SOURCE}/selected_startup.c" PROPERTIES COMPILE_OPTIONS "-Wno-misleading-indentation")
endif()
add_test(NAME cacheable_boot.cold COMMAND cacheable_boot)
