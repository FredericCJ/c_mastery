cmake_minimum_required(VERSION 3.20)
project(cmastery_execution C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMASTERY_ASSESSOR_ROOT "" CACHE PATH "Explicit assessor/execution directory")
set(CMASTERY_TARGET "host" CACHE STRING "host,cortex-m4-qemu,stm32f439,stm32h745")
set(CMASTERY_CPUS 1 CACHE STRING "Target CPU count (H745 AMP has two images)")
option(CMASTERY_SANITIZERS "Hosted address/undefined sanitizers" OFF)
set(EXEC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMASTERY_ASSESSOR_ROOT)
 if(NOT IS_ABSOLUTE "${CMASTERY_ASSESSOR_ROOT}")
  message(FATAL_ERROR "Assessor directory must be absolute")
 endif()
 set(EXEC_SOURCE "${CMASTERY_ASSESSOR_ROOT}")
endif()
enable_testing()
add_library(cmx STATIC "${EXEC_SOURCE}/cmx.c" "${EXEC_SOURCE}/concurrent.c" "${EXEC_SOURCE}/advanced.c")
target_include_directories(cmx PUBLIC include)
target_compile_options(cmx PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(NOT CMASTERY_ASSESSOR_ROOT)
 target_compile_options(cmx PRIVATE -Wno-unused-parameter)
endif()
add_executable(execution_tests tests/contracts.c)
target_link_libraries(execution_tests PRIVATE cmx)
target_compile_options(execution_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZERS)
 target_compile_options(cmx PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_compile_options(execution_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(execution_tests PRIVATE -fsanitize=address,undefined)
endif()
foreach(case admission enrollment lifecycle scheduler timers semaphore mutex inheritance ceiling condition queue events notification barrier once rwlock teardown faults response edf cbs spsc hazard collector_aba epoch)
 add_test(NAME execution.${case} COMMAND execution_tests ${case})
 set_tests_properties(execution.${case} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(execution_advanced tests/advanced.c)
target_link_libraries(execution_advanced PRIVATE cmx)
target_compile_options(execution_advanced PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMASTERY_SANITIZERS)
 target_compile_options(execution_advanced PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(execution_advanced PRIVATE -fsanitize=address,undefined)
endif()
foreach(case priorities partition server identity progress history rcu comparison)
 add_test(NAME execution.${case} COMMAND execution_advanced ${case})
 set_tests_properties(execution.${case} PROPERTIES TIMEOUT 15)
endforeach()
find_package(Threads REQUIRED)
add_executable(execution_parallel tests/parallel.c)
target_link_libraries(execution_parallel PRIVATE cmx Threads::Threads)
if(CMASTERY_SANITIZERS)
 target_compile_options(execution_parallel PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(execution_parallel PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME execution.parallel COMMAND execution_parallel)
set_tests_properties(execution.parallel PROPERTIES TIMEOUT 60)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(CMASTERY_ASSESSOR_ROOT)
 add_test(NAME execution.controls COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/tests/controls.py" --source "${EXEC_SOURCE}"
  --lab "${CMAKE_CURRENT_SOURCE_DIR}" --build "${CMAKE_CURRENT_BINARY_DIR}/controls"
  --cc "${CMAKE_C_COMPILER}")
 set_tests_properties(execution.controls PROPERTIES TIMEOUT 100 LABELS "mutation;host;emulator")
endif()
find_program(CMX_ARM arm-none-eabi-gcc)
find_program(CMX_QEMU qemu-system-arm)
if(CMX_ARM)
 foreach(board cortex-m4-qemu stm32f439 stm32h745)
  add_test(NAME execution.image_${board} COMMAND ${Python3_EXECUTABLE}
   "${CMAKE_CURRENT_SOURCE_DIR}/targets/build.py" --target ${board}
   --source "${EXEC_SOURCE}" --lab "${CMAKE_CURRENT_SOURCE_DIR}"
   --build "${CMAKE_CURRENT_BINARY_DIR}/targets/${board}")
  set_tests_properties(execution.image_${board} PROPERTIES TIMEOUT 60 LABELS "cross;image-audit")
 endforeach()
 if(CMX_QEMU)
  add_test(NAME execution.cortex_m4 COMMAND ${Python3_EXECUTABLE}
   "${CMAKE_CURRENT_SOURCE_DIR}/targets/build.py" --target cortex-m4-qemu --run
   --source "${EXEC_SOURCE}" --lab "${CMAKE_CURRENT_SOURCE_DIR}"
   --build "${CMAKE_CURRENT_BINARY_DIR}/targets/qemu-run")
  set_tests_properties(execution.cortex_m4 PROPERTIES TIMEOUT 60 LABELS "qemu;actual-context-switch")
  foreach(stage boot cooperative preemptive fp)
   add_test(NAME execution.cortex_m4_${stage} COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/targets/build.py" --target cortex-m4-qemu --run --stage ${stage}
    --source "${EXEC_SOURCE}" --lab "${CMAKE_CURRENT_SOURCE_DIR}"
    --build "${CMAKE_CURRENT_BINARY_DIR}/targets/qemu-${stage}")
   set_tests_properties(execution.cortex_m4_${stage} PROPERTIES TIMEOUT 60 LABELS "qemu;stage")
  endforeach()
 endif()
endif()
